/**
 * WorkdaySummaryService — podsumowanie czasu pracy (miesiąc) dla pracownika.
 *
 * Źródło: Workday_Log
 * - Miesiąc liczony wg CZASU URZĄDZENIA (tzOffsetMinutes = Date.getTimezoneOffset()).
 * - Sumujemy DURATION_SEC dla wierszy z danego miesiąca (wg START_AT).
 * - Jeśli DURATION_SEC brak / 0, to liczymy fallback z START_AT -> STOP (AUTO_CLOSE_AT > END_SCAN_AT > END_AT).
 */
function WorkdaySummary_getMonth_(worker, year, month1to12, tzOffsetMinutes) {
  const login = String(worker && worker.login || '').trim();
  const y = Number(year || 0);
  const m = Number(month1to12 || 0);
  const off = Number(tzOffsetMinutes || 0);

  if (!login || !y || !m) return { year: y, month: m, totalSec: 0 };

  const monthKey = String(y) + '-' + String(m).padStart(2, '0');

  const sh = Workday_sheet_();
  const h = Sheets_headerMap_(sh);
  const H = CONFIG.HEADERS.WORKDAY_LOG;

  const colLogin = h[H.WORKER_LOGIN];
  const colStart = h[H.START_AT];
  const colDur = h[H.DURATION_SEC];
  const colAuto = h[H.AUTO_CLOSE_AT];
  const colEndScan = h[H.END_SCAN_AT];
  const colEndAt = h[H.END_AT];

  if (!colLogin || !colStart) return { year: y, month: m, totalSec: 0 };

  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow < 2) return { year: y, month: m, totalSec: 0 };

  const values = sh.getRange(2, 1, lastRow - 1, lastCol).getValues();

  let total = 0;

  for (let i = 0; i < values.length; i++) {
    const row = values[i] || [];
    const rLogin = String(row[colLogin - 1] || '').trim();
    if (rLogin !== login) continue;

    const startAt = String(row[colStart - 1] || '').trim();
    if (!startAt) continue;

    const dateKey = localDateKey_(startAt, off);
    if (!dateKey || dateKey.slice(0, 7) !== monthKey) continue;

    // Prefer stored duration (already accounts for pauses)
    const dur = Number(colDur ? row[colDur - 1] : 0) || 0;
    if (dur > 0) { total += dur; continue; }

    // Fallback: compute by timestamps (without pause subtraction)
    const autoCloseAt = String(colAuto ? row[colAuto - 1] : '').trim();
    const endScanAt = String(colEndScan ? row[colEndScan - 1] : '').trim();
    const endAt = String(colEndAt ? row[colEndAt - 1] : '').trim();
    const stopAt = WorkdaySummary_pickStopAt_(autoCloseAt, endScanAt, endAt);
    if (!stopAt) continue;

    const sec = secondsBetweenIso_(startAt, stopAt);
    if (sec > 0) total += sec;
  }

  return { year: y, month: m, totalSec: Math.round(total) };
}

function WorkdaySummary_pickStopAt_(autoCloseAt, endScanAt, endAt) {
  const a = String(autoCloseAt || '').trim();
  if (a) return a;
  const s = String(endScanAt || '').trim();
  if (s) return s;
  const e = String(endAt || '').trim();
  if (e) return e;
  return '';
}

/**
 * Lista dni w miesiącu z sumą czasu pracy (tylko START/STOP Workday_Log).
 * Miesiąc liczony wg CZASU URZĄDZENIA (tzOffsetMinutes).
 */
function WorkdaySummary_getMonthDays_(worker, year, month1to12, tzOffsetMinutes) {
  const login = String(worker && worker.login || '').trim();
  const y = Number(year || 0);
  const m = Number(month1to12 || 0);
  const off = Number(tzOffsetMinutes || 0);

  if (!login || !y || !m) return { year: y, month: m, days: [] };

  const monthKey = String(y) + '-' + String(m).padStart(2, '0');

  const sh = Workday_sheet_();
  const h = Sheets_headerMap_(sh);
  const H = CONFIG.HEADERS.WORKDAY_LOG;

  const colLogin = h[H.WORKER_LOGIN];
  const colStart = h[H.START_AT];
  const colDur = h[H.DURATION_SEC];
  const colAuto = h[H.AUTO_CLOSE_AT];
  const colEndScan = h[H.END_SCAN_AT];
  const colEndAt = h[H.END_AT];

  if (!colLogin || !colStart) return { year: y, month: m, days: [] };

  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow < 2) return { year: y, month: m, days: [] };

  const values = sh.getRange(2, 1, lastRow - 1, lastCol).getValues();

  const map = {}; // dateKey -> totalSec

  for (let i = 0; i < values.length; i++) {
    const row = values[i] || [];
    const rLogin = String(row[colLogin - 1] || '').trim();
    if (rLogin !== login) continue;

    const startAt = String(row[colStart - 1] || '').trim();
    if (!startAt) continue;

    const dateKey = localDateKey_(startAt, off);
    if (!dateKey || dateKey.slice(0, 7) !== monthKey) continue;

    const dur = Number(colDur ? row[colDur - 1] : 0) || 0;
    let sec = 0;

    if (dur > 0) {
      sec = dur;
    } else {
      const autoCloseAt = String(colAuto ? row[colAuto - 1] : '').trim();
      const endScanAt = String(colEndScan ? row[colEndScan - 1] : '').trim();
      const endAt = String(colEndAt ? row[colEndAt - 1] : '').trim();
      const stopAt = WorkdaySummary_pickStopAt_(autoCloseAt, endScanAt, endAt);
      if (stopAt) sec = secondsBetweenIso_(startAt, stopAt);
    }

    if (sec > 0) {
      map[dateKey] = Number(map[dateKey] || 0) + Number(sec || 0);
    }
  }

  // Sort ascending (01..31) like on the screenshot
  const days = Object.keys(map)
    .sort()
    .map(dk => ({ dateKey: dk, totalSec: Math.round(map[dk] || 0) }))
    .filter(d => (d.totalSec || 0) > 0);

  return { year: y, month: m, days };
}
