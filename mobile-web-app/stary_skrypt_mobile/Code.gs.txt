/**
 * Best Clean Timer — Web App (Google Apps Script)
 * Entry points:
 *  - doGet: serves the mobile web UI
 *  - doPost: serves JSON API requests
 *
 * Deployment: Deploy as Web App and use the /exec URL on phones.
 */
function doGet(e) {
  return HtmlService.createTemplateFromFile('Index')
    .evaluate()
    .setTitle('Best Clean')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1, viewport-fit=cover');
}

function doPost(e) {
  return Router_handleApiRequest_(e);
}

/** HTML include helper */
function include(filename) {
  return HtmlService.createHtmlOutputFromFile(filename).getContent();
}
