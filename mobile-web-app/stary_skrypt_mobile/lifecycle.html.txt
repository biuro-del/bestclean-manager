<script>
window.BC = window.BC || {};
BC.Lifecycle = (function() {
  // Cel: wyeliminować sytuację "biały ekran" po dłuższym czasie / uśpieniu karty.
  // Działanie:
  // 1) Jeśli wracasz do aplikacji po dłuższej przerwie -> automatyczny reload (1x).
  // 2) Jeśli strona po starcie wygląda na "pustą" / DOM niepoprawny -> automatyczny reload (1x).
  // 3) Jeśli boot overlay wisi zbyt długo -> automatyczny reload (1x).

  const RELOAD_AFTER_MS = 30 * 60 * 1000; // 30 minut
  const WATCHDOG_MS = 8000;               // 8s
  const QUICK_CHECK_MS = 2000;            // 2s

  const LS_LAST_VISIBLE = 'bc_last_visible_at';
  const SS_RELOADED = 'bc_autoreload_once';

  function nowMs_() { return Date.now(); }

  function alreadyReloaded_() {
    try { return sessionStorage.getItem(SS_RELOADED) === '1'; } catch (e) { return false; }
  }

  function markReloaded_() {
    try { sessionStorage.setItem(SS_RELOADED, '1'); } catch (e) {}
  }

  function safeReload_(reason) {
    if (alreadyReloaded_()) return;
    markReloaded_();
    try { console.warn('[Lifecycle] Auto reload:', reason); } catch (e) {}
    try { location.reload(); } catch (e) {}
  }

  function getLastVisible_() {
    try {
      const v = localStorage.getItem(LS_LAST_VISIBLE);
      const n = Number(v);
      return Number.isFinite(n) ? n : 0;
    } catch (e) {
      return 0;
    }
  }

  function setLastVisible_(ts) {
    try { localStorage.setItem(LS_LAST_VISIBLE, String(ts)); } catch (e) {}
  }

  function checkStale_(reason) {
    const last = getLastVisible_();
    const delta = last ? (nowMs_() - last) : 0;
    if (last && delta > RELOAD_AFTER_MS) {
      safeReload_(reason + ' (' + Math.round(delta / 60000) + ' min)');
    } else {
      setLastVisible_(nowMs_());
    }
  }

  function looksBlank_() {
    // "Blank" heurystyka: brak głównych elementów lub brak treści (częsty przypadek po uśpieniu / discard tab).
    const shell = document.querySelector('.app-shell');
    const header = document.querySelector('.header');
    const main = document.querySelector('.main');
    const boot = document.getElementById('bootOverlay');

    if (!shell || !header || !main || !boot) return true;

    // Jeśli wszystko ma 0 wysokości (czasem zdarza się po przywróceniu karty)
    try {
      const h = shell.getBoundingClientRect().height;
      if (!h || h < 100) return true;
    } catch (e) {}

    // Jeśli kompletnie brak tekstu na stronie
    try {
      const t = (document.body && document.body.innerText) ? document.body.innerText.trim() : '';
      if (!t) return true;
    } catch (e) {}

    return false;
  }

  function healthCheck_(reason) {
    if (looksBlank_()) return safeReload_('blank: ' + reason);

    // Boot overlay stuck
    const boot = document.getElementById('bootOverlay');
    if (boot && !boot.classList.contains('hidden')) {
      // zostawiamy chwilę na bootstrap; watchdog poniżej zrobi reload po WATCHDOG_MS
    }
  }

  // Zapisz "jestem widoczny" od razu
  setLastVisible_(nowMs_());

  document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'visible') {
      // Po powrocie do karty: jeśli przerwa była długa -> reload
      checkStale_('resume');
      // Dodatkowo szybki check "czy nie jest biało"
      setTimeout(() => healthCheck_('resume'), 200);
    } else {
      setLastVisible_(nowMs_());
    }
  });

  window.addEventListener('focus', () => {
    checkStale_('focus');
    setTimeout(() => healthCheck_('focus'), 200);
  });

  // BFCache: jeśli persisted -> od razu reload (to często rozwiązuje biały ekran w mobile)
  window.addEventListener('pageshow', (e) => {
    if (e && e.persisted) return safeReload_('bfcache');
    checkStale_('pageshow');
    setTimeout(() => healthCheck_('pageshow'), 200);
  });

  // Szybki check po starcie
  setTimeout(() => healthCheck_('startup'), QUICK_CHECK_MS);

  // Boot watchdog: jeśli boot overlay nadal wisi po WATCHDOG_MS -> reload
  window.addEventListener('load', () => {
    setTimeout(() => {
      const boot = document.getElementById('bootOverlay');
      if (!boot) return safeReload_('missing bootOverlay');
      if (!boot.classList.contains('hidden')) safeReload_('boot watchdog');
      // nawet jeśli boot schowany, ale DOM wygląda na pusty -> reload
      healthCheck_('load');
    }, WATCHDOG_MS);
  });

  return {};
})();
</script>
