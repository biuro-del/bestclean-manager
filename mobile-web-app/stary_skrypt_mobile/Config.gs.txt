/**
 * WEB_Config (opcjonalny) jako "źródło prawdy" dla nagłówków.
 *
 * Arkusz: WEB_Config
 * Kolumny (rekomendowane): Module | Key | Value
 *
 * Przykłady (proste):
 *  Workday_Log | StartAtHeader | StartAt
 *  Workday_Log | StopHeader    | EndAt
 *  Workday_Log | StopAltHeader | AutoCloseAt
 *  Backup_Log  | StartAtHeader | StartAt
 *  Backup_Log  | StopHeader    | EndAt
 *  Backup_Log  | Notes         | Comment
 *  Backup_Log  | ZoneQrHeader  | RoomID
 *
 * Przykłady (pełna ścieżka):
 *  (Module puste) | HEADERS.BACKUP_LOG.START_AT | StartAt
 */
var __CONFIG_INIT_DONE = false;

function Config_init_() {
  if (__CONFIG_INIT_DONE) return;
  __CONFIG_INIT_DONE = true;

  try {
    const overrides = Config_loadWebConfigOverrides_();
    Config_applyWebConfigOverrides_(overrides);
  } catch (err) {
    // non-fatal; fallback to hardcoded CONFIG
    try {
      Logger.log('WEB_Config load failed: ' + (err && err.message ? err.message : String(err)));
    } catch (_) {}
  }
}

function Config_loadWebConfigOverrides_() {
  const cache = CacheService.getScriptCache();
  const cacheKey = 'webcfg:v1';
  const cached = cache.get(cacheKey);
  if (cached) {
    try { return JSON.parse(cached); } catch (_) {}
  }

  const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
  const sh = ss.getSheetByName(CONFIG.WEB_CONFIG_SHEET_NAME);
  if (!sh) {
    cache.put(cacheKey, JSON.stringify({}), CONFIG.WEB_CONFIG_CACHE_SECONDS);
    return {};
  }

  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow < 2 || lastCol < 2) {
    cache.put(cacheKey, JSON.stringify({}), CONFIG.WEB_CONFIG_CACHE_SECONDS);
    return {};
  }

  const values = sh.getRange(1, 1, lastRow, lastCol).getValues();
  const headerRow = values[0].map(v => String(v || '').trim().toLowerCase());

  const colModule = Config_pickCol_(headerRow, ['module', 'moduł', 'sekcja', 'sheet', 'zakładka', 'tab']);
  const colKey = Config_pickCol_(headerRow, ['key', 'param', 'property', 'nazwa', 'name']);
  const colValue = Config_pickCol_(headerRow, ['value', 'wartość', 'val']);

  const hasHeader = (colModule !== null && colKey !== null && colValue !== null);
  const startIndex = hasHeader ? 1 : 0;

  const overrides = {};

  for (let r = startIndex; r < values.length; r++) {
    const row = values[r];
    const moduleRaw = hasHeader ? row[colModule] : row[0];
    const keyRaw = hasHeader ? row[colKey] : row[1];
    const valueRaw = hasHeader ? row[colValue] : row[2];

    const module = String(moduleRaw || '').trim();
    const key = String(keyRaw || '').trim();
    const value = String(valueRaw || '').trim();

    if (!key || value === '') continue;

    const path = Config_translateWebConfigKey_(module, key);
    if (!path) continue;

    overrides[path] = value;
  }

  cache.put(cacheKey, JSON.stringify(overrides), CONFIG.WEB_CONFIG_CACHE_SECONDS);
  return overrides;
}

function Config_applyWebConfigOverrides_(overrides) {
  if (!overrides) return;
  Object.keys(overrides).forEach((path) => {
    try { Config_setByPath_(path, overrides[path]); } catch (_) {}
  });
}

function Config_translateWebConfigKey_(moduleRaw, keyRaw) {
  const key = String(keyRaw || '').trim();
  const module = String(moduleRaw || '').trim();

  // Direct paths supported
  if (/^HEADERS\./i.test(key)) return key;

  const m = module.toUpperCase().replace(/[^A-Z0-9]+/g, '_');
  const k = key.toUpperCase().replace(/[^A-Z0-9]+/g, '');

  // WORKDAY_LOG
  if (m.indexOf('WORKDAY') >= 0) {
    if (k === 'STARTATHEADER' || k === 'STARTHEADER') return 'HEADERS.WORKDAY_LOG.START_AT';
    if (k === 'STOPHEADER' || k === 'ENDATHEADER') return 'HEADERS.WORKDAY_LOG.END_AT';
    if (k === 'STOPALTHEADER' || k === 'AUTOCLOSEHEADER') return 'HEADERS.WORKDAY_LOG.AUTO_CLOSE_AT';
    if (k === 'WORKERLOGINHEADER') return 'HEADERS.WORKDAY_LOG.WORKER_LOGIN';
    if (k === 'WORKERNAMEHEADER') return 'HEADERS.WORKDAY_LOG.WORKER_NAME';
    if (k === 'UTILITYROOMIDHEADER') return 'HEADERS.WORKDAY_LOG.UTILITY_ROOM_ID';
    if (k === 'STATUSHEADER') return 'HEADERS.WORKDAY_LOG.STATUS';
    return null;
  }

  // BACKUP_LOG / CYCLES / EVENTS
  if (m.indexOf('BACKUP') >= 0 || m.indexOf('CYCLE') >= 0 || m.indexOf('ZDARZEN') >= 0) {
    if (k === 'STARTATHEADER' || k === 'STARTHEADER') return 'HEADERS.BACKUP_LOG.START_AT';
    if (k === 'STOPHEADER' || k === 'ENDATHEADER') return 'HEADERS.BACKUP_LOG.END_AT';
    if (k === 'NOTES' || k === 'COMMENT' || k === 'COMMENTHEADER') return 'HEADERS.BACKUP_LOG.COMMENT';
    if (k === 'ZONEQRHEADER' || k === 'ROOMIDHEADER' || k === 'QRHEADER') return 'HEADERS.BACKUP_LOG.ROOM_ID';
    if (k === 'ZONEHEADER') return 'HEADERS.BACKUP_LOG.ZONE';
    if (k === 'ROOMHEADER') return 'HEADERS.BACKUP_LOG.ROOM';
    if (k === 'STATUSHEADER') return 'HEADERS.BACKUP_LOG.STATUS';
    return null;
  }

  // WORKERS
  if (m.indexOf('WORKER') >= 0 || m.indexOf('PRACOWN') >= 0) {
    if (k === 'LOGINHEADER') return 'HEADERS.WORKERS.LOGIN';
    if (k === 'PASSWORDHEADER' || k === 'HASLOHEADER') return 'HEADERS.WORKERS.PASSWORD';
    if (k === 'NAMEHEADER' || k === 'IMIEINAZWISKOHEADER') return 'HEADERS.WORKERS.NAME';
    if (k === 'ACTIVEHEADER') return 'HEADERS.WORKERS.ACTIVE';
    if (k === 'WORKERIDHEADER' || k === 'IDHEADER') return 'HEADERS.WORKERS.WORKER_ID';
    return null;
  }

  // ZONES
  if (m.indexOf('ZONE') >= 0 || m.indexOf('STREF') >= 0) {
    if (k === 'ROOMIDHEADER' || k === 'IDHEADER' || k === 'QRHEADER') return 'HEADERS.ZONES.ROOM_ID';
    if (k === 'ZONEHEADER' || k === 'STREFAHEADER') return 'HEADERS.ZONES.ZONE';
    if (k === 'ROOMHEADER' || k === 'POMIESZCZENIEHEADER') return 'HEADERS.ZONES.ROOM';
    if (k === 'FUNCHEADER' || k === 'FUNKCJAHEADER' || k === 'FUNCTIONHEADER') return 'HEADERS.ZONES.FUNC';
    return null;
  }

  return null;
}

function Config_setByPath_(path, value) {
  if (!path) return false;
  const parts = String(path).trim().split('.').filter(Boolean);
  if (!parts.length) return false;

  let obj = CONFIG;
  for (let i = 0; i < parts.length - 1; i++) {
    const p = parts[i];
    if (!(p in obj)) return false;
    obj = obj[p];
    if (!obj) return false;
  }

  const last = parts[parts.length - 1];
  if (!(last in obj)) return false;

  obj[last] = String(value);
  return true;
}

function Config_pickCol_(headerRow, candidates) {
  if (!headerRow || !headerRow.length) return null;
  for (let i = 0; i < candidates.length; i++) {
    const idx = headerRow.indexOf(String(candidates[i] || '').toLowerCase());
    if (idx >= 0) return idx;
  }
  return null;
}
