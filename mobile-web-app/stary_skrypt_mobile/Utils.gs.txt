/** Utilities */
function nowIso_() {
  return new Date().toISOString();
}

/**
 * Client time helper.
 * Jeśli clientNow jest poprawnym timestampem ISO -> zwraca ISO (UTC).
 * W innym przypadku zwraca fallbackIso (jeśli podany) albo nowIso_().
 */
function clientNowIso_(clientNow, fallbackIso) {
  const raw = String(clientNow || '').trim();
  const fb = String(fallbackIso || '').trim() || nowIso_();
  if (!raw) return fb;
  try {
    const d = new Date(raw);
    if (!isNaN(d.getTime())) return d.toISOString();
  } catch (_) {}
  return fb;
}
function uuid_() {
  return Utilities.getUuid();
}
function addMinutesIso_(iso, minutes) {
  const d = new Date(iso);
  d.setMinutes(d.getMinutes() + Number(minutes || 0));
  return d.toISOString();
}
function secondsBetweenIso_(startIso, endIso) {
  const s = new Date(startIso).getTime();
  const e = new Date(endIso).getTime();
  return Math.max(0, Math.round((e - s) / 1000));
}
function parseJsonBody_(e) {
  const raw = e && e.postData && e.postData.contents ? e.postData.contents : '';
  if (!raw) return {};
  try {
    return JSON.parse(raw);
  } catch (err) {
    throw new Error('Invalid JSON body.');
  }
}
function jsonOk_(data) {
  const payload = { ok: true, data };
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}
function jsonError_(code, message, details) {
  const payload = { ok: false, error: { code, message } };
  if (details) payload.error.details = details;
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * Convert an ISO timestamp to a local date key (YYYY-MM-DD) using device tz offset (minutes).
 * tzOffsetMinutes: same as Date.getTimezoneOffset() (minutes behind UTC; e.g. Poland winter = -60? Actually getTimezoneOffset returns  -? it's positive for behind UTC.
 * We accept JS getTimezoneOffset() and apply: local = utc - offsetMinutes.
 */
function localDateKey_(iso, tzOffsetMinutes) {
  if (!iso) return '';
  try {
    const ms = new Date(String(iso)).getTime();
    if (!isFinite(ms)) return '';
    const offMin = Number(tzOffsetMinutes || 0);
    const localMs = ms - offMin * 60 * 1000;
    const d = new Date(localMs);
    const yyyy = d.getUTCFullYear();
    const mm = String(d.getUTCMonth() + 1).padStart(2, '0');
    const dd = String(d.getUTCDate()).padStart(2, '0');
    return yyyy + '-' + mm + '-' + dd;
  } catch (_) {
    return '';
  }
}

function isSameLocalDay_(isoA, isoB, tzOffsetMinutes) {
  const a = localDateKey_(isoA, tzOffsetMinutes);
  const b = localDateKey_(isoB, tzOffsetMinutes);
  return !!a && !!b && a === b;
}
