/**
 * QrResolver — "pień" do rozwiązywania kodów QR na obiekt strefy.
 *
 * Zasady:
 * - Jeśli kod pasuje do ^BCI\d{4}$ => zlecenie_indywidualne (dane z arkusza 957832581)
 * - W przeciwnym razie => standardowe strefy z arkusza Strefy (Zones.gs)
 * - Jeśli BCIxxxx nie istnieje => błąd "Nieznany kod zlecenia: ..."
 * - Jeśli zwykły QR nie istnieje => błąd "Nieznany kod QR: ..."
 */

function QrResolver_isIndOrderCode_(roomId) {
  const code = String(roomId || '').trim().toUpperCase();
  return /^BCI\d{4}$/.test(code);
}

function QrResolver_resolveRoom_(roomId) {
  const rid = String(roomId || '').trim();
  if (!rid) throw new Error('Missing roomId.');

  const code = rid.toUpperCase();

  // Individual order branch
  if (QrResolver_isIndOrderCode_(code)) {
    return IndOrdersService_resolveRoomByQr_(code); // throws if not found
  }

  // Zones branch
  try {
    return Zones_resolveRoom_(rid);
  } catch (err) {
    throw new Error('Nieznany kod QR: ' + rid);
  }
}

function QrResolver_getFuncForRoomId_(roomId) {
  const z = QrResolver_resolveRoom_(roomId);
  return String(z.func || 'CLEAN');
}
