<script>
window.BC = window.BC || {};
BC.TimePolicy = (function() {
  // "Dzisiaj" liczymy wg czasu urządzenia.
  // Na tym etapie tylko centralizujemy wartości przekazywane do API:
  // - clientNowIso() -> ISO string z urządzenia
  // - tzOffsetMinutes() -> new Date().getTimezoneOffset() (bez zmiany znaku / zachowania)

  function clientNowIso() {
    return new Date().toISOString();
  }

  function tzOffsetMinutes() {
    return new Date().getTimezoneOffset();
  }

  return {
    clientNowIso,
    tzOffsetMinutes
  };
})();
</script>
