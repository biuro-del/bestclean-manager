<script>
window.BC = window.BC || {};
BC.Store = (function() {
  const KEYS = {
    token: 'bc_token',
    worker: 'bc_worker',
    deviceId: 'bc_device_id',
    utilityRoomId: 'bc_utility_room_id',
    stopRules: 'bc_stop_rules',
    activeWorkdayId: 'bc_active_workday_id',
    activeWorkday: 'bc_active_workday',
    activeCycleId: 'bc_active_cycle_id',
    activeCycle: 'bc_active_cycle',
  };

  function getDeviceId() {
    let v = localStorage.getItem(KEYS.deviceId);
    if (!v) {
      v = (crypto && crypto.randomUUID)
        ? crypto.randomUUID()
        : String(Date.now()) + '_' + Math.random().toString(16).slice(2);
      localStorage.setItem(KEYS.deviceId, v);
    }
    return v;
  }

  function setToken(token) { localStorage.setItem(KEYS.token, token); }
  function getToken() { return localStorage.getItem(KEYS.token) || ''; }
  function clearToken() { localStorage.removeItem(KEYS.token); }

  function setWorker(worker) { localStorage.setItem(KEYS.worker, JSON.stringify(worker || {})); }
  function getWorker() {
    try { return JSON.parse(localStorage.getItem(KEYS.worker) || '{}'); }
    catch (_) { return {}; }
  }
  function clearWorker() { localStorage.removeItem(KEYS.worker); }

  function setUtilityRoomId(v) { localStorage.setItem(KEYS.utilityRoomId, String(v || '')); }
  function getUtilityRoomId() { return localStorage.getItem(KEYS.utilityRoomId) || ''; }

  function setStopRules(rules) { localStorage.setItem(KEYS.stopRules, JSON.stringify(rules || [])); }
  function getStopRules() {
    try { return JSON.parse(localStorage.getItem(KEYS.stopRules) || '[]'); }
    catch (_) { return []; }
  }
  function getStopGraceMin(roomId) {
    const id = String(roomId || '').trim();
    if (!id) return null;
    const rules = getStopRules();
    for (let i = 0; i < rules.length; i++) {
      if (String(rules[i].roomId) === id) return Number(rules[i].graceMin);
    }
    return null;
  }
  function isStopRoomId(roomId) { return getStopGraceMin(roomId) !== null; }

  function setActiveWorkday(workday) {
    if (workday && workday.workdayId) {
      localStorage.setItem(KEYS.activeWorkdayId, workday.workdayId);
      localStorage.setItem(KEYS.activeWorkday, JSON.stringify(workday));
    } else {
      localStorage.removeItem(KEYS.activeWorkdayId);
      localStorage.removeItem(KEYS.activeWorkday);
    }
  }
  function getActiveWorkdayId() { return localStorage.getItem(KEYS.activeWorkdayId) || ''; }
  function getActiveWorkday() {
    try { return JSON.parse(localStorage.getItem(KEYS.activeWorkday) || '{}'); }
    catch (_) { return {}; }
  }

  function setActiveCycle(cycle) {
    if (cycle && cycle.cycleId) {
      localStorage.setItem(KEYS.activeCycleId, cycle.cycleId);
      localStorage.setItem(KEYS.activeCycle, JSON.stringify(cycle));
    } else {
      localStorage.removeItem(KEYS.activeCycleId);
      localStorage.removeItem(KEYS.activeCycle);
    }
  }
  function getActiveCycleId() { return localStorage.getItem(KEYS.activeCycleId) || ''; }
  function getActiveCycle() {
    try { return JSON.parse(localStorage.getItem(KEYS.activeCycle) || '{}'); }
    catch (_) { return {}; }
  }

  function logout() {
    clearToken();
    clearWorker();
    setUtilityRoomId('');
    try { localStorage.removeItem(KEYS.stopRules); } catch (_) {}
    setActiveWorkday(null);
    setActiveCycle(null);
  }

  /**
   * If token is missing → auto logout + redirect to login screen (if UI exists)
   */
  function requireTokenOrLogin() {
    const token = getToken();
    if (!token) {
      logout();
      try {
        if (window.BC && BC.UI && typeof BC.UI.showLogin === 'function') {
          BC.UI.showLogin({ historyMode: 'replace' });
        }
      } catch (_) {}
      return false;
    }
    return true;
  }

  return {
    KEYS,
    getDeviceId,
    setToken, getToken, clearToken,
    setWorker, getWorker, clearWorker,
    setUtilityRoomId, getUtilityRoomId,
    setStopRules, getStopRules, getStopGraceMin, isStopRoomId,
    setActiveWorkday, getActiveWorkdayId, getActiveWorkday,
    setActiveCycle, getActiveCycleId, getActiveCycle,
    logout,
    requireTokenOrLogin
  };
})();
</script>
