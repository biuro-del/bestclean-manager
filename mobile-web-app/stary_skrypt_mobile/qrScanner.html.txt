<script>
window.BC = window.BC || {};
BC.QRScanner = (function() {
  const modal = document.getElementById('scanModal');
  const btnClose = document.getElementById('btnCloseScan');
  const btnTorch = document.getElementById('btnTorch'); // may be null if not present in HTML
  const video = document.getElementById('scanVideo');
  const statusEl = document.getElementById('scanStatus');

  // Legacy/manual fallback nodes might still exist in HTML — always keep them hidden.
  const manualBox = document.getElementById('manualFallback');
  const manualInput = document.getElementById('manualRoomId');
  const manualBtn = document.getElementById('btnManualSubmit');
  const manualCancel = document.getElementById('btnManualCancel');

  let stream = null;
  let videoTrack = null;
  let running = false;
  let onResult = null;
  let detector = null;

  let torchOn = false;
  let torchCapable = false;

  // Decoder selection:
  // - Prefer native BarcodeDetector when available.
  // - Fallback to jsQR (loaded from CDN) for iOS Safari / browsers without BarcodeDetector.
  let useJsQR = false;
  let jsqrLoaded = false;
  let scanCanvas = null;
  let scanCtx = null;

  function loadScript_(src) {
    return new Promise((resolve, reject) => {
      try {
        const s = document.createElement('script');
        s.src = src;
        s.async = true;
        s.onload = () => resolve(true);
        s.onerror = () => reject(new Error('Nie udało się załadować skryptu: ' + src));
        document.head.appendChild(s);
      } catch (e) {
        reject(e);
      }
    });
  }

  async function ensureDecoder_() {
    useJsQR = false;

    // Native path
    if ('BarcodeDetector' in window) {
      try {
        detector = new BarcodeDetector({ formats: ['qr_code'] });
        return;
      } catch (_) {
        // continue to jsQR fallback
      }
    }

    // Fallback path (jsQR)
    if (!jsqrLoaded) {
      statusEl.textContent = 'Ładuję moduł skanowania...';
      await loadScript_('https://unpkg.com/jsqr/dist/jsQR.js');
      jsqrLoaded = true;
    }

    if (typeof window.jsQR !== 'function') {
      throw new Error('Twoja przeglądarka nie obsługuje skanowania QR.');
    }

    useJsQR = true;
    if (!scanCanvas) scanCanvas = document.createElement('canvas');
    if (!scanCtx) scanCtx = scanCanvas.getContext('2d', { willReadFrequently: true });
  }


  function isSupported() {
    return navigator.mediaDevices && typeof navigator.mediaDevices.getUserMedia === 'function';
  }

  function normalizeQrValue_(raw) {
    const v = String(raw || '').trim();
    if (!v) return '';
    // Accept QR that contains a URL (extract roomId/room/id query param).
    if (/^https?:\/\//i.test(v)) {
      try {
        const u = new URL(v);
        const keys = ['roomId', 'room', 'id'];
        for (const k of keys) {
          const got = u.searchParams.get(k);
          if (got) return String(got).trim();
        }
      } catch (_) {}
    }
    return v;
  }

  async function getStream_() {
    const base = { audio: false };
    const attempts = [
      { video: { facingMode: { exact: 'environment' } } }, // force back camera (best for torch/QR)
      { video: { facingMode: { ideal: 'environment' } } }, // fallback
      { video: true },                                     // last resort
    ];

    let lastErr = null;
    for (const add of attempts) {
      try {
        return await navigator.mediaDevices.getUserMedia(Object.assign({}, base, add));
      } catch (e) {
        lastErr = e;
      }
    }
    throw lastErr || new Error('Nie udało się uruchomić kamery.');
  }

  function setTorchUi_() {
    if (!btnTorch) return;

    if (!videoTrack || !torchCapable) {
      // Hide/disable torch control when browser/device doesn't support it
      btnTorch.textContent = 'Latarka: niedostępna';
      btnTorch.disabled = true;
      btnTorch.classList.add('is-disabled');
      return;
    }

    btnTorch.classList.remove('is-disabled');
    btnTorch.textContent = torchOn ? 'Latarka: ON' : 'Latarka: OFF';
    btnTorch.disabled = false;
  }

  async function detectTorchSupport_(track) {
    torchCapable = false;
    torchOn = false;

    if (!track || typeof track.applyConstraints !== 'function') return false;

    // Preferred: capabilities flag
    try {
      if (typeof track.getCapabilities === 'function') {
        const caps = track.getCapabilities() || {};
        if (caps.torch === true) {
          torchCapable = true;
          return true;
        }
      }
    } catch (_) {}

    // Fallback: probe applyConstraints (some Android devices don't expose caps.torch)
    try {
      await track.applyConstraints({ advanced: [{ torch: false }] });
      torchCapable = true;
      return true;
    } catch (_) {
      torchCapable = false;
      return false;
    }
  }

  async function toggleTorch_() {
    if (!videoTrack || !torchCapable) {
      statusEl.textContent = 'Latarka niedostępna w tej przeglądarce/na tym urządzeniu.';
      return;
    }

    const want = !torchOn;
    try {
      await videoTrack.applyConstraints({ advanced: [{ torch: want }] });
      torchOn = want;
      setTorchUi_();
      statusEl.textContent = torchOn ? 'Latarka włączona.' : 'Latarka wyłączona.';
    } catch (_) {
      torchOn = false;
      torchCapable = false;
      setTorchUi_();
      statusEl.textContent = 'Latarka niedostępna w tej przeglądarce/na tym urządzeniu.';
    }
  }

  async function open(cb, opts) {
    const o = opts || {};
    const allowManual = !!o.allowManual;

    onResult = cb;
    modal.classList.remove('hidden');
    statusEl.textContent = 'Uruchamiam kamerę...';


// Manual input (paste/type) — ONLY when allowed (ADMIN)
if (manualBox) manualBox.classList.toggle('hidden', !allowManual);
if (manualInput) manualInput.value = '';
if (manualCancel) manualCancel.onclick = () => close();
if (manualBtn) {
  manualBtn.onclick = () => {
    if (!allowManual) return;
    const v = normalizeQrValue_(manualInput ? manualInput.value : '');
    if (!v) {
      statusEl.textContent = 'Wklej lub wpisz ID.';
      return;
    }
    close();
    onResult && onResult(v);
  };
}

    btnClose.onclick = () => close();

    if (btnTorch) {
      btnTorch.classList.remove('hidden');
      btnTorch.disabled = true;
      btnTorch.textContent = 'Latarka: OFF';
      btnTorch.onclick = () => toggleTorch_();
    }

    if (!isSupported()) {
      statusEl.textContent = 'Nie udało się uruchomić skanera. Na iPhone użyj Safari i zezwól na dostęp do aparatu (Ustawienia → Safari → Aparat).';
      return;
    }

    try {
      await ensureDecoder_();
      stream = await getStream_();
      video.srcObject = stream;

      // Save video track for torch
      videoTrack = (stream.getVideoTracks && stream.getVideoTracks()[0]) ? stream.getVideoTracks()[0] : null;

      // Detect torch support (device + browser). Then default-enable torch while scanning.
      if (videoTrack) {
        try { await detectTorchSupport_(videoTrack); } catch (_) {}
        setTorchUi_();

        // Default ON during scanning (best effort; ignore failures).
        if (torchCapable) {
          try {
            await videoTrack.applyConstraints({ advanced: [{ torch: true }] });
            torchOn = true;
          } catch (_) {
            torchOn = false;
          }
          setTorchUi_();
        }
      } else {
        setTorchUi_();
      }

      await video.play();

      running = true;
      statusEl.textContent = 'Skanuję...';
      tick();
    } catch (_) {
      statusEl.textContent = 'Nie udało się uruchomić kamery. Sprawdź uprawnienia (Ustawienia → Safari → Aparat).';
    }
  }

  async function tick(ts) {
    if (!running) return;

    // Throttle for jsQR (CPU heavy) ~8fps
    const nowTs = (typeof ts === 'number') ? ts : (window.performance && performance.now ? performance.now() : Date.now());
    if (useJsQR) {
      if (!tick._last) tick._last = 0;
      if (nowTs - tick._last < 120) {
        requestAnimationFrame(tick);
        return;
      }
      tick._last = nowTs;
    }

    try {
      if (!useJsQR) {
        if (detector) {
          const barcodes = await detector.detect(video);
          if (barcodes && barcodes.length) {
            const raw = (barcodes[0].rawValue || '').trim();
            const v = normalizeQrValue_(raw);
            if (v) {
              close();
              onResult && onResult(v);
              return;
            }
          }
        }
      } else {
        // jsQR fallback (iOS Safari etc.)
        const w = video.videoWidth || 0;
        const h = video.videoHeight || 0;
        if (w > 0 && h > 0 && scanCanvas && scanCtx) {
          if (scanCanvas.width !== w) scanCanvas.width = w;
          if (scanCanvas.height !== h) scanCanvas.height = h;

          scanCtx.drawImage(video, 0, 0, w, h);
          const img = scanCtx.getImageData(0, 0, w, h);
          const qr = window.jsQR(img.data, w, h, { inversionAttempts: 'dontInvert' });
          if (qr && qr.data) {
            const v = normalizeQrValue_(qr.data);
            if (v) {
              close();
              onResult && onResult(v);
              return;
            }
          }
        }
      }
    } catch (_) {}

    requestAnimationFrame(tick);
  }

  function close() {
    running = false;
    modal.classList.add('hidden');
    statusEl.textContent = '';
    torchOn = false;

    if (video) video.pause();
    if (stream) {
      try { stream.getTracks().forEach(t => t.stop()); } catch (_) {}
    }
    stream = null;
    videoTrack = null;
    detector = null;
    useJsQR = false;

    if (manualBox) manualBox.classList.add('hidden');
    if (btnTorch) {
      btnTorch.textContent = 'Latarka: OFF';
      btnTorch.disabled = true;
    }
  }

  return { open, close };
})();
</script>
