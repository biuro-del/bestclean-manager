/**
 * ChecklistService — business logic for checklist endpoints.
 */

function ChecklistService_getForQr_(worker, payload) {
  ChecklistRepo_ensureSheets_();
  const qrId = String((payload && payload.qrId) || '').trim();
  const daily = ChecklistRepo_listDaily_(qrId);
  const extra = ChecklistRepo_listExtraOpen_(qrId);

  const dailyTasks = daily.map(t => ({
    id: String(t.taskId || '').trim(),
    taskId: String(t.taskId || '').trim(),
    taskName: String(t.taskName || '').trim(),
    name: String(t.taskName || '').trim(),
    required: String(t.required || '').trim(),
    order: Number(t.order || 0),
  }));

  const extraTasks = extra.map(t => ({
    id: String(t.extraId || '').trim(),
    extraId: String(t.extraId || '').trim(),
    taskName: String(t.taskName || '').trim(),
    name: String(t.taskName || '').trim(),
    required: String(t.required || '').trim(),
    priority: Number(t.priority || 0),
  }));

  return { dailyTasks, extraTasks };
}

function ChecklistService_submit_(worker, payload) {
  ChecklistRepo_ensureSheets_();

  const qrId = String(payload.qrId || '').trim();
  const cycleId = String(payload.cycleId || '').trim();
  const workdayId = String(payload.workdayId || '').trim();

  if (cycleId && ChecklistRepo_logExistsByCycleId_(cycleId)) {
    return { ok: true, already: true };
  }

  const tzOff = Number(payload.tzOffsetMinutes || 0);
  const clientNow = String(payload.clientNow || nowIso_()).trim();
  const dateKey = localDateKey_(clientNow, tzOff) || localDateKey_(nowIso_(), tzOff);

  const wLabel = String((worker && (worker.name || worker.login)) || '').trim();

  const daily = Array.isArray(payload.daily) ? payload.daily : [];
  const extra = Array.isArray(payload.extra) ? payload.extra : [];

  const countStatus = (arr) => {
    let done = 0, notDone = 0;
    (arr || []).forEach(x => {
      const st = String((x && x.status) || '').trim().toUpperCase();
      if (st === 'DONE') done += 1;
      if (st === 'NOT_DONE') notDone += 1;
    });
    return { done, notDone, total: (arr||[]).length };
  };

  const dStat = countStatus(daily);
  const eStat = countStatus(extra);

  const dailyStats = dStat.total ? (dStat.done + ' OK / ' + dStat.notDone + ' NOT_DONE') : '';
  const extraStats = eStat.total ? (eStat.done + ' OK / ' + eStat.notDone + ' NOT_DONE') : '';

  const reasons = [];
  daily.forEach(x => {
    const st = String((x && x.status) || '').trim().toUpperCase();
    if (st !== 'NOT_DONE') return;
    reasons.push({ section: 'DAILY', id: String(x.taskId||''), name: String(x.taskName||''), reason: String(x.reason||'') });
  });
  extra.forEach(x => {
    const st = String((x && x.status) || '').trim().toUpperCase();
    if (st !== 'NOT_DONE') return;
    reasons.push({ section: 'EXTRA', id: String(x.extraId||''), name: String(x.taskName||''), reason: String(x.reason||'') });
  });

  // Store reasons in a human-readable way (better for Sheets view)
  // One line per NOT_DONE item: "SECTION: Task — reason"
  let reasonsText = '';
  try {
    reasonsText = (reasons || []).map(r => {
      const sec = String((r && r.section) || '').trim();
      const name = String((r && r.name) || '').trim();
      const id = String((r && r.id) || '').trim();
      const reason = String((r && r.reason) || '').trim();
      const left = (sec ? (sec + ': ') : '') + (name || id || '');
      return (left ? left : 'Powód') + ' — ' + reason;
    }).join('\n');
  } catch (_) {
    reasonsText = '';
  }
  if (reasonsText && reasonsText.length > 8000) reasonsText = reasonsText.slice(0, 8000);

  const meta = ChecklistRepo_zoneMetaByQrId_(qrId) || {};
  const client = String(meta.client || '').trim();
  const object = String(meta.object || '').trim();

  ChecklistRepo_appendLog_({
    logId: uuid_(),
    date: dateKey,
    worker: wLabel,
    workdayId,
    cycleId,
    client,
    object,
    qrId,
    dailyStats,
    extraStats,
    notDoneReasons: reasonsText,
    submittedAt: nowIso_(),
  });

  const updates = extra.map(x => ({
    extraId: String(x.extraId || '').trim(),
    status: String(x.status || '').trim(),
    reason: String(x.reason || '').trim(),
  })).filter(x => x.extraId && x.status);

  if (updates.length) ChecklistRepo_updateExtraStatuses_(updates, wLabel);

  return { ok: true };
}
