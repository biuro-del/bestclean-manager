<script>
window.BC = window.BC || {};
BC.Bootstrap = (function() {
  const DEFAULT_TIMEOUT_MS = 8000;

  function withTimeout(promise, ms, label) {
    const timeoutMs = Number(ms || 0) || DEFAULT_TIMEOUT_MS;
    return new Promise((resolve, reject) => {
      const id = setTimeout(() => {
        const e = new Error((label ? (label + ': ') : '') + 'timeout');
        e.code = 'TIMEOUT';
        reject(e);
      }, timeoutMs);

      Promise.resolve(promise).then(
        (v) => { clearTimeout(id); resolve(v); },
        (e) => { clearTimeout(id); reject(e); }
      );
    });
  }

  function authPayload_() {
    const token = (BC.Store && typeof BC.Store.getToken === 'function') ? BC.Store.getToken() : '';
    return token ? { token } : null;
  }

  function clientNowIso_() {
    try {
      if (BC.TimePolicy && typeof BC.TimePolicy.clientNowIso === 'function') return BC.TimePolicy.clientNowIso();
    } catch (_) {}
    return new Date().toISOString();
  }

  function tzOffsetMinutes_() {
    try {
      if (BC.TimePolicy && typeof BC.TimePolicy.tzOffsetMinutes === 'function') return BC.TimePolicy.tzOffsetMinutes();
    } catch (_) {}
    return new Date().getTimezoneOffset();
  }

  /**
   * Bootstrap po zalogowaniu:
   * - weryfikuje token (auth/me)
   * - hydratuje stan "dzisiaj" wg czasu urządzenia (workday/active, cycles/active)
   * - pobiera utilityId + stopRules
   *
   * Zasady:
   * - timeout (domyślnie 8s) -> wymagane ponowne logowanie (bez białego ekranu)
   * - brak komunikatów o zaległościach (biuro uzupełnia)
   */
  async function run(opts) {
    const o = opts || {};
    const timeoutMs = Number(o.timeoutMs || 0) || DEFAULT_TIMEOUT_MS;

    const auth = authPayload_();
    if (!auth) return { ok: false, needLogin: true, message: '' };

    // 1) Validate session
    try {
      const me = await withTimeout(BC.ApiClient.post('auth/me', auth), timeoutMs, 'auth/me');
      if (me && me.worker && BC.Store && typeof BC.Store.setWorker === 'function') {
        BC.Store.setWorker(me.worker);
      }
    } catch (e) {
      try { if (BC.Store && typeof BC.Store.logout === 'function') BC.Store.logout(); } catch (_) {}
      return { ok: false, needLogin: true, message: 'Wymagane ponowne logowanie.' };
    }

    // 2) Hydrate state (today)
    const clientNow = clientNowIso_();
    const tzOffsetMinutes = tzOffsetMinutes_();

    const pUtility = withTimeout(
      BC.ApiClient.post('workday/utilityId', auth),
      timeoutMs,
      'workday/utilityId'
    );

    const pWorkday = withTimeout(
      BC.ApiClient.post('workday/active', Object.assign({}, auth, { clientNow, tzOffsetMinutes })),
      timeoutMs,
      'workday/active'
    );

    const pCycle = withTimeout(
      BC.ApiClient.post('cycles/active', Object.assign({}, auth, { clientNow, tzOffsetMinutes })),
      timeoutMs,
      'cycles/active'
    );

    const settled = await Promise.allSettled([pUtility, pWorkday, pCycle]);

    const uRes = settled[0];
    const wdRes = settled[1];
    const cycleRes = settled[2];

    // Utility room & stop rules
    if (uRes.status === 'fulfilled') {
      const u = uRes.value || {};
      if (BC.Store && typeof BC.Store.setUtilityRoomId === 'function') {
        BC.Store.setUtilityRoomId(u.startRoomId || u.utilityRoomId || '');
      }
      if (BC.Store && typeof BC.Store.setStopRules === 'function') {
        BC.Store.setStopRules(u.stopRules || []);
      }
    } else {
      if (BC.Store && typeof BC.Store.setUtilityRoomId === 'function') BC.Store.setUtilityRoomId('');
      if (BC.Store && typeof BC.Store.setStopRules === 'function') BC.Store.setStopRules([]);
    }

    // Active workday (today only, backend decides using clientNow/tzOffsetMinutes)
    if (wdRes.status === 'fulfilled') {
      const wv = wdRes.value || {};
      if (BC.Store && typeof BC.Store.setActiveWorkday === 'function') {
        BC.Store.setActiveWorkday((wv && wv.workday) || null);
      }
      // NOTE: no staleOpen messaging (biuro uzupełnia)
    }

    // Active cycle (today only)
    if (cycleRes.status === 'fulfilled') {
      const cv = cycleRes.value || {};
      if (BC.Store && typeof BC.Store.setActiveCycle === 'function') {
        BC.Store.setActiveCycle((cv && cv.cycle) || null);
      }
    }

    // Hydrate cycle.func for an already running cycle (reload scenario)
    try {
      const c = (BC.Store && typeof BC.Store.getActiveCycle === 'function') ? BC.Store.getActiveCycle() : null;
      if (c && c.cycleId && c.roomId && !c.func) {
        const zr = await withTimeout(
          BC.ApiClient.post('zones/resolve', Object.assign({}, auth, { roomId: c.roomId })),
          timeoutMs,
          'zones/resolve'
        );
        if (zr && zr.zone && zr.zone.func) {
          c.func = zr.zone.func;
          if (BC.Store && typeof BC.Store.setActiveCycle === 'function') BC.Store.setActiveCycle(c);
        }
      }
    } catch (_) {}

    return { ok: true, needLogin: false };
  }

  return { run };
})();
</script>
