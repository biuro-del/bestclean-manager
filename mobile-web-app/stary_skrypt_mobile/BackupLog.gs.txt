/**
 * Backup_Log repository: append start rows and update stop data for a given CycleID.
 */
function BackupLog_appendStart_(record) {
  const sheet = Sheets_getBySheetId_(CONFIG.SHEET_ID_BACKUP_LOG);
  const H = CONFIG.HEADERS.BACKUP_LOG;

  const lastCol = sheet.getLastColumn();
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(x => String(x || '').trim());

  const byHeader = {};
  byHeader[H.CYCLE_ID] = record.cycleId;
  byHeader[H.WORKER_LOGIN] = record.workerLogin;
  byHeader[H.WORKER_NAME] = record.workerName;
  byHeader[H.ROOM_ID] = record.roomId;
  byHeader[H.ZONE] = record.zone;
  byHeader[H.ROOM] = record.room;
  byHeader[H.START_AT] = record.startAt;
  byHeader[H.STATUS] = record.status;
  byHeader[H.DEVICE_ID] = record.deviceId;
  byHeader[H.START_EVENT_ID] = record.startEventId;
  byHeader[H.CREATED_AT] = record.createdAt;

  const row = headers.map(h => (byHeader[h] !== undefined ? byHeader[h] : ''));
  sheet.appendRow(row);
}

function BackupLog_findRowByCycleId_(cycleId) {
  const sheet = Sheets_getBySheetId_(CONFIG.SHEET_ID_BACKUP_LOG);
  const h = Sheets_headerMap_(sheet);
  const col = h[CONFIG.HEADERS.BACKUP_LOG.CYCLE_ID];
  if (!col) throw new Error('Backup_Log missing header: CycleID');
  return Sheets_findRowIndexByValue_(sheet, col, cycleId);
}

function BackupLog_getByCycleId_(cycleId) {
  const sheet = Sheets_getBySheetId_(CONFIG.SHEET_ID_BACKUP_LOG);
  const rowIndex = BackupLog_findRowByCycleId_(cycleId);
  if (!rowIndex) return null;

  const lastCol = sheet.getLastColumn();
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(x => String(x || '').trim());
  const values = sheet.getRange(rowIndex, 1, 1, lastCol).getValues()[0];

  const obj = {};
  headers.forEach((name, i) => obj[name] = values[i]);
  return obj;
}

function BackupLog_updateStop_(cycleId, updates) {
  const sheet = Sheets_getBySheetId_(CONFIG.SHEET_ID_BACKUP_LOG);
  const h = Sheets_headerMap_(sheet);
  const rowIndex = BackupLog_findRowByCycleId_(cycleId);
  if (!rowIndex) throw new Error('Cycle not found: ' + cycleId);

  // Write row in one call (faster than many setValue calls)
  const lastCol = sheet.getLastColumn();
  const row = sheet.getRange(rowIndex, 1, 1, lastCol).getValues()[0];

  const H = CONFIG.HEADERS.BACKUP_LOG;
  const map = {
    endAt: H.END_AT,
    durationSec: H.DURATION_SEC,
    endReason: H.END_REASON,
    comment: H.COMMENT,
    status: H.STATUS,
    closeMarkedAt: H.CLOSE_MARKED_AT,
    endEventId: H.END_EVENT_ID,
  };

  let changed = false;
  Object.keys(updates || {}).forEach((k) => {
    const headerName = map[k];
    if (!headerName) return;
    const col = h[headerName];
    if (!col) return;
    row[col - 1] = updates[k];
    changed = true;
  });

  if (changed) sheet.getRange(rowIndex, 1, 1, lastCol).setValues([row]);
}

function BackupLog_findOpenWorkdayCycleForWorker_(workerLogin, opts) {
  const login = String(workerLogin || '').trim();
  if (!login) return null;

  const o = opts || {};
  const requireToday = !!o.requireToday;
  const clientNow = String(o.clientNow || '');
  const tzOffsetMinutes = Number(o.tzOffsetMinutes || 0);

  const sheet = Sheets_getBySheetId_(CONFIG.SHEET_ID_BACKUP_LOG);
  const h = Sheets_headerMap_(sheet);
  const H = CONFIG.HEADERS.BACKUP_LOG;

  const colLogin = h[H.WORKER_LOGIN];
  const colStatus = h[H.STATUS];
  const colEndAt = h[H.END_AT];
  const colStartAt = h[H.START_AT];
  const colCycleId = h[H.CYCLE_ID];

  if (!colLogin || !colCycleId || !colStatus || !colStartAt) return null;

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;

  const CHUNK = 2000;
  for (let endRow = lastRow; endRow >= 2; endRow -= CHUNK) {
    const startRow = Math.max(2, endRow - CHUNK + 1);
    const n = endRow - startRow + 1;

    const logins = sheet.getRange(startRow, colLogin, n, 1).getValues();
    const statuses = sheet.getRange(startRow, colStatus, n, 1).getValues();
    const endAts = colEndAt ? sheet.getRange(startRow, colEndAt, n, 1).getValues() : null;
    const startAts = sheet.getRange(startRow, colStartAt, n, 1).getValues();

    for (let i = n - 1; i >= 0; i--) {
      if (String(logins[i][0] || '').trim() !== login) continue;
      const st = String(statuses[i][0] || '').trim().toUpperCase();
      if (st !== 'WORKDAY') continue;

      const endAt = endAts ? String(endAts[i][0] || '').trim() : '';
      if (endAt) continue; // already closed

      const startAt = String(startAts[i][0] || '').trim();
      if (requireToday) {
        const nowIso = clientNow || nowIso_();
        const offMin = Number(tzOffsetMinutes || 0);
        if (!isSameLocalDay_(startAt, nowIso, offMin)) continue;
      }

      const rowIndex = startRow + i;
      const cycleId = sheet.getRange(rowIndex, colCycleId, 1, 1).getValues()[0][0];
      return {
        cycleId: String(cycleId || '').trim(),
        startAt: String(startAt || '').trim(),
        rowIndex: rowIndex,
      };
    }
  }
  return null;
}

function BackupLog_closeOpenWorkdayCycleForWorker_(workerLogin, opts) {
  const o = opts || {};
  const open = BackupLog_findOpenWorkdayCycleForWorker_(workerLogin, opts || {});
  if (!open || !open.cycleId) return null;

  const endAt = String(o.endAt || nowIso_());
  const startAt = String(open.startAt || '');
  let durationSec = '';
  try {
    if (startAt) durationSec = Math.max(0, Math.floor((new Date(endAt) - new Date(startAt)) / 1000));
  } catch (_) {}

  BackupLog_updateStop_(open.cycleId, {
    endAt: endAt,
    durationSec: durationSec,
    endReason: String(o.endReason || 'WORKDAY_STOP'),
    status: 'WORKDAY_CLOSED',
    endEventId: o.endEventId || uuid_(),
  });

  return { cycleId: open.cycleId, endAt: endAt, durationSec: durationSec };
}
