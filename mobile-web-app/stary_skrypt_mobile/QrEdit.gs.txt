 * Zakładka QR (gid=0):

 *  B: Nazwa klienta
 *  C: Strefa / pomieszczenie
 *  D: Funkcja kodu
 *  E: Edytował (login)
 *  F: Data i godzina (z urządzenia — clientAt, np. "YYYY-MM-DD HH:mm:ss")
 *
 * Zakładka klienci (gid=426771989):
 *  kolumna B: lista klientów (dropdown)
 */

const QREDIT_CFG = {
  SPREADSHEET_ID: '1aDllEIgfjTccPMtIormCwtfx0p2qxSQbSSlZiQBU3g0',
  SHEETID_QR: 0,
  SHEETID_CLIENTS: 426771989,
  COL_QR_ID: 1,      // A
  COL_CLIENT: 2,     // B
  COL_PLACE: 3,      // C
  COL_FUNCTION: 4,   // D
  COL_EDITED_BY: 5,  // E
  COL_EDITED_AT: 6,  // F
};


/**
 * Mapuje etykietę z UI (kol. D w formularzu) na wartość zapisywaną w bazie (kol. D w arkuszu QR).
 * Wartości docelowe (Twoje założenie):
 *  1) clean
 *  3) START
 *  4) STOP0
 *  5) STOP5
 *  6) STOP10
 *  7) STOP15
 *  8) podajnik_mydlo
 *  9) podajnik_papier_toaletowy
 * 10) podajnik_reczniki_papierowe
 * 11) podajnik_inne
 * 12) Indeks
 * 13) magazyn_lokalizacja
 */
function QrEdit_funcToCanonical_(raw) {
  const s0 = String(raw || '').trim();
  if (!s0) return '';

  // Jeśli już jest tokenem docelowym — zwróć.
  const sLower = s0.toLowerCase();
  const sUpper = s0.toUpperCase();

  if (sUpper === 'START') return 'START';
  if (sUpper === 'STOP0') return 'STOP0';
  if (sUpper === 'STOP5') return 'STOP5';
  if (sUpper === 'STOP10') return 'STOP10';
  if (sUpper === 'STOP15') return 'STOP15';

  if (sLower === 'clean') return 'clean';
  if (sLower === 'sprzatanie_indywidualne') return 'clean';
  if (sLower === 'zlecenie_indywidualne') return 'clean';
  if (sLower === 'podajnik_mydlo') return 'podajnik_mydlo';
  if (sLower === 'podajnik_papier_toaletowy') return 'podajnik_papier_toaletowy';
  if (sLower === 'podajnik_reczniki_papierowe') return 'podajnik_reczniki_papierowe';
  if (sLower === 'podajnik_inne') return 'podajnik_inne';
  if (sLower === 'magazyn_lokalizacja') return 'magazyn_lokalizacja';
  if (sLower === 'indeks') return 'Indeks';
  if (s0 === 'Indeks') return 'Indeks';

  // Mapowanie etykiet z UI -> tokeny
  const key = QrEdit_compactKey_(s0);

  const map = {
    'sprzatanie': 'clean',
    'zlecenieindywidualne': 'clean',
    'sprzatanieindywidualne': 'clean',
    'startczaspracy': 'START',
    'stop0czaspracy0min': 'STOP0',
    'stop5czaspracy5min': 'STOP5',
    'stop10czaspracy10min': 'STOP10',
    'stop15czaspracy15min': 'STOP15',
    'podajnikmydlo': 'podajnik_mydlo',
    'podajnikpapiertoaletowy': 'podajnik_papier_toaletowy',
    'podajnikrecznikipapierowe': 'podajnik_reczniki_papierowe',
    'podajnikinne': 'podajnik_inne',
    'indeksprodukt': 'Indeks',
    'magazynlokalizacjamagazynowa': 'magazyn_lokalizacja'
  };

  if (map[key]) return map[key];

  // Jeśli pojawi się nieznana wartość, nie blokujemy zapisu (zapisze się "jak było").
  return s0;
}

/**
 * Mapuje wartość zapisaną w bazie (token lub stara etykieta) na etykietę UI,
 * tak żeby dropdown w edycji pokazał poprawnie.
 */
function QrEdit_funcToUILabel_(raw) {
  const s0 = String(raw || '').trim();
  if (!s0) return '';

  const canon = QrEdit_funcToCanonical_(s0);

  const labels = {
    'clean': 'Sprzątanie',
    'START': 'START (czas pracy)',
    'STOP0': 'STOP0 (czas pracy + 0 min)',
    'STOP5': 'STOP5 (czas pracy + 5 min)',
    'STOP10': 'STOP10 (czas pracy + 10 min)',
    'STOP15': 'STOP15 (czas pracy + 15 min)',
    'podajnik_mydlo': 'Podajnik: mydło',
    'podajnik_papier_toaletowy': 'Podajnik: papier toaletowy',
    'podajnik_reczniki_papierowe': 'Podajnik: ręczniki papierowe',
    'podajnik_inne': 'Podajnik: inne',
    'Indeks': 'Indeks - Produkt',
    'magazyn_lokalizacja': 'Magazyn - lokalizacja magazynowa'
  };

  return labels[canon] || s0;
}

/**
 * Kompresuje tekst do klucza porównawczego (bez polskich znaków i separatorów).
 */
function QrEdit_compactKey_(s) {
  return String(s || '')
    .trim()
    .toLowerCase()
    .replace(/[ą]/g, 'a')
    .replace(/[ć]/g, 'c')
    .replace(/[ę]/g, 'e')
    .replace(/[ł]/g, 'l')
    .replace(/[ń]/g, 'n')
    .replace(/[ó]/g, 'o')
    .replace(/[ś]/g, 's')
    .replace(/[ż]/g, 'z')
    .replace(/[ź]/g, 'z')
    .replace(/[^a-z0-9]+/g, '');
}


function QrEdit_open_() {
  return SpreadsheetApp.openById(QREDIT_CFG.SPREADSHEET_ID);
}

function QrEdit_getSheetById_(ss, sheetId) {
  const target = Number(sheetId);
  const sheets = ss.getSheets();
  for (let i = 0; i < sheets.length; i++) {
    const sh = sheets[i];
    if (sh.getSheetId && Number(sh.getSheetId()) === target) return sh;
  }
  throw new Error('Sheet not found by sheetId: ' + sheetId);
}

/**
 * Lista klientów: unikalne wartości z kolumny B (od wiersza 2).
 */
function QrEdit_getClientNames_() {
  const ss = QrEdit_open_();
  const sh = QrEdit_getSheetById_(ss, QREDIT_CFG.SHEETID_CLIENTS);

  const lastRow = sh.getLastRow();
  if (lastRow < 2) return [];

  const values = sh.getRange(2, 2, lastRow - 1, 1).getValues(); // B2:B
  const uniq = {};
  for (let i = 0; i < values.length; i++) {
    const v = String(values[i][0] || '').trim();
    if (!v) continue;
    uniq[v] = true;
  }

  const out = Object.keys(uniq);
  out.sort((a, b) => a.localeCompare(b, 'pl'));
  return out;
}



/**
 * Pobierz rekord po QR_ID (kol. A).
 * Zwraca: { qrId, clientName, placeName, functionName, editedBy, editedAt } lub null.
 */
function QrEdit_getQrRecord_(qrId) {
  const id = String(qrId || '').trim();
  if (!id) return null;

  const ss = QrEdit_open_();
  const sh = QrEdit_getSheetById_(ss, QREDIT_CFG.SHEETID_QR);

  const lastRow = sh.getLastRow();
  if (lastRow < 1) return null;

  // Read column A (IDs)
  const ids = sh.getRange(1, QREDIT_CFG.COL_QR_ID, lastRow, 1).getValues();

  // If first row looks like a header, start from row 2.
  let start = 0;
  const first = String((ids[0] && ids[0][0]) || '').trim().toUpperCase();
  if (first === 'QR_ID' || first === 'QRID' || first === 'ID') start = 1;

  let foundRow = 0;
  for (let i = start; i < ids.length; i++) {
    if (String(ids[i][0] || '').trim() === id) { foundRow = i + 1; break; }
  }
  if (!foundRow) return null;

  const row = sh.getRange(foundRow, 1, 1, QREDIT_CFG.COL_EDITED_AT).getValues()[0];
  return {
    qrId: String(row[QREDIT_CFG.COL_QR_ID - 1] || '').trim(),
    clientName: String(row[QREDIT_CFG.COL_CLIENT - 1] || '').trim(),
    placeName: String(row[QREDIT_CFG.COL_PLACE - 1] || '').trim(),
    functionName: QrEdit_funcToUILabel_(String(row[QREDIT_CFG.COL_FUNCTION - 1] || '').trim()),
    editedBy: String(row[QREDIT_CFG.COL_EDITED_BY - 1] || '').trim(),
    editedAt: String(row[QREDIT_CFG.COL_EDITED_AT - 1] || '').trim(),
    rowNumber: foundRow
  };
}

/**
 * Upsert po QR_ID (kolumna A).
 * - jeśli istnieje: aktualizuje B..F
 * - jeśli nie: dopisuje A..F
 */
function QrEdit_upsertQrRecord_(rec) {
  const qrId = String(rec && rec.qrId || '').trim();
  if (!qrId) throw new Error('Missing qrId.');

  const ss = QrEdit_open_();
  const sh = QrEdit_getSheetById_(ss, QREDIT_CFG.SHEETID_QR);

  const lastRow = sh.getLastRow();
  let foundRow = 0;

  if (lastRow >= 1) {
    const ids = sh.getRange(1, QREDIT_CFG.COL_QR_ID, lastRow, 1).getValues(); // A1:A
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0] || '').trim() === qrId) { foundRow = i + 1; break; }
    }
  }

  const rowValues = [
    qrId,
    String(rec.clientName || '').trim(),
    String(rec.placeName || '').trim(),
    QrEdit_funcToCanonical_(String(rec.functionName || '').trim()),
    String(rec.editedBy || '').trim(),   // login
    String(rec.clientAt || '').trim(),   // device time
  ];

  if (!foundRow) {
    sh.appendRow(rowValues);
    return { ok: true, mode: 'insert', rowNumber: sh.getLastRow() };
  }

  // Update B..F
  sh.getRange(foundRow, QREDIT_CFG.COL_CLIENT, 1, 5).setValues([rowValues.slice(1)]);
  return { ok: true, mode: 'update', rowNumber: foundRow };
}


/**
 * Lokalizacja strefy (Strefy, kolumna G).
 * Wyszukuje po RoomID (kolumna A / ID) i zapisuje tekst lokalizacji w kolumnie 7.
 */
function QrEdit_setZoneLocationByRoomId_(roomId, location) {
  const rid = String(roomId || '').trim();
  if (!rid) throw new Error('Missing roomId for location.');

  const loc = String(location || '').trim();

  const sh = Sheets_getBySheetId_(CONFIG.SHEET_ID_ZONES);
  const h = Sheets_headerMap_(sh);
  const idCol = h[CONFIG.HEADERS.ZONES.ROOM_ID] || 1;

  const rowIndex = Sheets_findRowIndexByValue_(sh, idCol, rid);
  if (!rowIndex) throw new Error('Nie znaleziono kodu w arkuszu Strefy: ' + rid);

  // Column G = 7 (Lokalizacja)
  sh.getRange(rowIndex, 7).setValue(loc);
  return true;
}

function QrEdit_getZoneLocationByRoomId_(roomId) {
  const rid = String(roomId || '').trim();
  if (!rid) return '';

  const sh = Sheets_getBySheetId_(CONFIG.SHEET_ID_ZONES);
  const h = Sheets_headerMap_(sh);
  const idCol = h[CONFIG.HEADERS.ZONES.ROOM_ID] || 1;

  const rowIndex = Sheets_findRowIndexByValue_(sh, idCol, rid);
  if (!rowIndex) return '';

  // Column G = 7 (Lokalizacja)
  return String(sh.getRange(rowIndex, 7).getValue() || '').trim();
}
