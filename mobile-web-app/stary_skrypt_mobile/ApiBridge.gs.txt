/**
 * ApiBridge — frontend calls google.script.run.Api_post(payload).
 */
function Api_post(payload) {
  try {
    const fakeEvent = { postData: { contents: JSON.stringify(payload || {}) } };
    const out = Router_handleApiRequest_(fakeEvent);
    const text = out && out.getContent ? out.getContent() : String(out);
    return JSON.parse(text);
  } catch (err) {
    return {
      ok: false,
      error: { code: 'BRIDGE_ERROR', message: err && err.message ? err.message : String(err) }
    };
  }
}
