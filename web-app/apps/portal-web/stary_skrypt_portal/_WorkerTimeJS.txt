<script>
(function(){
  const $ = (sel)=>document.querySelector(sel);

  let _inited = false;

  function token(){
    return (window.FIDO_AUTH && window.FIDO_AUTH.getToken) ? window.FIDO_AUTH.getToken() : "";
  }

  function escapeHtml(s){
    return String(s ?? "")
      .replaceAll("&","&amp;")
      .replaceAll("<","&lt;")
      .replaceAll(">","&gt;")
      .replaceAll('"',"&quot;")
      .replaceAll("'","&#039;");
  }

  function yn(v){ return v ? "TAK" : "NIE"; }

  function baseExecUrl(){
    // działa dla /exec i /dev
    return window.location.origin + window.location.pathname;
  }

  function getRoute(){
    try{
      const u = new URL(window.location.href);
      return (u.searchParams.get('route') || '').trim();
    } catch(e){
      return '';
    }
  }

  function forceShowRoute(route){
    // bezpieczeństwo: jeśli główny router nie zadziała, pokaż właściwy widok ręcznie
    const targetId = 'view-' + route;
    const target = document.getElementById(targetId);
    if (!target) return;

    const main = document.querySelector('main.main');
    if (!main) return;

    const sections = main.querySelectorAll('section[id^="view-"]');
    sections.forEach(sec => {
      if (sec.id === targetId) sec.style.display = '';
      else if (sec.id !== 'view-placeholder') sec.style.display = 'none';
    });
  }


  function openWorkerDetail(worker){
    // Zamiast nowej karty (różne domeny GAS potrafią robić kłopoty z UI),
    // przechodzimy w ramach aplikacji na podstronę szczegółów.
    const u = new URL(window.location.href);
    u.searchParams.set('route', 'workerTimeDetail');
    u.searchParams.set('workerId', worker.workerId || '');
    u.searchParams.set('workerName', worker.name || '');
    window.history.pushState({}, '', u.toString());

    // jeśli jest globalny router, użyj go
    if (typeof window.go === 'function') {
      window.go('workerTimeDetail');
    } else {
      forceShowRoute('workerTimeDetail');
    }
  }

  function renderWorkers(items){
    const root = $("#wtRows");
    if (!root) return;

    const r = getRoute();
    if (r === "workerTime") forceShowRoute(r);
    root.innerHTML = "";

    if (!items || items.length === 0){
      const empty = document.createElement("div");
      empty.className = "events-row";
      empty.style.gridTemplateColumns = "140px 260px 220px 140px";
      empty.innerHTML = "<div>Brak wyników</div><div></div><div></div><div></div>";
      root.appendChild(empty);
      return;
    }

    for (const it of items){
      const row = document.createElement("div");
      row.className = "events-row";
      row.style.gridTemplateColumns = "140px 260px 220px 140px";
      row.innerHTML = `
        <div class="mono">${escapeHtml(it.workerId || "—")}</div>
        <div>${escapeHtml(it.name || "—")}</div>
        <div>${escapeHtml(it.type || "—")}</div>
        <div><button class="btn2" type="button">Pokaż</button></div>
      `;

      const btn = row.querySelector(".btn2");
      btn && btn.addEventListener("click", ()=>openWorkerDetail(it));

      root.appendChild(row);
    }
  }

  function readFilters(){
    return {
      token: token(),
      q: ($("#wtQ")?.value || "").trim(),
      type: ($("#wtType")?.value || "").trim(),
      page: 1,
      pageSize: 500
    };
  }

  function fetchWorkers(){
    const params = readFilters();
    google.script.run
      .withSuccessHandler((res)=>{
        const items = (res && res.items) ? res.items : [];
        renderWorkers(items);
      })
      .withFailureHandler((err)=>{
        const msg = (err && err.message) ? err.message : String(err || "Błąd");
        alert("Błąd pobierania pracowników: " + msg);
      })
      .api_listWorkers(params);
  }

  function bind(){
    const root = document.getElementById("view-workerTime");
    if (!root) return;

    const r = getRoute();
    if (r === "workerTime") forceShowRoute(r);

    const btnS = $("#wtSearchBtn");
    const btnR = $("#wtRefreshBtn");

    btnS && btnS.addEventListener("click", fetchWorkers);
    btnR && btnR.addEventListener("click", ()=>{
      if ($("#wtQ")) $("#wtQ").value = "";
      if ($("#wtType")) $("#wtType").value = "";
      fetchWorkers();
    });

    // Enter w polu szukania
    const q = $("#wtQ");
    q && q.addEventListener("keydown", (e)=>{
      if (e.key === "Enter") fetchWorkers();
    });

    fetchWorkers();
  }

  function init(){
    // wywoływane przez router (gdy przełączasz moduły w menu)
    if (_inited) {
      fetchWorkers();
      return;
    }
    _inited = true;
    bind();
  }

  // Udostępnij dla routera
  window.WorkerTimeModule = window.WorkerTimeModule || {};
  window.WorkerTimeModule.init = init;

  if (document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", ()=>{
      // auto-init tylko jeśli jesteśmy już na tej trasie
      const r = getRoute();
      if (r === "workerTime") init();
    });
  } else {
    const r = getRoute();
    if (r === "workerTime") init();
  }
})();
</script>
