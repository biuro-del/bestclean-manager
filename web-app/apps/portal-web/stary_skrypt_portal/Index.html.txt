      // Global helper used across modules (safe ISO date string)
      window._iso_ = window._iso_ || function(v){
        try{
          if(!v) return '';
          var d = (v instanceof Date) ? v : new Date(v);
          if (isNaN(d.getTime())) return String(v).trim();
          var y = d.getFullYear();
          var m = ('0' + (d.getMonth()+1)).slice(-2);
          var day = ('0' + d.getDate()).slice(-2);
          return y + '-' + m + '-' + day;
        }catch(e){
          try{ return String(v).trim(); }catch(_e){ return ''; }
        }
      };
