<script>
(function(){
  const $ = (sel)=>document.querySelector(sel);

  const state = {
    total: 0,
    limit: 200,
    mode: "edit",   // "edit" | "add"
    current: null   // edytowany wiersz (tryb edit)
  };

  function token(){
    return (window.FIDO_AUTH && window.FIDO_AUTH.getToken) ? window.FIDO_AUTH.getToken() : "";
  }
  function userRoleLevel(){
    try{
      return (window.FIDO_AUTH && window.FIDO_AUTH.roleLevel && window.FIDO_AUTH.getUserRole)
        ? (window.FIDO_AUTH.roleLevel(window.FIDO_AUTH.getUserRole()) || 0)
        : 0;
    } catch(e){
      return 0;
    }
  }
  function canEdit(){ return userRoleLevel() >= 2; }     // Admin/Kierownik
  function canDelete(){ return userRoleLevel() >= 3; }   // Admin

  function escapeHtml(s){
    return String(s ?? "")
      .replaceAll("&","&amp;")
      .replaceAll("<","&lt;")
      .replaceAll(">","&gt;")
      .replaceAll('"',"&quot;")
      .replaceAll("'","&#039;");
  }

  function setShownLabel(shown, total, limit){
    const el = $("#znShownLabel");
    if (!el) return;
    const lim = limit || state.limit || 200;
    el.textContent = `Wyświetlono: ${shown} (max ${lim}) · Wszystkie: ${total}`;
  }

  function readFilters(){
    return {
      token: token(),
      qr: ($("#znQr").value || "").trim(),
      klient: ($("#znClient").value || "").trim(),
      strefa: ($("#znStrefa").value || "").trim(),
      lokalizacja: ($("#znLoc").value || "").trim(),
      funkcja: ($("#znFunkcja").value || "").trim(),
      edytowal: ($("#znEditedBy").value || "").trim(),
      q: ($("#znQ").value || "").trim(),
      page: 1,
      pageSize: 200
    };
  }

  function resetFilters(){
    $("#znQr").value = "";
    $("#znClient").value = "";
    $("#znStrefa").value = "";
    $("#znLoc").value = "";
    $("#znFunkcja").value = "";
    $("#znEditedBy").value = "";
    $("#znQ").value = "";
  }

  function renderRows(items){
    const root = $("#znRows");
    if (!root) return;
    root.innerHTML = "";

    const showActions = canEdit() || canDelete();

    if (!items || items.length === 0){
      const empty = document.createElement("div");
      empty.className = "zones-row";
      empty.innerHTML = showActions
        ? "<div>Brak wyników</div><div></div><div></div><div></div><div></div><div></div><div></div><div></div>"
        : "<div>Brak wyników</div><div></div><div></div><div></div><div></div><div></div><div></div><div></div>";
      root.appendChild(empty);
      return;
    }

    for (const it of items){
      const row = document.createElement("div");
      row.className = "zones-row";

      const actionsHtml = (showActions)
        ? `<div class="zones-actions">
            ${canEdit() ? `<button class="btn2 zn-edit" type="button">Edytuj</button>` : ``}
            ${canDelete() ? `<button class="btn2 btn-del zn-del" type="button">Usuń</button>` : ``}
           </div>`
        : `<div></div>`;

      row.innerHTML = `
        <div class="mono">${escapeHtml(it.qr || "—")}</div>
        <div>${escapeHtml(it.klient || "—")}</div>
        <div title="${escapeHtml(it.strefa || "")}">${escapeHtml(it.strefa || "—")}</div>
        <div title="${escapeHtml(it.lokalizacja || "")}">${escapeHtml(it.lokalizacja || "—")}</div>
        <div>${escapeHtml(it.funkcja || "—")}</div>
        <div class="mono">${escapeHtml(it.data || "—")}</div>
        <div>${escapeHtml(it.edytowal || "—")}</div>
        ${actionsHtml}
      `;

      const editBtn = row.querySelector(".zn-edit");
      editBtn && editBtn.addEventListener("click", ()=>openEditorEdit(it));

      const delBtn = row.querySelector(".zn-del");
      delBtn && delBtn.addEventListener("click", ()=>deleteRow(it));

      root.appendChild(row);
    }
  }

  function fetchZones(){
    const params = readFilters();
    google.script.run
      .withSuccessHandler((res)=>{
        const items = (res && res.items) ? res.items : [];
        state.total = (res && typeof res.total === "number") ? res.total : items.length;
        state.limit = (res && typeof res.limit === "number") ? res.limit : 200;
        renderRows(items);
        setShownLabel(items.length, state.total, state.limit);
      })
      .withFailureHandler((err)=>{
        const msg = (err && err.message) ? err.message : String(err || "Błąd");
        alert("Błąd pobierania stref: " + msg);
      })
      .api_listZones(params);
  }

  function ensureEditorWired(){
    const ov = $("#znEditorOverlay");
    if (!ov) return;

    ov.addEventListener("click", (e)=>{
      if (e.target && e.target.id === "znEditorOverlay") closeEditor();
    });

    $("#znCancelBtn") && $("#znCancelBtn").addEventListener("click", closeEditor);
    $("#znSaveBtn") && $("#znSaveBtn").addEventListener("click", saveEditor);
    $("#znDeleteBtn") && $("#znDeleteBtn").addEventListener("click", deleteFromModal);

    if (window.FIDO_CONFIG && window.FIDO_CONFIG.logoUrl){
      const img = $("#znLogo");
      if (img) img.src = window.FIDO_CONFIG.logoUrl;
    }
  }

  function openEditorEdit(it){
    if (!canEdit()){
      alert("Brak uprawnień do edycji (wymagany ADMIN lub Kierownik).");
      return;
    }

    state.mode = "edit";
    state.current = it;

    $("#znQrLabel").textContent = it.qr || "—";
    $("#znRowNumberLabel").textContent = String(it.rowNumber || "—");

    $("#znEditQr").value = it.qr || "";
    $("#znEditClient").value = it.klient || "";
    $("#znEditStrefa").value = it.strefa || "";
    $("#znEditLoc").value = it.lokalizacja || "";
    $("#znEditFunkcja").value = it.funkcja || "";

    $("#znEditData").value = it.data || "";
    $("#znEditEdytowal").value = it.edytowal || "";

    const delBtn = $("#znDeleteBtn");
    if (delBtn) delBtn.style.display = canDelete() ? "inline-flex" : "none";

    $("#znEditorOverlay").style.display = "flex";
  }

  function openEditorAdd(){
    if (!canEdit()){
      alert("Brak uprawnień do dodawania (wymagany ADMIN lub Kierownik).");
      return;
    }

    state.mode = "add";
    state.current = null;

    $("#znQrLabel").textContent = "AUTO";
    $("#znRowNumberLabel").textContent = "—";

    $("#znEditQr").value = "AUTO";
    $("#znEditClient").value = "";
    $("#znEditStrefa").value = "";
    $("#znEditLoc").value = "";
    $("#znEditFunkcja").value = "";

    $("#znEditData").value = "AUTO";
    $("#znEditEdytowal").value = "AUTO";

    const delBtn = $("#znDeleteBtn");
    if (delBtn) delBtn.style.display = "none";

    $("#znEditorOverlay").style.display = "flex";
  }

  function closeEditor(){
    const ov = $("#znEditorOverlay");
    if (ov) ov.style.display = "none";
    state.current = null;
    state.mode = "edit";
  }

  function saveEditor(){
    if (!canEdit()){
      alert("Brak uprawnień do zapisu (wymagany ADMIN lub Kierownik).");
      return;
    }

    const btn = $("#znSaveBtn");
    const oldTxt = btn ? btn.textContent : "";
    if (btn){ btn.disabled = true; btn.textContent = "Zapisywanie..."; }

    const payloadCommon = {
      token: token(),
      klient: ($("#znEditClient").value || "").trim(),
      strefa: ($("#znEditStrefa").value || "").trim(),
      lokalizacja: ($("#znEditLoc").value || "").trim(),
      funkcja: ($("#znEditFunkcja").value || "").trim(),
    };

    if (state.mode === "add"){
      google.script.run
        .withSuccessHandler((res)=>{
          if (btn){ btn.disabled = false; btn.textContent = oldTxt || "Zapisz"; }
          closeEditor();
          fetchZones();
          if (res && res.qr) alert("Dodano strefę. Nadany numer QR: " + res.qr);
        })
        .withFailureHandler((err)=>{
          if (btn){ btn.disabled = false; btn.textContent = oldTxt || "Zapisz"; }
          const msg = (err && err.message) ? err.message : String(err || "Błąd");
          alert("Błąd dodawania strefy: " + msg);
        })
        .api_createZone(payloadCommon);
      return;
    }

    if (!state.current){
      if (btn){ btn.disabled = false; btn.textContent = oldTxt || "Zapisz"; }
      alert("Brak wybranego wiersza do edycji.");
      return;
    }

    const payload = Object.assign({}, payloadCommon, {
      rowNumber: state.current.rowNumber,
      qr: state.current.qr
    });

    google.script.run
      .withSuccessHandler(()=>{
        if (btn){ btn.disabled = false; btn.textContent = oldTxt || "Zapisz"; }
        closeEditor();
        fetchZones();
      })
      .withFailureHandler((err)=>{
        if (btn){ btn.disabled = false; btn.textContent = oldTxt || "Zapisz"; }
        const msg = (err && err.message) ? err.message : String(err || "Błąd");
        alert("Błąd zapisu strefy: " + msg);
      })
      .api_updateZone(payload);
  }

  function deleteRow(it){
    if (!canDelete()){
      alert("Brak uprawnień do usuwania (wymagany ADMIN).");
      return;
    }
    const ok = confirm(`Usunąć dane strefy ${it.qr}?\n(Zostanie tylko ID/QR w kolumnie A)`);
    if (!ok) return;

    google.script.run
      .withSuccessHandler(()=>{
        fetchZones();
      })
      .withFailureHandler((err)=>{
        const msg = (err && err.message) ? err.message : String(err || "Błąd");
        alert("Błąd usuwania: " + msg);
      })
      .api_deleteZone({ token: token(), rowNumber: it.rowNumber, qr: it.qr });
  }

  function deleteFromModal(){
    if (!state.current) return;
    deleteRow(state.current);
    closeEditor();
  }

  function bind(){
    const addBtn = $("#znAddBtn");
    if (addBtn){
      if (canEdit()){
        addBtn.style.display = "inline-flex";
        addBtn.addEventListener("click", openEditorAdd);
      } else {
        addBtn.style.display = "none";
      }
    }

    $("#znSearchBtn") && $("#znSearchBtn").addEventListener("click", fetchZones);
    $("#znResetBtn") && $("#znResetBtn").addEventListener("click", ()=>{ resetFilters(); fetchZones(); });

    ["#znQr","#znClient","#znStrefa","#znLoc","#znFunkcja","#znEditedBy","#znQ"].forEach(sel=>{
      const el = $(sel);
      el && el.addEventListener("keydown", (e)=>{ if (e.key === "Enter") fetchZones(); });
    });
  }

  function init(){
    if (!window.__FIDO_ZONES_WIRED__){
      window.__FIDO_ZONES_WIRED__ = true;
      bind();
      ensureEditorWired();
    }
    fetchZones();
  }

  window.ZonesModule = { init };
})();
</script>
