/**
 * WorkerTimeWorkday.gs
 * Moduł: Czas pracy pracownika (podsumowanie dzienne na bazie Workday_Log)
 * Wymaga istniejących: _authRequireRead_(), _authRequireWrite_(), EVENTS_SHEET
 */

function _wt_getSheetByNameOrGid_(ss, name, gidOpt) {
  let sh = null;
  if (gidOpt !== null && gidOpt !== undefined) {
    try { sh = ss.getSheetById(gidOpt); } catch (e) { sh = null; }
  }
  if (!sh) sh = ss.getSheetByName(name);
  if (!sh) throw new Error('Nie znaleziono zakładki: ' + name + (gidOpt != null ? (' (gid=' + gidOpt + ')') : ''));
  return sh;
}

function _wt_getWorkdaySheet_() {
  const ss = SpreadsheetApp.openById(EVENTS_SHEET.spreadsheetId);
  // gid niepewny → nazwa pewniejsza
  return _wt_getSheetByNameOrGid_(ss, 'Workday_Log', null);
}

/**
 * Backup_Log: map "workerName|dayKey" -> last non-empty "Edytowano przez" (kolumna R).
 * dayKey format: YYYY-MM-DD (local, from StartAt).
 * Wybiera ostatni (najświeższy) wpis, tzn. największy StartAt/CreatedAt jeśli dostępne.
 */
function _wt_backupEditorsByWorkerDay_(workerName, fromDayKey, toDayKey) {
  const wn = String(workerName || '').trim();
  const out = {}; // key -> { atIso, by }
  if (!wn) return out;

  // Backup_Log sheet (w tym samym skoroszycie co Workday_Log)
  const ss = SpreadsheetApp.openById(EVENTS_SHEET.spreadsheetId);
  const sh = _wt_getSheetByNameOrGid_(ss, 'Backup_Log', null);
  if (!sh) return out;

  const lastRow = sh.getLastRow();
  if (lastRow <= 1) return out;

  // Kolumny (po nagłówku)
  const header = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
  const col = {};
  for (let c = 0; c < header.length; c++) col[String(header[c] || '').trim()] = c;

  // wymagane: WorkerName, StartAt, "Edytowano przez"
  const idxWorkerName = (col['WorkerName'] != null) ? col['WorkerName'] : null;
  const idxStartAt = (col['StartAt'] != null) ? col['StartAt'] : null;
  const idxCreatedAt = (col['CreatedAt'] != null) ? col['CreatedAt'] : null;
  // kolumna R na screenie: "Edytowano przez" (dokładna nazwa)
  const idxEditedBy = (col['Edytowano przez'] != null) ? col['Edytowano przez'] : (col['Edytowano_przez'] != null ? col['Edytowano_przez'] : null);

  if (idxWorkerName == null || idxEditedBy == null) return out;

  const data = sh.getRange(2, 1, lastRow - 1, sh.getLastColumn()).getValues();
  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    const rWn = String(r[idxWorkerName] || '').trim();
    if (!rWn || rWn !== wn) continue;

    const atIso = String((idxCreatedAt != null ? r[idxCreatedAt] : (idxStartAt != null ? r[idxStartAt] : '')) || '').trim();
    const dayKey = _wt_toLocalDateKey_(String((idxStartAt != null ? r[idxStartAt] : atIso) || '').trim());
    if (!dayKey) continue;

    // zakres dat (opcjonalny)
    if (fromDayKey && dayKey < fromDayKey) continue;
    if (toDayKey && dayKey > toDayKey) continue;

    const by = String(r[idxEditedBy] || '').trim();
    if (!by) continue;

    const key = wn + '|' + dayKey;
    const cur = out[key];
    if (!cur || (atIso && String(atIso) > String(cur.atIso || ''))) {
      out[key] = { atIso: atIso, by: by };
    }
  }

  const flat = {};
  Object.keys(out).forEach(k => flat[k] = out[k].by);
  return flat;
}

function _wt_getZonesSheet_() {
  // Preferuj stałą ZONES_SHEET (z Code.gs), ale miej fallback po nazwie.
  const spreadsheetId = (typeof ZONES_SHEET !== 'undefined' && ZONES_SHEET && ZONES_SHEET.spreadsheetId)
    ? ZONES_SHEET.spreadsheetId
    : EVENTS_SHEET.spreadsheetId;

  const gid = (typeof ZONES_SHEET !== 'undefined' && ZONES_SHEET && ZONES_SHEET.gid != null)
    ? ZONES_SHEET.gid
    : 0;

  const ss = SpreadsheetApp.openById(spreadsheetId);
  let sh = null;
  try { sh = ss.getSheetById(gid); } catch (e) { sh = null; }
  if (!sh) sh = ss.getSheetByName('Strefy');
  if (!sh) throw new Error('Nie znaleziono zakładki Strefy (gid=' + gid + ' / nazwa "Strefy").');
  return sh;
}

function _wt_normZoneKey_(raw) {
  const s0 = String(raw || '').trim().toUpperCase();
  if (!s0) return '';
  // Najczęściej BC0001
  const m = s0.match(/BC\d{4}/);
  if (m) return m[0];
  // Fallback: usuń białe znaki
  return s0.replace(/\s+/g, '');
}

function _wt_buildZoneClientMap_() {
  const sh = _wt_getZonesSheet_();
  const lastRow = sh.getLastRow();
  if (lastRow <= 1) return {};

  // Strefy: A=QR, B=Klient
  const lastCol = Math.max(2, Math.min(5, sh.getLastColumn()));
  const data = sh.getRange(2, 1, lastRow - 1, lastCol).getValues();

  const map = {};
  for (let i = 0; i < data.length; i++) {
    const qr = _wt_normZoneKey_(data[i][0]);
    const klient = String(data[i][1] || '').trim();
    if (!qr) continue;
    if (klient) map[qr] = klient;
  }
  return map;
}

function _wt_resolveClientName_(utilityRoomId, zoneClientMap) {
  const key = _wt_normZoneKey_(utilityRoomId);
  if (!key) return '';
  if (zoneClientMap && zoneClientMap[key]) return zoneClientMap[key];
  // fallback: pokaż kod, zamiast pustego
  return key;
}


function _wt_alertLevelFromWorkSec_(workSec){
  const sec = Math.max(0, _wt_safeInt_(workSec, 0));
  if (sec > 10*3600) return 2;
  if (sec > 9*3600) return 1;
  return 0;
}
function _wt_isTrue_(v){
  const s = String(v ?? '').trim().toLowerCase();
  return s === 'true' || s === '1' || s === 'tak' || s === 'yes';
}
function _wt_ensureCols_(sheet, need){
  need = need || [];
  let hi = _wt_getHeaderMap_(sheet);
  let lastCol = sheet.getLastColumn();
  let changed = false;
  for (const h of need){
    if (hi.map[h] == null){
      lastCol += 1;
      sheet.getRange(1, lastCol).setValue(h);
      changed = true;
    }
  }
  return changed ? _wt_getHeaderMap_(sheet) : hi;
}

function _wt_getHeaderMap_(sheet) {
  const headerRow = 1;
  const lastCol = sheet.getLastColumn();
  const headers = sheet.getRange(headerRow, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
  const map = {};
  headers.forEach((h, i) => { if (h) map[h] = i; });
  return { headers, map, headerRow, lastCol };
}

function _wt_isoToDate_(iso) {
  if (!iso) return null;
  const d = new Date(String(iso));
  if (!isFinite(d.getTime())) return null;
  return d;
}

function _wt_fmtDate_(d) {
  // yyyy-mm-dd
  const yyyy = d.getFullYear();
  const mm = ('0' + (d.getMonth() + 1)).slice(-2);
  const dd = ('0' + d.getDate()).slice(-2);
  return yyyy + '-' + mm + '-' + dd;
}

function _wt_toLocalDateKey_(iso) {
  const d = _wt_isoToDate_(iso);
  if (!d) return '';
  // Klucz dnia wg strefy skryptu (ustaw w projekcie na Europe/Warsaw)
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function _wt_safeInt_(v, def) {
  const n = parseInt(v, 10);
  return isFinite(n) ? n : (def || 0);
}

function _wt_parseISODate_(s) {
  // przyjmujemy yyyy-mm-dd
  const t = String(s || '').trim();
  if (!t) return '';
  if (!/^\d{4}-\d{2}-\d{2}$/.test(t)) return '';
  return t;
}

function _wt_hmToSec_(hm) {
  const m = String(hm || '').trim().match(/^(\d{1,3}):(\d{2})$/);
  if (!m) return 0;
  const hh = _wt_safeInt_(m[1], 0);
  const mm = _wt_safeInt_(m[2], 0);
  return Math.max(0, hh * 3600 + mm * 60);
}

function _wt_findRowByWorkdayId_(sheet, workdayId, headerInfo) {
  const id = String(workdayId || '').trim();
  if (!id) return 0;

  const idx = headerInfo.map['WorkdayID'];
  if (idx == null) throw new Error('Brak kolumny WorkdayID w Workday_Log.');

  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return 0;
  const values = sheet.getRange(2, idx + 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0] || '').trim() === id) return 2 + i;
  }
  return 0;
}

/**
 * Lista dni (podsumowanie) dla pracownika w zakresie dat.
 * 1 wiersz = 1 dzień (agregacja ze wszystkich wpisów Workday_Log w danym dniu dla pracownika).
 *
 * Definicje:
 * - Start pracy: najwcześniejszy StartAt w danym dniu
 * - Koniec pracy: najpóźniejszy EndAt w danym dniu
 * - Czas pracy: (Koniec - Start) dla danego dnia
 * - Przerwa: suma PauseTotalSec (Workday_Log) w danym dniu
 *
 * Dodatkowo zwraca sumy miesięczne (dla miesiąca z pola "Do" lub bieżącego):
 * - monthWorkSec: suma (endMax-startMin) po dniach w miesiącu (dla pracownika)
 * - monthBreakSec: suma PauseTotalSec w miesiącu (dla pracownika)
 * - monthNetSec: monthWorkSec - monthBreakSec
 */
function api_wt_listDays(params) {
  params = params || {};
  _authRequireRead_(String(params.token || '').trim());

  const tz = Session.getScriptTimeZone();

  const workerName = String(params.workerName || '').trim();
  if (!workerName) throw new Error('Brak workerName.');

  const fromIso = _wt_parseISODate_(params.fromIso);
  const toIso = _wt_parseISODate_(params.toIso);

  const page = Math.max(1, _wt_safeInt_(params.page, 1));
  const pageSize = Math.min(200, Math.max(10, _wt_safeInt_(params.pageSize, 50)));

  // Miesiąc pod sumę: bierzemy "to" jeśli jest, inaczej dziś
  const toForMonth = (toIso || Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'));
  const monthLabel = toForMonth.slice(0, 7); // yyyy-MM
  const y = parseInt(monthLabel.slice(0, 4), 10);
  const m = parseInt(monthLabel.slice(5, 7), 10);
  const monthFrom = `${monthLabel}-01`;
  const monthTo = Utilities.formatDate(new Date(y, m, 0), tz, 'yyyy-MM-dd');

  const sh = _wt_getWorkdaySheet_();
  let hi = _wt_getHeaderMap_(sh);
  const needCols = ['WT_AlertAck','WT_AlertAckAt','WT_AlertAckBy','UpdatedAt','UpdatedBy','Status'];
  if (hi.map['Comment'] == null && hi.map['Komentarz'] == null) needCols.push('Comment');
  hi = _wt_ensureCols_(sh, needCols);
const lastRow = sh.getLastRow();
  if (lastRow <= 1) {
    return {
      items: [],
      total: 0,
      page,
      pageSize,
      monthLabel,
      monthFrom,
      monthTo,
      monthWorkSec: 0,
      monthBreakSec: 0,
      monthNetSec: 0,
    };
  }

  const col = hi.map;
  const idxWorkdayId = col['WorkdayID'];
  const idxWorkerName = col['WorkerName'];
  const idxStartAt = col['StartAt'];
  const idxEndAt = col['EndAt'];
  const idxPauseTotal = col['PauseTotalSec'];

  const idxUpdatedAt = col['UpdatedAt'];
  const idxUpdatedBy = col['UpdatedBy'];
  const idxComment = (col['Comment'] != null) ? col['Comment'] : ((col['Komentarz'] != null) ? col['Komentarz'] : null);
  const idxAck = col['WT_AlertAck'];
  const idxAckAt = col['WT_AlertAckAt'];
  const idxAckBy = col['WT_AlertAckBy'];

  if (idxWorkdayId == null || idxWorkerName == null || idxStartAt == null) {
    throw new Error('Workday_Log: brakuje wymaganych kolumn (WorkdayID/WorkerName/StartAt).');
  }

  const data = sh.getRange(2, 1, lastRow - 1, hi.lastCol).getValues();


  // Backup_Log: kto edytował (kolumna R: 'Edytowano przez')
  const fromDayKey = fromIso || '';
  const toDayKey = toIso || '';
  const backupEditors = _wt_backupEditorsByWorkerDay_(workerName, fromDayKey, toDayKey);
  // group: dzień (dla listy)
  const groups = {};

  // group: dzień (dla sumy miesięcznej)
  const monthGroups = {};

  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    const wn = String(r[idxWorkerName] || '').trim();
    if (!wn || wn !== workerName) continue;

    const startAt = String(r[idxStartAt] || '').trim();
    if (!startAt) continue;

    const dayKey = _wt_toLocalDateKey_(startAt);
    if (!dayKey) continue;

    const editorName = String(backupEditors[String(workerName)+'|'+dayKey] || '').trim();

    const endAt = (idxEndAt != null) ? String(r[idxEndAt] || '').trim() : '';
    const pauseTotalSec = (idxPauseTotal != null) ? _wt_safeInt_(r[idxPauseTotal], 0) : 0;

    // --- sumy miesięczne (po dniu) ---
    if (dayKey >= monthFrom && dayKey <= monthTo) {
      let mg = monthGroups[dayKey];
      if (!mg) {
        mg = monthGroups[dayKey] = { minStartIso: startAt, maxEndIso: endAt, breakSec: 0 };
      } else {
        const dsNew = _wt_isoToDate_(startAt);
        const dsOld = _wt_isoToDate_(mg.minStartIso);
        if (dsNew && (!dsOld || dsNew.getTime() < dsOld.getTime())) mg.minStartIso = startAt;

        const deNew = _wt_isoToDate_(endAt);
        const deOld = _wt_isoToDate_(mg.maxEndIso);
        if (deNew && (!deOld || deNew.getTime() > deOld.getTime())) mg.maxEndIso = endAt;
      }
      mg.breakSec += Math.max(0, pauseTotalSec);
    }

    // --- filtr zakresu dla listy ---
    if (fromIso && dayKey < fromIso) continue;
    if (toIso && dayKey > toIso) continue;

    let g = groups[dayKey];
    if (!g) {
      g = groups[dayKey] = {
        dateKey: dayKey,
        workerName: wn,
        minStartIso: startAt,
        maxEndIso: endAt,
        breakSec: 0,
        updatedBy: editorName,
        // ostatnia edycja w tym dniu (najświeższe UpdatedAt)
        lastUpdatedAtIso: String((idxUpdatedAt != null ? r[idxUpdatedAt] : '') || '').trim(),
        lastUpdatedBy: String((idxUpdatedBy != null ? r[idxUpdatedBy] : '') || '').trim(),
        // jako bazowy WorkdayID wybieramy rekord z najwcześniejszym StartAt
        workdayId: String(r[idxWorkdayId] || '').trim(),
        comment: String((idxComment != null ? r[idxComment] : '') || '').trim(),
      };
    } else {
      const dsNew = _wt_isoToDate_(startAt);
      const dsOld = _wt_isoToDate_(g.minStartIso);
      if (dsNew && (!dsOld || dsNew.getTime() < dsOld.getTime())) {
        g.minStartIso = startAt;
        g.workdayId = String(r[idxWorkdayId] || '').trim();
      }

      const deNew = _wt_isoToDate_(endAt);
      const deOld = _wt_isoToDate_(g.maxEndIso);
      if (deNew && (!deOld || deNew.getTime() > deOld.getTime())) {
        g.maxEndIso = endAt;
      }

      // last editor (max UpdatedAt)
      if (idxUpdatedAt != null) {
        const uAt = String(r[idxUpdatedAt] || '').trim();
        if (uAt) {
          const cur = String(g.lastUpdatedAtIso || '').trim();
          if (!cur || uAt > cur) {
            g.lastUpdatedAtIso = uAt;
            g.lastUpdatedBy = String((idxUpdatedBy != null ? r[idxUpdatedBy] : '') || '').trim();
            if (idxComment != null) g.comment = String(r[idxComment] || '').trim();
          }
        }
      }
    }

    // ACK: jeśli którykolwiek wpis w dniu ma WT_AlertAck=true, to cały dzień uznajemy za ACK
    if (idxAck != null && _wt_isTrue_(r[idxAck])) g.alertAck = true;
    if (idxAckAt != null){
      const vAt = String(r[idxAckAt] || '').trim();
      if (vAt && !g.alertAckAt) g.alertAckAt = vAt;
    }
    if (idxAckBy != null){
      const vBy = String(r[idxAckBy] || '').trim();
      if (vBy && !g.alertAckBy) g.alertAckBy = vBy;
    }

    g.breakSec += Math.max(0, pauseTotalSec);
  }

  // zbuduj items
  const all = [];
  Object.keys(groups).forEach((k) => {
    const g = groups[k];
    let workSec = 0;
    const ds = _wt_isoToDate_(g.minStartIso);
    const de = _wt_isoToDate_(g.maxEndIso);
    if (ds && de) {
      workSec = Math.max(0, Math.round((de.getTime() - ds.getTime()) / 1000));
    }

    const alertLevel = _wt_alertLevelFromWorkSec_(workSec);

    all.push({
      workdayId: g.workdayId || '',
      workerName: g.workerName,
      dateKey: g.dateKey,
      startAt: g.minStartIso,
      endAt: g.maxEndIso,
      workSec: workSec,
      breakSec: g.breakSec,
      netSec: Math.max(0, workSec - Math.max(0, _wt_safeInt_(g.breakSec, 0))),
      alertLevel: alertLevel,
      alertAck: !!g.alertAck,
      alertAckAt: g.alertAckAt || '',
      alertAckBy: g.alertAckBy || '',
    
      updatedBy: g.updatedBy || '',
      comment: g.comment || '',
    });
});

  // sort: najnowsze na górze
  all.sort((a, b) => {
    if (a.dateKey !== b.dateKey) return (a.dateKey < b.dateKey) ? 1 : -1;
    return (String(a.startAt) < String(b.startAt)) ? 1 : -1;
  });

  const total = all.length;
  const start = (page - 1) * pageSize;
  const items = all.slice(start, start + pageSize);

  // suma miesięczna (po dniach)
  let monthWorkSec = 0;
  let monthBreakSec = 0;
  Object.keys(monthGroups).forEach((dk) => {
    const mg = monthGroups[dk];
    monthBreakSec += Math.max(0, _wt_safeInt_(mg.breakSec, 0));

    const ds = _wt_isoToDate_(mg.minStartIso);
    const de = _wt_isoToDate_(mg.maxEndIso);
    if (ds && de) monthWorkSec += Math.max(0, Math.round((de.getTime() - ds.getTime()) / 1000));
  });
  const monthNetSec = Math.max(0, monthWorkSec - monthBreakSec);

  return { items, total, page, pageSize, monthLabel, monthWorkSec, monthBreakSec, monthNetSec, monthFrom, monthTo };
}

/**
 * Edycja dnia (workday) — tylko ADMIN/Kierownik
 *
 * Widok agreguje po DNIU (bez klienta). Jeśli w danym dniu istnieje
 * więcej niż 1 wpis w Workday_Log, to przy zapisie:
 * - ustawiamy StartAt/EndAt na wskazanym WorkdayID
 * - wszystkie inne wpisy z tego samego dnia "przycinamy" do zakresu
 *   (żeby nie psuły min/max) oraz zerujemy PauseTotalSec
 */

/**
 * ALERTY — mapa pracowników z przekroczeniami normy w Workday_Log
 * Zwraca mapę: { "Imię Nazwisko": 2 } gdzie:
 *  - 2 => >10h (czerwony)
 *  - 1 => >9h (pomarańczowy)
 *
 * Uwaga: na liście pracowników chcemy tylko sygnał ostrzegawczy,
 * więc frontend może traktować oba poziomy jako "czerwony".
 *
 * Filtry:
 *  - month (yyyy-MM) opcjonalnie, domyślnie bieżący miesiąc
 *  - workerNames (array) opcjonalnie, jeśli podasz to liczy tylko dla tych nazw
 *
 * Nie pokazujemy ACK (WT_AlertAck=true nie liczy się jako aktywny alert).
 */
function api_wt_workersAlertMap(params){
  params = params || {};
  _authRequireRead_(String(params.token || '').trim());

  const tz = Session.getScriptTimeZone();
  const month = String(params.month || '').trim(); // yyyy-MM
  const workerNames = (params.workerNames && Array.isArray(params.workerNames)) ? params.workerNames : null;

  const now = new Date();
  const monthLabel = (/^\d{4}-\d{2}$/.test(month)) ? month : Utilities.formatDate(now, tz, 'yyyy-MM');
  const y = parseInt(monthLabel.slice(0,4),10);
  const m = parseInt(monthLabel.slice(5,7),10);
  const fromYmd = `${monthLabel}-01`;
  const toYmd = Utilities.formatDate(new Date(y, m, 0), tz, 'yyyy-MM-dd');

  const sh = _wt_getWorkdaySheet_();
  let hi = _wt_getHeaderMap_(sh);
  const needCols = ['WT_AlertAck','WT_AlertAckAt','WT_AlertAckBy','UpdatedAt','UpdatedBy','Status'];
  if (hi.map['Comment'] == null && hi.map['Komentarz'] == null) needCols.push('Comment');
  hi = _wt_ensureCols_(sh, needCols);

  const col = hi.map;
  const idxWorkerName = col['WorkerName'];
  const idxStartAt = col['StartAt'];
  const idxEndAt = col['EndAt'];
  const idxAck = col['WT_AlertAck'];

  if (idxWorkerName == null || idxStartAt == null || idxEndAt == null){
    throw new Error('Workday_Log: brakuje wymaganych kolumn (WorkerName/StartAt/EndAt).');
  }

  const lastRow = sh.getLastRow();
  if (lastRow <= 1){
    return { ok:true, monthLabel, fromYmd, toYmd, map:{} };
  }

  // normalizacja nazw pracowników do szybkiego filtra (opcjonalnie)
  let only = null;
  if (workerNames && workerNames.length){
    only = {};
    workerNames.forEach(n=>{
      const k = String(n||'').trim();
      if (k) only[k] = 1;
    });
  }

  const data = sh.getRange(2, 1, lastRow-1, hi.lastCol).getValues();
  const out = {}; // workerName -> maxLevel

  for (let i=0;i<data.length;i++){
    const r = data[i];
    const wn = String(r[idxWorkerName]||'').trim();
    if (!wn) continue;
    if (only && !only[wn]) continue;

    const startIso = String(r[idxStartAt]||'').trim();
    const endIso = String(r[idxEndAt]||'').trim();
    if (!startIso || !endIso) continue;

    const dayKey = _wt_toLocalDateKey_(startIso);
    if (!dayKey) continue;
    if (dayKey < fromYmd || dayKey > toYmd) continue;

    // jeśli ACK=true, nie traktujemy jako aktywny alert
    if (idxAck != null && _wt_isTrue_(r[idxAck])) continue;

    const ds = _wt_isoToDate_(startIso);
    const de = _wt_isoToDate_(endIso);
    if (!ds || !de) continue;

    const workSec = Math.max(0, Math.round((de.getTime() - ds.getTime())/1000));
    const lvl = _wt_alertLevelFromWorkSec_(workSec);
    if (lvl <= 0) continue;

    const prev = out[wn] || 0;
    if (lvl > prev) out[wn] = lvl;
  }

  return { ok:true, monthLabel, fromYmd, toYmd, map: out };
}




/**
 * Dodanie ręczne dnia pracy (1 dzień = 1 rekord w Workday_Log) — tylko ADMIN/Kierownik
 *
 * payload:
 *  - token (required)
 *  - editorName (optional)
 *  - workerName (required)
 *  - startAt (ISO) required
 *  - endAt (ISO) required
 *  - breakSec (number) optional
 *
 * Blokada duplikatu: jeśli istnieje już wpis dla workerName w tym dniu -> błąd.
 */
function api_wt_createDay(payload){
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || '').trim());

  const workerName = String(payload.workerName || '').trim();
  if (!workerName) throw new Error('Brak workerName.');

  const startAt = String(payload.startAt || '').trim();
  const endAt = String(payload.endAt || '').trim();
  const breakSec = Math.max(0, _wt_safeInt_(payload.breakSec, 0));
  const comment = String(payload.comment || '').trim();

  const ds = _wt_isoToDate_(startAt);
  const de = _wt_isoToDate_(endAt);
  if (!ds || !de) throw new Error('Podaj poprawny Start i Koniec (ISO).');
  if (de.getTime() < ds.getTime()) throw new Error('Koniec < Start.');

  const workSec = Math.max(0, Math.round((de.getTime() - ds.getTime())/1000));
  if (workSec > 24*3600) throw new Error('Czas pracy > 24h — sprawdź dane.');
  if (breakSec > 12*3600) throw new Error('Przerwa > 12h — sprawdź dane.');

  const sh = _wt_getWorkdaySheet_();
  let hi = _wt_getHeaderMap_(sh);
  hi = _wt_ensureCols_(sh, ['WT_AlertAck','WT_AlertAckAt','WT_AlertAckBy','UpdatedAt','UpdatedBy']);

  const col = hi.map;
  const idxWorkdayId = col['WorkdayID'];
  const idxWorkerName = col['WorkerName'];
  const idxStartAt = col['StartAt'];
  const idxEndAt = col['EndAt'];
  const idxDuration = col['DurationSec'];
  const idxPauseTotal = col['PauseTotalSec'];
  const idxUpdatedAt = col['UpdatedAt'];
  const idxUpdatedBy = col['UpdatedBy'];
  const idxStatus = col['Status'];
  const idxComment = (col['Comment'] != null) ? col['Comment'] : ((col['Komentarz'] != null) ? col['Komentarz'] : null);

  if (idxWorkdayId == null || idxWorkerName == null || idxStartAt == null || idxEndAt == null || idxDuration == null){
    throw new Error('Workday_Log: brakuje wymaganych kolumn (WorkdayID/WorkerName/StartAt/EndAt/DurationSec).');
  }

  const dayKey = _wt_toLocalDateKey_(startAt);
  if (!dayKey) throw new Error('Nie mogę ustalić dnia (StartAt).');

  // Blokada duplikatu dnia
  const lastRow = sh.getLastRow();
  if (lastRow > 1){
    const data = sh.getRange(2, 1, lastRow - 1, hi.lastCol).getValues();
    for (let i=0;i<data.length;i++){
      const r = data[i];
      const wn = String(r[idxWorkerName] || '').trim();
      if (wn !== workerName) continue;
      const sIso = String(r[idxStartAt] || '').trim();
      if (!sIso) continue;
      const dk = _wt_toLocalDateKey_(sIso);
      if (dk && dk === dayKey){
        throw new Error('Taki dzień już istnieje dla pracownika (' + dayKey + '). Użyj edycji.');
      }
    }
  }

  const nowIso = new Date().toISOString();
  const newId = Utilities.getUuid();

  const row = new Array(hi.lastCol).fill('');
  row[idxWorkdayId] = newId;
  row[idxWorkerName] = workerName;
  row[idxStartAt] = startAt;
  row[idxEndAt] = endAt;
  row[idxDuration] = workSec;
  if (idxPauseTotal != null) row[idxPauseTotal] = breakSec;
  if (idxStatus != null) row[idxStatus] = 'CLOSED';
  if (idxComment != null) row[idxComment] = comment;

  // domyślnie brak ACK
  if (col['WT_AlertAck'] != null) row[col['WT_AlertAck']] = false;
  if (col['WT_AlertAckAt'] != null) row[col['WT_AlertAckAt']] = '';
  if (col['WT_AlertAckBy'] != null) row[col['WT_AlertAckBy']] = '';

  if (idxUpdatedAt != null) row[idxUpdatedAt] = nowIso;
  if (idxUpdatedBy != null) row[idxUpdatedBy] = String(payload.editorName || u.name || u.login || '');

  sh.appendRow(row);
  SpreadsheetApp.flush();

  return { ok:true, workdayId:newId, dayKey: dayKey };
}


function api_wt_updateDay(payload) {
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || '').trim());

  const workdayId = String(payload.workdayId || '').trim();
  if (!workdayId) throw new Error('Brak workdayId.');

  let startAt = String(payload.startAt || '').trim();
  let endAt = String(payload.endAt || '').trim();
  let workSec = _wt_safeInt_(payload.workSec, 0);
  let breakSec = _wt_safeInt_(payload.breakSec, 0);
  const comment = String(payload.comment || '').trim();

  // walidacja i wyliczenia
  const ds = _wt_isoToDate_(startAt);
  const de = _wt_isoToDate_(endAt);

  if (ds && de) {
    workSec = Math.max(0, Math.round((de.getTime() - ds.getTime()) / 1000));
  } else if (ds && workSec > 0 && !de) {
    endAt = new Date(ds.getTime() + workSec * 1000).toISOString();
  } else if (de && workSec > 0 && !ds) {
    startAt = new Date(de.getTime() - workSec * 1000).toISOString();
  }

  // ponowna normalizacja
  const ds2 = _wt_isoToDate_(startAt);
  const de2 = _wt_isoToDate_(endAt);
  if (!ds2 || !de2) throw new Error('Podaj poprawny Start i Koniec (ISO).');

  workSec = Math.max(0, Math.round((de2.getTime() - ds2.getTime()) / 1000));
  if (workSec > 24 * 3600) throw new Error('Czas pracy > 24h — sprawdź dane.');

  breakSec = Math.max(0, breakSec);
  if (breakSec > 12 * 3600) throw new Error('Przerwa > 12h — sprawdź dane.');

  const sh = _wt_getWorkdaySheet_();
  let hi = _wt_getHeaderMap_(sh);
  const needCols = ['WT_AlertAck','WT_AlertAckAt','WT_AlertAckBy','UpdatedAt','UpdatedBy','Status'];
  if (hi.map['Comment'] == null && hi.map['Komentarz'] == null) needCols.push('Comment');
  hi = _wt_ensureCols_(sh, needCols);

  const col = hi.map;
  const idxWorkdayId = col['WorkdayID'];
  const idxWorkerName = col['WorkerName'];
  const idxStartAt = col['StartAt'];
  const idxEndAt = col['EndAt'];
  const idxDuration = col['DurationSec'];
  const idxPauseTotal = col['PauseTotalSec'];

  const idxUpdatedAt = col['UpdatedAt'];
  const idxUpdatedBy = col['UpdatedBy'];

  const idxStatus = col['Status'];
  const idxComment = (col['Comment'] != null) ? col['Comment'] : ((col['Komentarz'] != null) ? col['Komentarz'] : null);
  if (idxWorkdayId == null || idxWorkerName == null || idxStartAt == null || idxEndAt == null || idxDuration == null) {
    throw new Error('Workday_Log: brakuje wymaganych kolumn (WorkdayID/WorkerName/StartAt/EndAt/DurationSec).');
  }

  const lastRow = sh.getLastRow();
  if (lastRow <= 1) throw new Error('Workday_Log jest pusty.');

  const data = sh.getRange(2, 1, lastRow - 1, hi.lastCol).getValues();

  // znajdź rekord bazowy po WorkdayID
  let base = null;
  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    const wid = String(r[idxWorkdayId] || '').trim();
    if (wid === workdayId) {
      base = {
        rowNumber: 2 + i,
        workerName: String(r[idxWorkerName] || '').trim(),
        startAtOld: String(r[idxStartAt] || '').trim(),
      };
      break;
    }
  }
  if (!base) throw new Error('Nie znaleziono WorkdayID: ' + workdayId);
  if (!base.workerName) throw new Error('Workday_Log: pusty WorkerName dla WorkdayID: ' + workdayId);

  const dayKeyOld = _wt_toLocalDateKey_(base.startAtOld);
  const dayKeyNew = _wt_toLocalDateKey_(startAt);
  const dayKeys = [];
  if (dayKeyOld) dayKeys.push(dayKeyOld);
  if (dayKeyNew && dayKeyNew !== dayKeyOld) dayKeys.push(dayKeyNew);
  const dayKey = dayKeys[0] || '';
  if (!dayKey) throw new Error('Nie mogę ustalić dnia (StartAt puste/niepoprawne).');

  // zbierz wszystkie wiersze tego samego dnia dla tego pracownika
  const group = [];
  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    const wn = String(r[idxWorkerName] || '').trim();
    if (!wn || wn !== base.workerName) continue;

    const sIso = String(r[idxStartAt] || '').trim();
    if (!sIso) continue;

    const dk = _wt_toLocalDateKey_(sIso);
    if (dayKeys.length && dayKeys.indexOf(dk) === -1) continue;

    group.push({
      rowNumber: 2 + i,
      workdayId: String(r[idxWorkdayId] || '').trim(),
      startAt: sIso,
      endAt: String(r[idxEndAt] || '').trim(),
    });
  }

  // Upewnij się, że w grupie jest nasz bazowy workdayId (nawet jeśli StartAt było puste)
  if (!group.some(g => g.workdayId === workdayId)) {
    group.push({
      rowNumber: base.rowNumber,
      workdayId: workdayId,
      startAt: base.startAtOld || startAt,
      endAt: String(sh.getRange(base.rowNumber, idxEndAt + 1).getValue() || '').trim(),
    });
  }

  const nowIso = new Date().toISOString();

  // Zapis: bazowy wpis = wartości z edytora, reszta = clamp + PauseTotalSec=0
  for (let j = 0; j < group.length; j++) {
    const g = group[j];

    let newStartIso = g.startAt;
    let newEndIso = g.endAt;
    let newPause = 0;

    if (g.workdayId === workdayId) {
      newStartIso = startAt;
      newEndIso = endAt;
      newPause = breakSec;
    } else {
      // Inne wpisy z tego samego dnia: wymuś zakres edytowany,
      // żeby agregacja (min/max) zawsze odzwierciedlała ręczną edycję.
      newStartIso = startAt;
      newEndIso = endAt;
      newPause = 0;
    }

    // jeżeli po przycięciu End < Start → ustaw End = Start
    const dsRow = _wt_isoToDate_(newStartIso);
    const deRow = _wt_isoToDate_(newEndIso);
    if (dsRow && deRow && deRow.getTime() < dsRow.getTime()) {
      newEndIso = newStartIso;
    }

    // duration dla tego wiersza
    let durSec = 0;
    const dsRow2 = _wt_isoToDate_(newStartIso);
    const deRow2 = _wt_isoToDate_(newEndIso);
    if (dsRow2 && deRow2) durSec = Math.max(0, Math.round((deRow2.getTime() - dsRow2.getTime()) / 1000));

    // zapis
    sh.getRange(g.rowNumber, idxStartAt + 1).setValue(newStartIso);
    sh.getRange(g.rowNumber, idxEndAt + 1).setValue(newEndIso);
    sh.getRange(g.rowNumber, idxDuration + 1).setValue(durSec);


    // Status zawsze CLOSED (ręczna edycja dnia)
    if (idxStatus != null) {
      sh.getRange(g.rowNumber, idxStatus + 1).setValue('CLOSED');
    }
    if (idxComment != null) {
      sh.getRange(g.rowNumber, idxComment + 1).setValue(comment);
    }
    if (idxPauseTotal != null) {
      sh.getRange(g.rowNumber, idxPauseTotal + 1).setValue(newPause);
    }

    if (idxUpdatedAt != null) {
      sh.getRange(g.rowNumber, idxUpdatedAt + 1).setValue(nowIso);
    }
    if (idxUpdatedBy != null) {
      sh.getRange(g.rowNumber, idxUpdatedBy + 1).setValue(String(u.name || u.login || ''));
    }
  }

  SpreadsheetApp.flush();

  return {
    ok: true,
    workdayId: workdayId,
    updatedBy: String(u.name || u.login || ''),
    updatedRows: group.length,
    dayKey: dayKey,
    dayKeyOld: dayKeyOld || '',
    dayKeyNew: dayKeyNew || '',
  };
}


/**
 * ACK alertu przekroczenia normy — tylko ADMIN/Kierownik
 * Ustawia WT_AlertAck=TRUE oraz zapisuje czas i użytkownika.
 */
function api_wt_ackAlert(payload){
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || '').trim());

  const workdayId = String(payload.workdayId || '').trim();
  if (!workdayId) throw new Error('Brak workdayId.');

  const sh = _wt_getWorkdaySheet_();
  let hi = _wt_getHeaderMap_(sh);
  hi = _wt_ensureCols_(sh, ['WT_AlertAck','WT_AlertAckAt','WT_AlertAckBy','UpdatedAt','UpdatedBy','Status']);

  const rn = _wt_findRowByWorkdayId_(sh, workdayId, hi);
  if (!rn) throw new Error('Nie znaleziono WorkdayID: ' + workdayId);

  const idxStartAt = hi.map['StartAt'];
  const idxEndAt = hi.map['EndAt'];
  const idxAck = hi.map['WT_AlertAck'];
  const idxAckAt = hi.map['WT_AlertAckAt'];
  const idxAckBy = hi.map['WT_AlertAckBy'];

  const idxStatus = hi.map['Status'];
  const row = sh.getRange(rn, 1, 1, hi.lastCol).getValues()[0];
  const startIso = String(row[idxStartAt] || '').trim();
  const endIso = String(row[idxEndAt] || '').trim();
  const ds = _wt_isoToDate_(startIso);
  const de = _wt_isoToDate_(endIso);
  let workSec = 0;
  if (ds && de) workSec = Math.max(0, Math.round((de.getTime() - ds.getTime())/1000));

  const lvl = _wt_alertLevelFromWorkSec_(workSec);
  if (lvl <= 0){
    if (idxAck != null) sh.getRange(rn, idxAck+1).setValue(false);
    if (idxAckAt != null) sh.getRange(rn, idxAckAt+1).setValue('');
    if (idxAckBy != null) sh.getRange(rn, idxAckBy+1).setValue('');
    return { ok:true, workdayId, alertLevel:0, alertAck:false };
  }

  const nowIso = new Date().toISOString();
  if (idxAck != null) sh.getRange(rn, idxAck+1).setValue(true);
  if (idxAckAt != null) sh.getRange(rn, idxAckAt+1).setValue(nowIso);
  if (idxAckBy != null) sh.getRange(rn, idxAckBy+1).setValue(String(u.name || u.login || ''));

  return { ok:true, workdayId, alertLevel:lvl, alertAck:true, alertAckAt: nowIso, alertAckBy: String(u.name || u.login || '') };
}


/**
 * PDF — Ewidencja czasu pracy (miesięczna)
 * Generuje PDF podobny do wzoru (bez kolumn: Przerwa, Realny czas pracy).
 *
 * params:
 *  - token (required)
 *  - workerName (required)
 *  - workerType (optional, np. ADMIN)
 *  - month (optional, yyyy-MM; default: bieżący)
 *
 * return: { ok:true, fileName, b64 }
 */
function api_wt_exportEwidencjaPdf(params){
  params = params || {};
  _authRequireRead_(String(params.token || '').trim());

  const tz = Session.getScriptTimeZone();
  const now = new Date();

  const workerName = String(params.workerName || '').trim();
  if (!workerName) throw new Error('Brak workerName.');

  const workerType = String(params.workerType || '').trim();

  const monthIn = String(params.month || '').trim();
  const monthLabel = (/^\d{4}-\d{2}$/.test(monthIn)) ? monthIn : Utilities.formatDate(now, tz, 'yyyy-MM');
  const y = parseInt(monthLabel.slice(0,4), 10);
  const m = parseInt(monthLabel.slice(5,7), 10);
  const monthFrom = monthLabel + '-01';
  const monthTo = Utilities.formatDate(new Date(y, m, 0), tz, 'yyyy-MM-dd');
  const lastDay = parseInt(monthTo.slice(8,10), 10);

  // pobierz dni z backendu (agregacja po min/max)
  const res = api_wt_listDays({
    token: String(params.token || '').trim(),
    workerName: workerName,
    fromIso: monthFrom,
    toIso: monthTo,
    page: 1,
    pageSize: 200
  });

  const items = (res && res.items) ? res.items : [];
  const map = {};
  for (let i=0;i<items.length;i++){
    const it = items[i];
    if (it && it.dateKey) map[String(it.dateKey)] = it;
  }

  function _fmtDateDM_(ymd){
    // ymd: yyyy-MM-dd
    if (!ymd || String(ymd).length < 10) return '';
    const dd = ymd.slice(8,10);
    const mm2 = ymd.slice(5,7);
    return dd + '.' + mm2;
  }

  function _fmtRange_(isoA, isoB){
    const da = _wt_isoToDate_(isoA);
    const db = _wt_isoToDate_(isoB);
    if (!da || !db) return '';
    const a = Utilities.formatDate(da, tz, 'H:mm');
    const b = Utilities.formatDate(db, tz, 'H:mm');
    return a + ' - ' + b;
  }

  function _fmtDurHm_(sec){
    const s = Math.max(0, _wt_safeInt_(sec, 0));
    const h = Math.floor(s / 3600);
    const m2 = Math.floor((s % 3600) / 60);
    if (h <= 0 && m2 <= 0) return '';
    if (m2 === 0) return h + 'h';
    return h + 'h ' + m2 + 'm';
  }

  function _escape_(s){
    return String(s || '')
      .replace(/&/g,'&amp;')
      .replace(/</g,'&lt;')
      .replace(/>/g,'&gt;')
      .replace(/"/g,'&quot;')
      .replace(/'/g,'&#039;');
  }

  let totalSec = 0;

  let rowsHtml = '';
  for (let d=1; d<=lastDay; d++){
    const day = String(d).padStart(2,'0');
    const ymd = monthLabel + '-' + day;
    const it = map[ymd];

    let zd = '';
    let dur = '';
    if (it){
      const startIso = String(it.startAt || '').trim();
      const endIso = String(it.endAt || '').trim();

      if (!startIso || !endIso){
        zd = 'Uzupełnij status';
      } else {
        zd = _fmtRange_(startIso, endIso);
        dur = _fmtDurHm_(it.workSec || 0);
        totalSec += Math.max(0, _wt_safeInt_(it.workSec, 0));
      }
    }

    rowsHtml += '<tr>' +
      '<td class="c-date">' + _escape_(_fmtDateDM_(ymd)) + '</td>' +
      '<td class="c-ev">' + _escape_(zd) + '</td>' +
      '<td class="c-work">' + _escape_(dur) + '</td>' +
    '</tr>';
  }

  const titleLeft = 'Ewidencja czasu pracy - ' + workerName + (workerType ? ' (' + workerType + ')' : '');
  const rangeText = Utilities.formatDate(new Date(y, m-1, 1), tz, 'dd.MM.yyyy') + ' - ' + Utilities.formatDate(new Date(y, m-1, lastDay), tz, 'dd.MM.yyyy');
  const totalText = _fmtDurHm_(totalSec);
  const genAt = Utilities.formatDate(now, tz, 'yyyy-MM-dd HH:mm:ss');

  const html = `<!doctype html>
<html>
<head>
<meta charset="utf-8">
<style>
  body{ font-family: Arial, sans-serif; font-size: 11px; color:#111; }
  .top{ width:100%; border-collapse:collapse; margin-bottom:8px; }
  .top td{ padding:2px 0; }
  .title{ font-size: 15px; font-weight: 700; }
  .range{ font-size: 12px; font-weight: 700; text-align:right; }

  table.main{ width:100%; border-collapse: collapse; table-layout: fixed; }
  table.main th, table.main td{ border:1px solid #111; padding:6px 8px; }
  table.main thead th{ background:#efefef; font-weight:700; text-align:center; }
  td.c-date{ width:70px; font-weight:700; text-align:left; }
  td.c-ev{ width:auto; text-align:center; }
  td.c-work{ width:140px; text-align:center; font-weight:700; }

  tfoot td{ border:1px solid #111; padding:8px; font-weight:700; }
  tfoot td.sum-l{ text-align:left; }
  tfoot td.sum-r{ text-align:center; }

  .footer{ margin-top:14px; display:flex; justify-content:space-between; align-items:center; font-size:10px; color:#333; }
</style>
</head>
<body>
  <table class="top">
    <tr>
      <td class="title">${_escape_(titleLeft)}</td>
      <td class="range">${_escape_(rangeText)}</td>
    </tr>
  </table>

  <table class="main">
    <thead>
      <tr>
        <th style="width:70px"></th>
        <th>Zdarzenia</th>
        <th style="width:140px">Czas pracy</th>
      </tr>
    </thead>
    <tbody>
      ${rowsHtml}
    </tbody>
    <tfoot>
      <tr>
        <td class="sum-l">W sumie</td>
        <td></td>
        <td class="sum-r">${_escape_(totalText)}</td>
      </tr>
    </tfoot>
  </table>

  <div class="footer">
    <div>Wygenerowano w Best Clean / FIDO iClean - ${_escape_(genAt)}</div>
    <div>1 z 1</div>
  </div>
</body>
</html>`;

  const pdfBlob = HtmlService.createHtmlOutput(html).getBlob().getAs(MimeType.PDF);

  function _sanitizeFilePart_(s){
    const v = String(s || '').trim();
    if (!v) return '';
    try{
      return v.normalize('NFD').replace(/[\u0300-\u036f]/g,'').replace(/[^\w]+/g,'_').replace(/^_+|_+$/g,'');
    } catch(e){
      return v.replace(/[^\w]+/g,'_').replace(/^_+|_+$/g,'');
    }
  }

  const mmYyyy = monthLabel.slice(5,7) + '_' + monthLabel.slice(0,4);
  const fileName = 'Ewidencja_pracy_' + _sanitizeFilePart_(mmYyyy) + '_' + _sanitizeFilePart_(workerName) + '.pdf';
  pdfBlob.setName(fileName);

  const b64 = Utilities.base64Encode(pdfBlob.getBytes());

  return { ok:true, fileName:fileName, b64:b64 };
}
