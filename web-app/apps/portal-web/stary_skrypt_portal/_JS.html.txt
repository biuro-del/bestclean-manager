<script>
  // Hydrate hidden views from inert <template> blocks.
  // This prevents malformed HTML in a hidden view from breaking the whole layout at parse time.
  (function () {
    function hydrateViewsFromTemplates() {
      document.querySelectorAll('template[id^="tpl-view-"]').forEach(function (tpl) {
        var sid = tpl.id.replace(/^tpl-/, '');
        var section = document.getElementById(sid);
        if (!section) return;
        if (section.children && section.children.length > 0) return;
        section.appendChild(tpl.content.cloneNode(true));
      });
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', hydrateViewsFromTemplates);
    } else {
      hydrateViewsFromTemplates();
    }
  })();
</script>

    try{
      if(!v) return '';
      var d = (v instanceof Date) ? v : new Date(v);
      if (isNaN(d.getTime())) return String(v).trim();
      var y = d.getFullYear();
      var m = String(d.getMonth()+1).padStart(2,'0');
      var day = String(d.getDate()).padStart(2,'0');
      return y + '-' + m + '-' + day;
    }catch(e){
      try{ return String(v).trim(); }catch(_e){ return ''; }

      const mm = Math.floor(s / 60);
      const ss = s % 60;
      const mmStr = (mm < 10 ? "0" : "") + mm;
      const ssStr = (ss < 10 ? "0" : "") + ss;

    function clearIdle(){

        idleTimer = null;
      }
    }

    function clearCountdown(){
      if (countdownTimer){
        clearInterval(countdownTimer);
        countdownTimer = null;
      }
      countdownActive = false;
      remainingSec = 0;
      hideTimer();
    }

    function scheduleIdle(){
      clearIdle();
      if (!getToken()) return;
      idleTimer = setTimeout(beginCountdown, IDLE_BEFORE_COUNTDOWN_MS);
    }

    function beginCountdown(){
      if (!getToken()) return;
      clearCountdown();
      remainingSec = COUNTDOWN_TOTAL_SEC;
      countdownActive = true;
      showTimer();

      countdownTimer = setInterval(()=>{
        // jeżeli user został wylogowany gdzieś indziej
        if (!getToken()){
          clearCountdown();
          return;
        }
        remainingSec -= 1;
        showTimer();

        if (remainingSec <= 0){
          clearCountdown();
          // wyloguj
          doLogout();
        }
      }, 1000);
    }

    function onUserActivity(){
      // lekki throttling, żeby nie resetować 1000x na sekundę (mousemove)
      const now = Date.now();
      if (now - lastTouch < 800) return;
      lastTouch = now;

      if (!getToken()) return;

      if (countdownActive){
        clearCountdown();
      }
      scheduleIdle();
    }

    function addListeners(){
      if (listenersAdded) return;
      listenersAdded = true;

      });

  function _dashTimeAgo_(iso){
    if(!iso) return "—";
    const t = new Date(iso).getTime();
    if(!isFinite(t)) return "—";
    const diff = Date.now() - t;
    const sec = Math.max(0, Math.round(diff/1000));
    if (sec < 60) return sec + "s temu";
    const min = Math.round(sec/60);
    if (min < 60) return min + "m temu";
    const h = Math.round(min/60);
    if (h < 24) return h + "h temu";
    const d = Math.round(h/24);
    return d + "d temu";
  }

  function _dashGotoEvents_(w){
    // Prefill for Events module
    try{
      window.__EV_PREFILL__ = { workerId: w.workerId || "", workerName: w.workerName || "" };
    }catch(e){}

    // Go to events
    if (typeof window.go === "function") window.go("events");

    // Try to apply worker filter in Events view (best-effort)
    setTimeout(()=>{
      const p = window.__EV_PREFILL__ || {};
      const name = (p.workerName || "").trim();
      const id = (p.workerId || "").trim();

      // 1) If module exposes a method
      try{
        const M = window.EventsModule || {};
        const candidates = ["setWorker","selectWorker","filterByWorker","openForWorker","prefillWorker","setWorkerFilter"];
        for (const fn of candidates){
          if (typeof M[fn] === "function"){
            M[fn](id || name);
            return;
          }
        }
      }catch(e){}

      // 2) Heuristic: find a select/input in #view-events and set value
      const root = document.getElementById("view-events");
      if(!root) return;

      // Try selects first (options matching name)
      const selects = Array.from(root.querySelectorAll("select"));
      for (const sel of selects){
        const opts = Array.from(sel.options || []);
        const hit = opts.find(o => (o.value && String(o.value).trim() === id) || (o.text && String(o.text).trim() === name));
        if (hit){
          sel.value = hit.value;
          sel.dispatchEvent(new Event("change", {bubbles:true}));
          return;
        }
        // fallback: contains name
        const hit2 = opts.find(o => (o.text && String(o.text).toLowerCase().includes(name.toLowerCase())));
        if (hit2 && name){
          sel.value = hit2.value;
          sel.dispatchEvent(new Event("change", {bubbles:true}));
          return;
        }
      }

      // Try text inputs (search)

  function _dashGotoWorkerTime_(w){
    try{
      window.__WT_PREFILL__ = { workerId: w.workerId || "", workerName: w.workerName || "" };
    }catch(e){}

    // Przejdź do czasu pracy
    if (typeof window.go === "function") window.go("workerTime");

    // Spróbuj ustawić pracownika w module (jeśli jest API) lub heurystycznie w DOM
    setTimeout(()=>{
      const p = window.__WT_PREFILL__ || {};
      const name = p.workerName || "";
      const id = p.workerId || "";

      // 1) jeśli moduł udostępnia metodę
      try{
        const M = window.WorkerTimeModule || {};
        const candidates = ["setWorker","selectWorker","openForWorker","prefillWorker"];
        for (const fn of candidates){
          if (typeof M[fn] === "function"){
            M[fn](id || name);
            return;
          }
        }
      }catch(e){}

      // 2) heurystyka: znajdź select w view-workerTime i ustaw opcję pasującą
      const root = document.getElementById("view-workerTime");
      if(!root) return;

      const selects = Array.from(root.querySelectorAll("select"));
      for (const sel of selects){
        const opts = Array.from(sel.options || []);
        // exact match on text
        let opt = opts.find(o => String(o.textContent||"").trim() === name);
        if (!opt && id) opt = opts.find(o => String(o.value||"").trim() === id);
        if (opt){
          sel.value = opt.value;
          sel.dispatchEvent(new Event("change", {bubbles:true}));
          return;
        }
      }

      // 3) heurystyka: input search
      const inp = root.querySelector('input[type="search"], input[placeholder*="prac"], input[placeholder*="Prac"], input[name*="worker"], input[id*="worker"]');
      if (inp && name){
        inp.value = name;
        inp.dispatchEvent(new Event("input", {bubbles:true}));
      }
    }, 60);
  }

  function _ymNow_(){
    const d = new Date();
    const mm = String(d.getMonth() + 1).padStart(2, "0");
    return String(d.getFullYear()) + "-" + mm;
  }

  function _pickMonthFromView_(view){
    // jeśli kiedyś dodasz select miesiąca na liście, automatycznie go wykorzystamy
    const sel = view ? view.querySelector("select") : null;
    if (!sel) return "";
    const v = String(sel.value || "").trim();
    return (/^\d{4}-\d{2}$/.test(v)) ? v : "";

  function _isIdLike_(t){
    const s = String(t || "").trim();
    // przykłady: w001, W010, BC0001, OBJ0001
    if (/^[a-zA-Z]{1,4}\d{3,}$/i.test(s)) return true;
    return false;
  }

  function _findRowFromButton_(btn, view){
    if (!btn) return null;

    // 1) typowe przypadki
    let row = btn.closest(".workers-row") || btn.closest("tr") || btn.closest("[role='row']");
    if (row) return row;

    // 2) heurystyka: idź w górę po rodzicach i szukaj elementu,
    // który wygląda jak wiersz (ma kilka dzieci, pierwsze przypomina ID)
    let el = btn;
    for (let i = 0; i < 10; i++){
      el = el.parentElement;
      if (!el) break;
      if (el === view) break;

      const kids = Array.from(el.children || []).filter(n => n && n.nodeType === 1);
      if (kids.length >= 3){
        const t0 = String(kids[0].textContent || "").trim();
        // nagłówek tabeli ma zwykle tekst "ID" — pomijamy
        if (t0.toLowerCase() === "id") continue;
        if (_isIdLike_(t0)) return el;
      }
    }

    return null;
  }

  function _extractWorkerNameFromRow_(rowEl){
    if (!rowEl) return "";

    // tabela
    if (rowEl.tagName && rowEl.tagName.toLowerCase() === "tr"){
      const tds = Array.from(rowEl.querySelectorAll("td"));
      if (tds.length >= 2){
        const name = String(tds[1].textContent || "").trim();
        return name;
      }
    }

    // div/grid: dzieci = komórki
    const kids = Array.from(rowEl.children || []).filter(n => n && n.nodeType === 1);
    if (kids.length >= 2){
      // zakładamy: 0 = ID, 1 = Imię i nazwisko
      const name = String(kids[1].textContent || "").trim();
      if (name) return name;
    }

    // fallback: szukaj najdłuższego tekstu niebędącego ID / typem / "Pokaż"
    const texts = [];
    const elems = rowEl.querySelectorAll("*");
    elems.forEach(e=>{
      const t = String(e.textContent || "").trim();
      if (!t) return;
      const tl = t.toLowerCase();
      if (tl === "pokaż" || tl === "pokaz") return;
      if (tl === "admin" || tl === "kierownik" || tl === "koordynator") return;
      if (_isIdLike_(t)) return;
      texts.push(t);
    });
    texts.sort((a,b)=>b.length-a.length);
    return texts[0] || "";
  }

  function _getWorkerRowsFromView_(view){
    if (!view) return [];

    const btns = Array.from(view.querySelectorAll("button"));
    const pokaBtns = btns.filter(b=>{
      const t = String(b.textContent || "").trim().toLowerCase();
      return t === "pokaż" || t === "pokaz";
    });

    const rows = [];
    for (const b of pokaBtns){
      const r = _findRowFromButton_(b, view);
      if (r && !rows.includes(r)) rows.push(r);
    }

    // fallback: jeśli jest tabela, bierz wszystkie TR oprócz nagłówka
    if (!rows.length){
      const trs = Array.from(view.querySelectorAll("tr"));
      trs.forEach(tr=>{
        const t0 = String((tr.children && tr.children[0] ? tr.children[0].textContent : "") || "").trim();
        if (t0.toLowerCase() === "id") return;
        if (_isIdLike_(t0)) rows.push(tr);
      });
    }

    return rows;
  }

  
  function _normNameKey_(s){
    const v = String(s || "").trim().toLowerCase().replace(/\s+/g, " ");
    // usuń diakrytyki (np. Rafał -> rafal)
    try{
      return v.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    } catch(e){
      return v;
    }
  }

  function _wtHasAlert_(row){
    if (!row) return false;
    if (row.classList && row.classList.contains("wt-worker-alert")) return true;
    // fallback: jeśli klasa była na komórce
    try{
      if (row.querySelector && row.querySelector(".wt-worker-alert")) return true;
    }catch(e){}
    return false;
  }

  function _wtApplySortMode_(){
    const view = document.getElementById("view-workerTime");
    if (!view) return;

    // tylko gdy widok jest widoczny
    try{
      const disp = window.getComputedStyle(view).display;
      if (disp === "none") return;
    }catch(e){}

    const mode = _wtGetSortMode_();
    if (!mode) return;

    const rows = _getWorkerRowsFromView_(view);
    if (!rows || !rows.length) return;

    // reset display
    rows.forEach(r=>{ try{ r.style.display = ""; }catch(e){} });

    // sort key
    function keyOf(row){
      const nm = _extractWorkerNameFromRow_(row);
      return _normNameKey_(nm || "");
    }

    let list = rows.slice();

    if (mode === "alerts"){
      list = list.filter(_wtHasAlert_);
    }

    list.sort((a,b)=>{
      const ka = keyOf(a);
      const kb = keyOf(b);
      if (ka < kb) return -1;
      if (ka > kb) return 1;
      return 0;
    });

    // ukryj niepasujące (alerts)
    if (mode === "alerts"){
      rows.forEach(r=>{
        if (!_wtHasAlert_(r)){
          try{ r.style.display = "none"; }catch(e){}
        }
      });
    }

    // re-order in DOM
    const first = rows[0];
    const parent = first ? first.parentElement : null;
    if (!parent) return;

    // div/grid: keep header on top
    const head = parent.querySelector && parent.querySelector(".workers-head");
    if (head && head.parentElement === parent){
      // upewnij się że head jest pierwszy
      try{ parent.insertBefore(head, parent.firstChild); }catch(e){}
    }

    // przenieś tylko widoczne (list)
    list.forEach(r=>{
      try{ parent.appendChild(r); }catch(e){}
    });
  }


function _ensureWorkersAlertMap_(token, month, cb){
    if (window.__wtWorkersAlertMap && window.__wtWorkersAlertMapMonth === month){
      cb(window.__wtWorkersAlertMap);
      return;
    }

    google.script.run
      .withSuccessHandler((res)=>{
        const rawMap = (res && res.map) ? res.map : {};
        const normMap = {};
        Object.keys(rawMap).forEach(k=>{
          const nk = _normNameKey_(k);
          if (!nk) return;
          const lvl = Number(rawMap[k] || 0);
          if (!normMap[nk] || lvl > normMap[nk]) normMap[nk] = lvl;
        });

        window.__wtWorkersAlertMap = normMap;
        window.__wtWorkersAlertMapMonth = month;

        cb(normMap);
      })
      .withFailureHandler((err)=>{
        try{ console.log("api_wt_workersAlertMap error", err); }catch(e){}
        if (!window.__wtWorkersAlertMapWarned){
          window.__wtWorkersAlertMapWarned = true;
          try{ alert("Nie mogę pobrać mapy alertów czasu pracy (sprawdź Console)."); }catch(e){}
        }
        cb({});
      })
      .api_wt_workersAlertMap({ token: token, month: month });
  }

          const n = _extractWorkerNameFromRow_(r);
          if (!n) return;
          const nk = _normNameKey_(n);
          const lvl = map[nk] || 0;

          if (lvl > 0){
            // wymaganie: na liście pracowników zawsze CZERWONY, jeśli jest jakikolwiek alert
            if (r.classList) r.classList.add("wt-worker-alert");
          }
        });

        // po odświeżeniu alertów — ponownie zastosuj sortowanie / filtrowanie
        try{ _wtRefreshSortBtn_(); }catch(e){}
        try{ _wtApplySortMode_(); }catch(e){}
      });

    } catch(e){
      try{ console.log("applyWorkerTimeListAlerts error", e); }catch(_){}
    }
  }

  function scheduleWorkerTimeListAlerts(){

    setTimeout(applyWorkerTimeListAlerts, 200);
    setTimeout(applyWorkerTimeListAlerts, 700);
    setTimeout(applyWorkerTimeListAlerts, 1400);

    // obserwator zmian DOM (np. po filtrach / odśwież)

/* BC_LOADING_PATCH: global loading + auto wrapper for google.script.run */
(function(){
  var el = null;
  var pending = 0;

  function getEl(){
    if (el) return el;
    el = document.getElementById("bcGlobalLoading");
    return el;
  }

  // Do not show global loading overlay when user is on Dashboard (Pulpit)
  function shouldShowOnThisView(){
    try{
      var dash = document.getElementById("view-dashboard");
      if (!dash) return true;
      var cs = window.getComputedStyle ? getComputedStyle(dash) : null;
      var visible = cs ? (cs.display !== "none" && cs.visibility !== "hidden" && cs.opacity !== "0") : (dash.style.display !== "none");
      if (visible) return false; // dashboard active -> suppress loading overlay
    }catch(e){}
    return true;
  }

  function show(){
    if (!shouldShowOnThisView()) return;
    var e = getEl();
    if (!e) return;
    pending++;
    e.classList.remove("hidden");
    e.setAttribute("aria-hidden","false");
  }

  function hide(){
    var e = getEl();
    if (!e) return;
    pending = Math.max(0, pending - 1);
    if (pending === 0){
      e.classList.add("hidden");
      e.setAttribute("aria-hidden","true");
    }
  }

  window.BCLoading = window.BCLoading || { show: show, hide: hide };

  function patchGAS(){
    try{
      if (!window.google || !google.script || !google.script.run) return;
      if (google.script.run.__bc_patched) return;

      function wrapRunner(runner){
        return new Proxy(runner, {
          get: function(target, prop){
            var v = target[prop];

            if (prop === "withSuccessHandler"){
              return function(fn){
                return wrapRunner(target.withSuccessHandler(function(res){
                  try{ fn && fn(res); } finally { hide(); }
                }));
              };
            }
            if (prop === "withFailureHandler"){
              return function(fn){
                return wrapRunner(target.withFailureHandler(function(err){
                  try{ fn && fn(err); } finally { hide(); }
                }));
              };
            }

            if (typeof v === "function"){
              return function(){
                show();
                return v.apply(target, arguments);
              };
            }

            return v;
          }
        });
      }

      var proxied = wrapRunner(google.script.run);
      proxied.__bc_patched = true;
      google.script.run = proxied;
      google.script.run.__bc_patched = true;
    }catch(e){}
  }

  if (document.readyState === "loading"){
    document.addEventListener("DOMContentLoaded", patchGAS);
  } else {
    patchGAS();
  }
})();
  window.AUTO_LOGOUT = AUTO_LOGOUT;

</script>
