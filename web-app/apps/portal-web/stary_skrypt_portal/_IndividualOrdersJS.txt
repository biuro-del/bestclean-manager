<script>
/* Klienci -> Zlecenie indywidualne (Klient_Ind) — CRUD + Pobieranie QR */
(function(){
  function $(sel){ return document.querySelector(sel); }
  function esc(s){
    return String(s ?? "").replace(/[&<>"']/g, m=>({ "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;" }[m]));
  }

  let __io_req = 0;
    let __io_originalId = "";
const API = {
    list: "api_listIndividualOrders",
    create: "api_createIndividualOrder",
    update: "api_updateIndividualOrder",
    del: "api_deleteIndividualOrder",
  };

  function setStatus(msg){
    const el = $("#ioStatus");
    if (el) el.textContent = "Status: " + String(msg || "");
  }

  function setFooter(shown, total, limit){
    const el = $("#ioFooter");
    if (el) el.textContent = `Wyświetlono: ${shown} (max ${limit}) · Wszystkie: ${total}`;
  }

  function getToken(){
    // global helper in _JS.html
    try{
      if (window.FIDO_AUTH && typeof window.FIDO_AUTH.getToken === "function"){
        return String(window.FIDO_AUTH.getToken() || "");
      }
    }catch(e){}
    // fallback from sessionStorage
    try{
      const a = JSON.parse(sessionStorage.getItem("fido_auth") || "null");
      return a && a.token ? String(a.token) : "";
    }catch(e){ return ""; }
  }

  function callServer(fnName, payload, ok, fail){
    try{
      if (!(window.google && google.script && google.script.run)){
        setStatus("BŁĄD: brak google.script.run");
        return;
      }
      const runner = google.script.run
        .withSuccessHandler(ok)
        .withFailureHandler(fail);

      const fn = runner[fnName];
      if (typeof fn !== "function"){
        console.error("[IND] Backend function missing:", fnName, "available keys:", Object.keys(runner||{}));
        setStatus("BŁĄD: backend nie ma funkcji " + fnName + " (sprawdź Code.gs + wdrożenie)");
        return;
      }
      return fn.call(runner, payload);
    }catch(e){
      console.error("[IND] callServer exception:", e);
      setStatus("BŁĄD callServer: " + (e && (e.message||e.toString()) ? (e.message||e.toString()) : "nieznany"));
    }
  }

  function collectFilters(){
    return {
      q: $("#ioQ") ? ($("#ioQ").value || "") : ""
    };
  }

  // ---------- QR download ----------

  // pending QR code for modal
  let __io_qrPending = "";

  function showQrModal(show){
    const m = $("#ioQrModal");
    if (m) m.style.display = show ? "flex" : "none";
  }
  function openQrModal(qrText){
    const code = String(qrText || "").trim();
    if (!code){
      setStatus("Brak kodu QR w tym rekordzie.");
      return;
    }
    __io_qrPending = code;
    const elCode = $("#ioQrModalCode");
    const elSub  = $("#ioQrModalSub");
    if (elCode) elCode.textContent = code;
    if (elSub)  elSub.textContent  = "Wybierz format pobrania:";
    const elImg = $("#ioQrModalImg");
    if (elImg){
      elImg.alt = "QR: " + code;
      elImg.src = _qrUrl_(code, 260);
    }
    showQrModal(true);
  }
  function closeQrModal(){
    __io_qrPending = "";
    const elImg = $("#ioQrModalImg");
    if (elImg) elImg.src = "";
    showQrModal(false);
  }

  function _qrUrl_(qrText, size){
    const txt = String(qrText || "").trim();
    const s = (Number(size) && Number(size) > 0) ? Math.round(Number(size)) : 600;
    // quickchart QR (PNG)
    return "https://quickchart.io/qr?text=" + encodeURIComponent(txt) + "&size=" + s + "&margin=10&format=png";
  }

  function downloadQrPng(qrText){
    const txt = String(qrText || "").trim();
    if (!txt){
      setStatus("Brak kodu QR w tym rekordzie.");
      return;
    }

    const url = _qrUrl_(txt);
    setStatus("generuję QR…");

    fetch(url, { method: "GET", mode: "cors" })
      .then(res=>{
        if (!res.ok) throw new Error("HTTP " + res.status);
        return res.blob();
      })
      .then(blob=>{
        const a = document.createElement("a");
        const objUrl = URL.createObjectURL(blob);
        a.href = objUrl;
        a.download = txt + ".png";
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(()=>{ try{ URL.revokeObjectURL(objUrl); }catch(e){} }, 1500);
        setStatus("OK · pobrano: " + txt + ".png");
      })
      .catch(err=>{
        console.error("[IND] QR download error:", err);
        setStatus("BŁĄD pobierania QR (Console)");
        try{ alert("Nie mogę pobrać QR. Sprawdź połączenie / Console."); }catch(e){}
      });
  }
  function _ensureJsPDF_(){
    return new Promise(function(resolve, reject){
      try{
        if (window.jspdf && window.jspdf.jsPDF) return resolve(window.jspdf.jsPDF);
        const s = document.createElement("script");
        s.src = "https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js";
        s.async = true;
        s.onload = function(){
          if (window.jspdf && window.jspdf.jsPDF) resolve(window.jspdf.jsPDF);
          else reject(new Error("jsPDF not available after load"));
        };
        s.onerror = function(){ reject(new Error("jsPDF load error")); };
        document.head.appendChild(s);
      }catch(e){ reject(e); }
    });
  }

  function downloadQrPdf(qrText){
    const txt = String(qrText || "").trim();
    if (!txt){
      setStatus("Brak kodu QR w tym rekordzie.");
      return;
    }

    const url = _qrUrl_(txt);
    setStatus("generuję PDF…");

    fetch(url, { method: "GET", mode: "cors" })
      .then(res=>{
        if (!res.ok) throw new Error("HTTP " + res.status);
        return res.blob();
      })
      .then(blob=>{
        // convert to dataURL
        return new Promise(function(resolve, reject){
          const r = new FileReader();
          r.onload = function(){ resolve(String(r.result || "")); };
          r.onerror = function(){ reject(new Error("FileReader error")); };
          r.readAsDataURL(blob);
        });
      })
      .then(function(dataUrl){
        return _ensureJsPDF_().then(function(jsPDF){
          // A4, mm
          const doc = new jsPDF({ orientation: "p", unit: "mm", format: "a4" });
          const pageW = doc.internal.pageSize.getWidth();
          const pageH = doc.internal.pageSize.getHeight();

          // QR size in PDF (mm)
          const imgSize = Math.min(160, pageW - 40);
          const x = (pageW - imgSize) / 2;
          const y = 40;

          doc.setFontSize(14);
          doc.text("Kod QR: " + txt, pageW/2, 20, { align: "center" });
          doc.addImage(dataUrl, "PNG", x, y, imgSize, imgSize);
          doc.setFontSize(10);
          doc.text("Wygenerowano w Best Clean Monitor", pageW/2, pageH - 15, { align: "center" });

          doc.save(txt + ".pdf");
          setStatus("OK · pobrano: " + txt + ".pdf");
        });
      })
      .catch(err=>{
        console.error("[IND] QR PDF error:", err);
        setStatus("BŁĄD pobierania PDF (Console)");
        try{ alert("Nie mogę pobrać PDF. Sprawdź połączenie / Console."); }catch(e){}
      });
  }



  // ---------- render ----------
  function render(items){
    const box = $("#ioRows");
    if (!box) return;

    if (!items || !items.length){
      box.innerHTML = `
        <div class="io-row io-empty">
          <div>—</div><div>—</div><div>Brak wyników</div><div>—</div><div>—</div><div>—</div><div>—</div><div>—</div><div>—</div>
          <div class="io-actions"></div>
        </div>`;
      return;
    }

    box.innerHTML = items.map(it=>{
      const rowNumber = Number(it.rowNumber || 0) || 0;
      return `
        <div class="io-row" data-row="${esc(rowNumber)}" data-date="${esc(it.data || "")}" data-qrcode="${esc(it.qrCode || "")}">
          <div>${esc(it.klientIndId || "—")}</div>
          <div>${esc(it.dataFmt || it.data || "—")}</div>
          <div title="${esc(it.nazwa || "")}">${esc(it.nazwa || "—")}</div>
          <div>${esc(it.nip || "")}</div>
          <div>${esc(it.miasto || "")}</div>
          <div>${esc(it.adres || "")}</div>
          <div>${esc(it.kontakt || "")}</div>
          <div title="${esc(it.informacje || "")}">${esc(it.informacje || "")}</div>
          <div class="mono">${esc(it.qrCode || "")}</div>
          <div class="io-actions">
            <button class="mini-btn qr" data-act="qr" type="button" title="Pobierz QR">QR</button>
            <button class="mini-btn edit" data-act="edit" type="button">Edytuj</button>
            <button class="mini-btn danger" data-act="del" type="button">Usuń</button>
          </div>
        </div>
      `;
    }).join("");
  }

  // ---------- load ----------
  function load(){
    const token = getToken();
    const limit = 200;
    const f = collectFilters();

    if (!token){
      setStatus("BŁĄD: brak tokena (zaloguj się ponownie)");
      setFooter(0,0,limit);
      render([]);
      return;
    }

    const reqId = (++__io_req);
    setStatus("pobieram dane…");

    let done = false;
    const watchdog = setTimeout(function(){
      if (done) return;
      done = true;
      if (reqId !== __io_req) return;
      console.error("[IND] timeout: brak odpowiedzi z backendu");
      setStatus("TIMEOUT (brak odpowiedzi) — Console");
    }, 12000);

    function onOk(res){
      if (done) return;
      done = true;
      clearTimeout(watchdog);
      if (reqId !== __io_req) return;

      try{
        const items = (res && res.items) ? res.items : [];
        render(items);
        const total = (res && typeof res.total === "number") ? res.total : items.length;
        const lim = (res && (res.limit || res.pageSize)) ? (res.limit || res.pageSize) : limit;
        setFooter(items.length, total, lim);
        setStatus("OK · rekordów: " + items.length);
      }catch(e){
        console.error("[IND] render error:", e);
        render([]);
        setFooter(0,0,limit);
        setStatus("błąd renderowania (Console)");
      }
    }

    function onFail(err){
      if (done) return;
      done = true;
      clearTimeout(watchdog);
      if (reqId !== __io_req) return;

      console.error("[IND] api error:", err);
      render([]);
      setFooter(0,0,limit);
      setStatus("BŁĄD API (Console)");
      try{ alert("Błąd pobierania zleceń indywidualnych. Sprawdź Console."); }catch(e){}
    }

    callServer(API.list, {
      token: token,
      page: 1,
      limit: limit,
      pageSize: limit,
      q: f.q
    }, onOk, onFail);
  }

  // ---------- modal ----------
  function showModal(show){
    const m = $("#ioModal");
    if (m) m.style.display = show ? "flex" : "none";
  }
  function closeModal(){ showModal(false); }

  function openModal(mode, it){
    const isEdit = (mode === "edit");

    $("#ioFormRow").value = isEdit ? String(it.rowNumber || "") : "";
    $("#ioFormId").value = isEdit ? String(it.klientIndId || "") : "";
    __io_originalId = isEdit ? String(it.klientIndId || "").trim() : "";

    // ID: w edycji tylko Admin/Kierownik mogą zmienić KlientIndID
    const canEditId = (typeof roleLevel === "function" && typeof getUserRole === "function")
      ? (roleLevel(getUserRole()) >= 2)
      : false;
    $("#ioFormId").disabled = (isEdit && !canEditId);

    // Data: ustaw jako yyyy-MM-dd (preferuj ISO z backendu)
    const dIso = String(it.data || "").trim();
    if (dIso && /^\d{4}-\d{2}-\d{2}$/.test(dIso)){
      $("#ioFormDate").value = dIso;
    } else {
      $("#ioFormDate").value = (window._iso_ ? window._iso_(new Date()) : "");
    }

    $("#ioFormName").value = isEdit ? String(it.nazwa || "") : "";
    $("#ioFormNip").value = isEdit ? String(it.nip || "") : "";
    $("#ioFormCity").value = isEdit ? String(it.miasto || "") : "";
    $("#ioFormAddr").value = isEdit ? String(it.adres || "") : "";
    $("#ioFormContact").value = isEdit ? String(it.kontakt || "") : "";
    $("#ioFormInfo").value = isEdit ? String(it.informacje || "") : "";
    $("#ioFormQrCode").value = isEdit ? String(it.qrCode || "") : "";

    $("#ioModalTitle").textContent = isEdit ? "Edytuj zlecenie" : "Dodaj zlecenie";
    showModal(true);
  }

  function apiCreate(){
    const payload = {
      token: getToken(),
      klientIndId: ($("#ioFormId").value || "").trim(),
      data: ($("#ioFormDate").value || "").trim(),
      nazwa: ($("#ioFormName").value || "").trim(),
      nip: ($("#ioFormNip").value || "").trim(),
      miasto: ($("#ioFormCity").value || "").trim(),
      adres: ($("#ioFormAddr").value || "").trim(),
      kontakt: ($("#ioFormContact").value || "").trim(),
      informacje: ($("#ioFormInfo").value || "").trim(),
      qrCode: ($("#ioFormQrCode").value || "").trim()
    };

    const btn = $("#ioModalSave");
    if (btn){ btn.disabled = true; btn.textContent = "Zapisywanie..."; }

    callServer(API.create, payload, function(){
      if (btn){ btn.disabled = false; btn.textContent = "Zapisz"; }
      closeModal();
      load();
    }, function(err){
      if (btn){ btn.disabled = false; btn.textContent = "Zapisz"; }
      console.error(err);
      try{ alert(String(err && err.message ? err.message : err)); }catch(e){}
    });
  }

  function apiUpdate(rowNumber){
    const payload = {
      token: getToken(),
      rowNumber: rowNumber,
      klientIndId: ($("#ioFormId").value || "").trim(),
      originalKlientIndId: (__io_originalId || "").trim(),
      data: ($("#ioFormDate").value || "").trim(),
      nazwa: ($("#ioFormName").value || "").trim(),
      nip: ($("#ioFormNip").value || "").trim(),
      miasto: ($("#ioFormCity").value || "").trim(),
      adres: ($("#ioFormAddr").value || "").trim(),
      kontakt: ($("#ioFormContact").value || "").trim(),
      informacje: ($("#ioFormInfo").value || "").trim(),
      qrCode: ($("#ioFormQrCode").value || "").trim()
    };

    const btn = $("#ioModalSave");
    if (btn){ btn.disabled = true; btn.textContent = "Zapisywanie..."; }

    callServer(API.update, payload, function(){
      if (btn){ btn.disabled = false; btn.textContent = "Zapisz"; }
      closeModal();
      load();
    }, function(err){
      if (btn){ btn.disabled = false; btn.textContent = "Zapisz"; }
      console.error(err);
      try{ alert(String(err && err.message ? err.message : err)); }catch(e){}
    });
  }

  function apiDelete(rowNumber){
    callServer(API.del, { token: getToken(), rowNumber: rowNumber }, function(){
      load();
    }, function(err){
      console.error(err);
      try{ alert(String(err && err.message ? err.message : err)); }catch(e){}
    });
  }

  // ---------- bind ----------
  function bindOnce(){
    if (window.__io_bound__) return;
    const addBtn = $("#ioAdd");
    const rowsBox = $("#ioRows");
    const refreshBtn = $("#ioRefresh");

    if (!addBtn || !rowsBox || !refreshBtn){
      setStatus("brak elementów widoku (czy wgrałeś _ViewIndividualOrders?)");
      return;
    }

    window.__io_bound__ = true;
    setStatus("gotowe");

    refreshBtn.addEventListener("click", function(){ load(); });
    addBtn.addEventListener("click", function(){ openModal("add", {}); });

    const q = $("#ioQ");
    if (q && !q.__bound){
      q.__bound = true;
      q.addEventListener("keydown", function(e){
        if (e.key === "Enter") load();
      });
      // delikatny debounce na wpisywaniu
      let t = null;
      q.addEventListener("input", function(){
        clearTimeout(t);
        t = setTimeout(load, 300);
      });
    }

    rowsBox.addEventListener("click", function(e){
      const btn = e.target && e.target.closest ? e.target.closest("button") : null;
      const row = e.target && e.target.closest ? e.target.closest(".io-row") : null;
      if (!row) return;

      const rowNumber = parseInt(row.getAttribute("data-row") || "0", 10) || 0;
      const qrCode = String(row.getAttribute("data-qrcode") || "").trim();

      // 1) Klik na przycisku akcji
      if (btn){
        const act = btn.getAttribute("data-act");
        if (!act) return;

        if (act === "del"){
          if (!confirm("Usunąć ten rekord?")) return;
          apiDelete(rowNumber);
          return;
        }

        if (act === "edit"){
          const cells = row.querySelectorAll(":scope > div");
          const it = {
            rowNumber: rowNumber,
            klientIndId: (cells[0] ? cells[0].textContent.trim() : ""),
            dataFmt: (cells[1] ? cells[1].textContent.trim() : ""),
            nazwa: (cells[2] ? cells[2].textContent.trim() : ""),
            nip: (cells[3] ? cells[3].textContent.trim() : ""),
            miasto: (cells[4] ? cells[4].textContent.trim() : ""),
            adres: (cells[5] ? cells[5].textContent.trim() : ""),
            kontakt: (cells[6] ? cells[6].textContent.trim() : ""),
            informacje: (cells[7] ? cells[7].textContent.trim() : ""),
            qrCode: (cells[8] ? cells[8].textContent.trim() : "")
          };

          // data ISO nie jest w komórce (tam jest format PL), więc pobierzemy ją z data-qrcode? nie
          // Najprościej: odświeżymy listę po zapisie; w edycji zostawiamy date pole jak w rekordzie jeśli backend zwróci w load().
          // Dla pewności — spróbuj odczytać iso z dataset (jeśli kiedyś dodasz).
          it.data = String(row.getAttribute("data-date") || "").trim();

          openModal("edit", it);
          return;
        }

        if (act === "qr"){
          openQrModal(qrCode);
          return;
        }

        return;
      }

      // 2) Klik w wiersz — bez akcji (QR tylko z przycisku)
      return;
    });

    const closeBtn = $("#ioModalClose");
    const cancelBtn= $("#ioModalCancel");
    const overlay  = $("#ioModal");
    if (closeBtn) closeBtn.addEventListener("click", closeModal);
    if (cancelBtn) cancelBtn.addEventListener("click", closeModal);
    if (overlay) overlay.addEventListener("click", function(e){ if (e.target === overlay) closeModal(); });
    // --- QR format modal bindings ---
    const qrOverlay = $("#ioQrModal");
    const qrClose   = $("#ioQrModalClose");
    const qrCancel  = $("#ioQrModalCancel");
    const qrPngBtn  = $("#ioQrModalPng");
    const qrPdfBtn  = $("#ioQrModalPdf");

    if (qrClose)  qrClose.addEventListener("click", closeQrModal);
    if (qrCancel) qrCancel.addEventListener("click", closeQrModal);
    if (qrOverlay) qrOverlay.addEventListener("click", function(e){ if (e.target === qrOverlay) closeQrModal(); });

    if (qrPngBtn) qrPngBtn.addEventListener("click", function(){
      const code = String(__io_qrPending || "").trim();
      closeQrModal();
      downloadQrPng(code);
    });

    if (qrPdfBtn) qrPdfBtn.addEventListener("click", function(){
      const code = String(__io_qrPending || "").trim();
      closeQrModal();
      downloadQrPdf(code);
    });


    const saveBtn = $("#ioModalSave");
    if (saveBtn) saveBtn.addEventListener("click", function(){
      const row = String($("#ioFormRow").value || "").trim();
      const rn = parseInt(row, 10) || 0;
      if (rn) apiUpdate(rn);
      else apiCreate();
    });
  }

  function tryInit(){
    if (!document.getElementById("ioRows")) return false;
    bindOnce();
    load();
    return true;
  }

  // public
  window.IndividualOrdersModule = window.IndividualOrdersModule || {};
  window.IndividualOrdersModule.init = function(){ tryInit(); };

  document.addEventListener("DOMContentLoaded", function(){ tryInit(); });

  let tries = 0;
  const t = setInterval(function(){
    tries++;
    if (tryInit() || tries >= 20) clearInterval(t);
  }, 300);

})();
</script>
