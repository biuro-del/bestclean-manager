  function token(){
    return (window.FIDO_AUTH && typeof window.FIDO_AUTH.getToken === "function")
      ? window.FIDO_AUTH.getToken()
      : "";
  }

  function esc(s){
    return String(s ?? "")
      .replaceAll("&","&amp;")
      .replaceAll("<","&lt;")
      .replaceAll(">","&gt;")
      .replaceAll('"',"&quot;")
      .replaceAll("'","&#039;");
  }
  function norm(s){ return String(s ?? "").trim().toLowerCase(); }

  function pad2(n){ return String(n).padStart(2,"0"); }

  function getMonthRange(monthValue){
    // monthValue: "YYYY-MM"
    const mv = String(monthValue || "").trim();
    if (!/^\d{4}-\d{2}$/.test(mv)) return { fromDate:"", toDate:"" };
    const [yStr, mStr] = mv.split("-");
    const y = parseInt(yStr, 10);
    const m = parseInt(mStr, 10);
    if (!isFinite(y) || !isFinite(m) || m < 1 || m > 12) return { fromDate:"", toDate:"" };

    const fromDate = `${yStr}-${mStr}-01`;
    const lastDay = new Date(y, m, 0).getDate(); // day 0 of next month = last day of this month
    const toDate = `${yStr}-${mStr}-${pad2(lastDay)}`;
    return { fromDate, toDate };
  }

  function setDefaultMonth(){
    const el = document.getElementById("logMonth");
    if (!el) return;
    if (el.value) return;
    const d = new Date();
    const y = d.getFullYear();
    const m = pad2(d.getMonth()+1);
    el.value = `${y}-${m}`;
  }

    if (pl) pl.style.display = (which === "logs") ? "" : "none";

    if (d) d.classList.toggle("primary", which === "daily");
    if (e) e.classList.toggle("primary", which === "extra");
    if (l) l.classList.toggle("primary", which === "logs");
  }

  // ========= MODAL (normalne okno w portalu, bez prompt) =========
  function modalEls(){
    return {
      overlay: document.getElementById("chkModalOverlay"),
      title: document.getElementById("chkModalTitle"),
      meta: document.getElementById("chkModalMeta"),
      body: document.getElementById("chkModalBody"),
      foot: document.getElementById("chkModalFoot"),
      close: document.getElementById("chkModalClose")
    };
  }

  function modalOpen({title, meta, bodyHtml, buttons}){
    const m = modalEls();
    if (!m.overlay) return;

    m.title.textContent = title || "—";
    if (m.meta) m.meta.textContent = meta || "";
    if (m.body) m.body.innerHTML = bodyHtml || "";
    if (m.foot){
      m.foot.innerHTML = "";
      (buttons || []).forEach(btn=>{
        const b = document.createElement("button");
        b.type = "button";
        b.className = btn.className || "btn2";
        b.textContent = btn.label || "OK";
        b.onclick = () => btn.onClick && btn.onClick();
        m.foot.appendChild(b);
      });
    }

    m.overlay.style.display = "flex";
    m.overlay.setAttribute("aria-hidden","false");
  }

  function modalClose(){
    const m = modalEls();
    if (!m.overlay) return;
    m.overlay.style.display = "none";
    m.overlay.setAttribute("aria-hidden","true");
  }

  function bindModal(){
    const m = modalEls();
    if (!m.overlay) return;

    if (m.close) m.close.onclick = modalClose;

    m.overlay.addEventListener("click", (e)=>{
      if (e.target === m.overlay) modalClose();
    });

    document.addEventListener("keydown", (e)=>{
      if (e.key === "Escape") modalClose();
    });
  }

  function loadClients(){

  function loadZonesForClient(clientName){
    const zSel = $("#chkZoneSelect");
    if (!zSel) return;

    zSel.innerHTML = `<option value="">— wybierz —</option>`;

    if (!clientName){
      state.zones = [];
      state.currentQr = "";
      return;
    }

    api()
      .withSuccessHandler(res=>{
        const items = (res && res.items) ? res.items : [];
        state.zones = items;

        // label: Strefa (kol C), value: QrId (kol A)
        const counts = {};
        items.forEach(z=>{
          const name = String(z.strefa || "").trim();
          const k = norm(name);
          if (!k) return;
          counts[k] = (counts[k] || 0) + 1;
        });

        items.sort((a,b)=>{
          const as = norm(a.strefa), bs = norm(b.strefa);
          if (as === bs){
            const aq = norm(a.qr), bq = norm(b.qr);
            return aq < bq ? -1 : (aq > bq ? 1 : 0);
          }
          return as < bs ? -1 : 1;
        });

        items.forEach(z=>{
          const qr = String(z.qr || "").trim();
          const strefa = String(z.strefa || "").trim();
          const lok = String(z.lokalizacja || "").trim();
          if (!qr) return;

          const key = norm(strefa);
          let label = strefa || qr;

          if (strefa && counts[key] > 1) label = `${strefa} • ${qr}`;
          else if (strefa && lok) label = `${strefa} — ${lok}`;

          const opt = document.createElement("option");
          opt.value = qr;          // backend używa QrId
          opt.textContent = label; // UI pokazuje strefę z C
          zSel.appendChild(opt);
        });

        if (state.currentQr) zSel.value = state.currentQr;
        onZoneChange();
      })
      .withFailureHandler(err=>{
        console.error("Checklisty: api_listZones error", err);
        zSel.innerHTML = `<option value="">Błąd pobierania stref</option>`;
      })
      .api_listZones({ token: state.token, klient: clientName, limit: 500 });
  }

// ---------- REFRESH ----------
  function refreshDaily(){
    if (!state.currentQr){
      state.daily = [];
      state.dailyHist = [];
      renderDaily();
      return;
    }

    api()
      .withSuccessHandler(res=>{ state.daily = (res && res.items) ? res.items : []; renderDaily(); })
      .withFailureHandler(err=>{ console.error("Checklisty: daily_list error", err); state.daily=[]; renderDaily(); })
      .api_checklists_daily_list({ token: state.token, qr: state.currentQr });

    refreshDailyHistory();
  }

  function refreshDailyHistory(){
    if (!state.currentQr) return renderDailyHistory();

    api()
      .withSuccessHandler(res=>{
        state.dailyHist = (res && res.items) ? res.items : [];
        renderDailyHistory();
      })
      .withFailureHandler(err=>{
        console.error("Checklisty: daily_history error", err);
        state.dailyHist = [];
        renderDailyHistory();
      })
      .api_checklists_daily_listHistory({ token: state.token, qr: state.currentQr });
  }

  function refreshExtra(){

  function exportLogs(){
    if (!state.currentQr) return;

    setDefaultMonth();
    const mv = (document.getElementById("logMonth").value || "").trim();
    const rng = getMonthRange(mv);
    const fromDate = rng.fromDate;
    const toDate = rng.toDate;
const worker = (document.getElementById("logWorker").value || "").trim();

    api()
      .withSuccessHandler(res=>{
        const csv = (res && res.csv) ? res.csv : "";
        const filename = (res && res.filename) ? res.filename : "Checklist_Log_export.csv";
        const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = filename;
        document.body.appendChild(a);
        a.click();
        a.remove();
        setTimeout(()=> URL.revokeObjectURL(url), 5000);
      })
      .withFailureHandler(err=>{ console.error("Checklisty: export error", err); alert("Błąd eksportu CSV."); })
      .api_checklists_logs_exportCsv({ token: state.token, qr: state.currentQr, fromDate, toDate, worker });
  }

    `;
  }

  function openDailyModal(model){
    if (!state.currentQr) return alert("Najpierw wybierz strefę.");

    const isEdit = !!(model && model.TaskId);
    const title = isEdit ? "Edytuj czynność DAILY" : "Dodaj czynność DAILY";
    const meta = `Strefa (QrId): ${state.currentQr}`;

    modalOpen({
      title,
      meta,
      bodyHtml: dailyFormHtml(model),
      buttons: [
        { label: "Anuluj", className: "btn2", onClick: modalClose },
        { label: "Zapisz", className: "btn2 primary", onClick: ()=> saveDaily(model) }
      ]
    });
  }

  function saveDaily(model){
    const taskName = (document.getElementById("chkTaskName").value || "").trim();
    const required = (document.getElementById("chkRequired").value || "").trim();
    const order = (document.getElementById("chkOrder").value || "").trim();
    const activeEl = document.getElementById("chkActive");
    const active = activeEl ? String(activeEl.value || "TAK").trim() : "TAK";

    if (!taskName) return alert("Podaj nazwę zadania.");

    if (model && model.TaskId){
      api().withSuccessHandler(()=>{
          modalClose();
          refreshDaily();
        })
        .withFailureHandler(err=>{
          console.error(err);
          alert("Błąd zapisu.");
        })
        .api_checklists_daily_update({
          token: state.token,
          taskId: model.TaskId,
          taskName,
          required,
          order,
          active
        });
    } else {
      api().withSuccessHandler(()=>{
          modalClose();
          refreshDaily();
        })
        .withFailureHandler(err=>{
          console.error(err);
          alert("Błąd zapisu.");
        })
        .api_checklists_daily_create({
          token: state.token,
          qr: state.currentQr,
          taskName,
          required,
          order
        });
    }
  }

  function extraFormHtml(){
    return `
      <div class="form-grid">
        <div class="form-field span-2">
          <label>Nazwa zadania</label>
          <input id="chkExtraTaskName" type="text" placeholder="np. Uzupełnij ręczniki papierowe">
        </div>

        <div class="form-field">
          <label>Wymagane</label>
          <select id="chkExtraRequired">
            <option>TAK</option>
            <option>NIE</option>
          </select>
        </div>

        <div class="form-field">
          <label>Priorytet</label>
          <select id="chkExtraPriority">
            <option value="3">3 (wysoki)</option>
            <option value="2" selected>2 (średni)</option>
            <option value="1">1 (niski)</option>
          </select>
        </div>
      </div>

      <p class="muted" style="margin-top: 10px;">Nowe EXTRA zawsze zapisuje się jako <b>Status = OPEN</b>.</p>
    `;
  }

  function openExtraModal(){
    if (!state.currentQr) return alert("Najpierw wybierz strefę.");

    modalOpen({
      title: "Dodaj EXTRA",
      meta: `Strefa (QrId): ${state.currentQr}`,
      bodyHtml: extraFormHtml(),
      buttons: [
        { label: "Anuluj", className: "btn2", onClick: modalClose },
        { label: "Zapisz", className: "btn2 primary", onClick: saveExtra }
      ]
    });
  }

  function saveExtra(){
    const taskName = (document.getElementById("chkExtraTaskName").value || "").trim();
    const required = (document.getElementById("chkExtraRequired").value || "").trim();
    const priority = (document.getElementById("chkExtraPriority").value || "").trim();

    if (!taskName) return alert("Podaj nazwę zadania.");

    api().withSuccessHandler(()=>{
        modalClose();
        refreshExtra();
      })
      .withFailureHandler(err=>{
        console.error(err);
        alert("Błąd zapisu.");
      })
      .api_checklists_extra_create({
        token: state.token,
        qr: state.currentQr,
        taskName,
        required,
        priority
      });
  }

  function openDailyCopyModal(){
    if (!state.currentQr) return;
    const from = prompt("Podaj źródłowy QrId (from):", "");
    if (!from) return;
    const mode = prompt("Tryb: MERGE lub REPLACE", "MERGE");
    if (!mode) return;

    api()
      .withSuccessHandler(res=>{
        alert("Skopiowano: " + (res && res.copied ? res.copied : 0));
        refreshDaily();
      })
      .withFailureHandler(err=>{
        console.error(err);
        alert("Błąd kopiowania.");
      })
      .api_checklists_daily_copy({ token: state.token, fromQr: String(from).trim(), toQr: state.currentQr, mode: String(mode).trim() });
  }

  // ---------- EVENTS ----------
  function onClientChange(){
    const sel = $("#chkClientSelect");
    state.currentClient = (sel && sel.value) ? String(sel.value).trim() : "";
    state.currentQr = "";
    const zSel = $("#chkZoneSelect");
    if (zSel) zSel.value = "";
    loadZonesForClient(state.currentClient);
    // Historia klienta ładuje się od razu po wyborze klienta
    refreshClientDaily();
    refreshClientExtra();
    refreshClientLogs();
  }

  function onZoneChange(){
    const sel = $("#chkZoneSelect");
    state.currentQr = (sel && sel.value) ? String(sel.value).trim() : "";
    refreshAll();
  }

  function refreshAll(){
    refreshDaily();
    refreshExtra();
    refreshLogs(); refreshClientLogs();
  }

  function bind(){
    bindModal();

    $("#chkClientSelect").addEventListener("change", onClientChange);
    $("#chkZoneSelect").addEventListener("change", onZoneChange);
    $("#chkRefreshBtn").addEventListener("click", refreshAll);

    $("#chkTabDaily").addEventListener("click", ()=> setActiveTab("daily"));
    $("#chkTabExtra").addEventListener("click", ()=> setActiveTab("extra"));
    $("#chkTabLogs").addEventListener("click", ()=> { setActiveTab("logs"); setDefaultMonth(); refreshLogs(); refreshClientLogs(); });

    $("#dailyAddBtn").addEventListener("click", ()=> openDailyModal(null));
    $("#dailyCopyBtn").addEventListener("click", openDailyCopyModal);

    $("#extraAddBtn").addEventListener("click", openExtraModal);

    $("#logsSearchBtn").addEventListener("click", refreshLogs);
    $("#logsExportBtn").addEventListener("click", exportLogs);
    const lm = document.getElementById("logMonth");
    if (lm) lm.addEventListener("change", ()=>{ refreshLogs(); refreshClientLogs(); refreshClientLogs(); });
  }

  window.ChecklistsModule = {
    init: function(){
      if (!ensureDom()) return;
      state.token = token();
      bind();
      setDefaultMonth();
      setActiveTab("daily");
      loadClients();
    }
  };
})();
