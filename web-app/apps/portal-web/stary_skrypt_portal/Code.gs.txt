const AUTH_CONFIG = {
  spreadsheetId: EVENTS_SHEET.spreadsheetId,
  headerRow: 1,
  preferSheetNames: ["Użytkownicy WEB", "Użytkownicy_web"],
  tokenTtlSec: 6 * 60 * 60,
};

function api_login(payload) {
  payload = payload || {};
  const login = String(payload.login || "").trim();
  const password = String(payload.password || "").trim();
  if (!login || !password) throw new Error("Podaj login i hasło.");

  const ss = SpreadsheetApp.openById(AUTH_CONFIG.spreadsheetId);
  const sheetInfo = _authGetUsersSheetInfo_(ss);
  if (!sheetInfo || !sheetInfo.sheet) {
    throw new Error('Nie znalazłem arkusza użytkowników (szukam nagłówków: "Login" + "Hasło/Password").');
  }

  const sheet = sheetInfo.sheet;
  const idx = sheetInfo.idx;

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= AUTH_CONFIG.headerRow) throw new Error("Arkusz użytkowników jest pusty.");

  const data = sheet.getRange(AUTH_CONFIG.headerRow + 1, 1, lastRow - AUTH_CONFIG.headerRow, lastCol).getValues();
  const loginNorm = login.toLowerCase();

  for (let i = 0; i < data.length; i++) {
    const row = data[i];

    const rowLogin = String(row[idx.login] ?? "").trim();
    if (!rowLogin) continue;
    if (rowLogin.toLowerCase() !== loginNorm) continue;

    if (idx.active >= 0) {
      const activeVal = String(row[idx.active] ?? "").trim().toLowerCase();
      if (activeVal) {
        const isFalse = ["0","false","nie","no","inactive","disabled","off"].includes(activeVal);
        if (isFalse) throw new Error("Konto jest nieaktywne.");
      }
    }

    const stored = String(row[idx.password] ?? "").trim();
    if (!stored) throw new Error("Brak hasła w arkuszu dla tego użytkownika.");

    if (!_authCheckPassword_(password, stored)) {
      throw new Error("Błędny login lub hasło.");
    }

    const name = (idx.name >= 0) ? String(row[idx.name] ?? "").trim() : "";
    const roleRaw = (idx.role >= 0) ? String(row[idx.role] ?? "").trim() : "";
    const role = _normalizePortalRole_(roleRaw);

    if (_roleLevel_(role) < 1) {
      throw new Error("Brak dostępu do portalu dla tego typu konta.");
    }

    const token = _authIssueToken_({
      login: rowLogin,
      name: name || rowLogin,
      role: role,
      sheetName: sheet.getName(),
    });

    return {
      ok: true,
      user: {
        login: rowLogin,
        name: name || rowLogin,
        role: role,
        token: token,
      }
    };
  }

  throw new Error("Błędny login lub hasło.");
}

function _authGetUsersSheetInfo_(ss) {
  for (const nm of AUTH_CONFIG.preferSheetNames) {
    const sh = ss.getSheetByName(nm);
    if (!sh) continue;
    const info = _authTryBuildUsersIdx_(sh);
    if (info) return info;
  }

  let best = null;
  const sheets = ss.getSheets();

  for (let s = 0; s < sheets.length; s++) {
    const sh = sheets[s];
    const info = _authTryBuildUsersIdx_(sh);
    if (!info) continue;

    const score = info.score || 0;
    if (!best || score > best.score) best = info;
  }

  return best;
}

function _authTryBuildUsersIdx_(sheet) {
  const lastCol = sheet.getLastColumn();
  const lastRow = sheet.getLastRow();
  if (lastCol <= 0 || lastRow < AUTH_CONFIG.headerRow) return null;

  const headers = sheet.getRange(AUTH_CONFIG.headerRow, 1, 1, lastCol).getValues()[0]
    .map(v => String(v ?? "").trim());

  const norm = headers.map(h => String(h || "").trim().toLowerCase());

  const findAny = (keysExact, keysContains) => {
    for (let i = 0; i < norm.length; i++) {
      const h = norm[i];
      for (const k of keysExact) {
        if (h === k) return i;
      }
    }
    for (let i = 0; i < norm.length; i++) {
      const h = norm[i];
      for (const k of keysContains) {
        if (h.indexOf(k) >= 0) return i;
      }
    }
    return -1;
  };

  const idxLogin = findAny(
    ["login","username","user","użytkownik","uzytkownik","email","e-mail","mail"],
    ["login","user","użytk","uzyt","mail","email"]
  );

  const idxPass = findAny(
    ["hasło","haslo","password","pass","pwd","sha256","sha-256","hash"],
    ["has","pass","pwd","sha","hash"]
  );

  if (idxLogin < 0 || idxPass < 0) return null;

  const idxName = findAny(["imię","imie","name","nazwa"], ["imi","name","nazw"]);
  const idxRole = findAny(["rola","role","uprawnienia","typ"], ["rol","uprawn","typ"]);
  const idxActive = findAny(["aktywny","active","enabled","status"], ["akty","active","enabl","status"]);

  let score = 50;
  if (idxName >= 0) score += 10;
  if (idxRole >= 0) score += 8;
  if (idxActive >= 0) score += 6;

  return {
    sheet,
    score,
    idx: {
      login: idxLogin,
      password: idxPass,
      name: idxName,
      role: idxRole,
      active: idxActive,
    }
  };
}

function _authIssueToken_(userObj) {
  const token = Utilities.getUuid() + "-" + (new Date().getTime().toString(36));
  try {
    CacheService.getScriptCache().put(
      "fido_token_" + token,
      JSON.stringify({ ...userObj, issuedAt: new Date().toISOString() }),
      AUTH_CONFIG.tokenTtlSec
    );
  } catch (e) {}
  return token;
}

function _authCheckPassword_(inputPassword, stored) {
  const inP = String(inputPassword || "");
  const st = String(stored || "").trim();

  const isSha256Hex = /^[a-f0-9]{64}$/i.test(st);
  if (isSha256Hex) {
    const h = _authSha256Hex_(inP);
    return h.toLowerCase() === st.toLowerCase();
  }
  return inP === st;
}

function _authSha256Hex_(text) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8);
  return bytes.map(b => {
    const v = (b < 0) ? b + 256 : b;
    return ("0" + v.toString(16)).slice(-2);
  }).join("");
}

function _authGetUserByToken_(token) {
  if (!token) return null;
  try {
    const raw = CacheService.getScriptCache().get("fido_token_" + token);
    if (!raw) return null;
    return JSON.parse(raw);
  } catch (e) {
    return null;
  }
}

/** ROLE (ważne: Stażysta = read-only; Stały personel/Zespół mobilny = jak Pracownik) */
function _normalizePortalRole_(roleRaw) {
  const r = String(roleRaw || "").trim();
  const rl = r.toLowerCase();

  if (rl === "admin") return "Admin";
  if (rl === "kierownik") return "Kierownik";
  if (rl === "koordynator") return "Koordynator";
  if (rl === "stażysta" || rl === "stazysta") return "Stażysta";
  if (rl === "pracownik") return "Pracownik";

  if (rl === "stały personel na obiekcie" || rl === "staly personel na obiekcie") return "Stały personel na obiekcie";
  if (rl === "zespół mobilny" || rl === "zespol mobilny") return "Zespół mobilny";

  return r;
}

function _roleLevel_(role) {
  const r = _normalizePortalRole_(role);
  if (r === "Admin") return 3;
  if (r === "Kierownik") return 2;
  if (r === "Koordynator") return 1;
  if (r === "Stażysta") return 1;
  return 0;
}

function _authRequireRead_(token) {
  const u = _authGetUserByToken_(token);
  if (!u) throw new Error("Sesja wygasła. Zaloguj się ponownie.");
  if (_roleLevel_(u.role) < 1) throw new Error("Brak uprawnień.");
  return u;
}
function _authRequireWrite_(token) {
  const u = _authGetUserByToken_(token);
  if (!u) throw new Error("Sesja wygasła. Zaloguj się ponownie.");
  if (_roleLevel_(u.role) < 2) throw new Error("Brak uprawnień do edycji.");
  return u;
}
function _authRequireAdmin_(token) {
  const u = _authGetUserByToken_(token);
  if (!u) throw new Error("Sesja wygasła. Zaloguj się ponownie.");
  if (_roleLevel_(u.role) < 3) throw new Error("Brak uprawnień administratora.");
  return u;
}

/**
 * Zwraca Set loginów, które mają rozpoczętą pracę (jest StartAt w kol. E),
 * ale nie mają jeszcze zakończenia (brak EndScanAt w kol. F).
 */
function _buildRunningWorkersSet_() {
  const sheet = _getWorkdayLogSheet_();
  const lastRow = sheet.getLastRow();
  if (lastRow <= WORKDAY_LOG_SHEET.headerRow) return new Set();

  // B=WorkerLogin, E=StartAt, F=EndScanAt
  const values = sheet.getRange(WORKDAY_LOG_SHEET.headerRow + 1, 2, lastRow - WORKDAY_LOG_SHEET.headerRow, 5).getValues();
  const running = new Set();

  for (let i = 0; i < values.length; i++) {
    const row = values[i];
    const login = String(row[0] ?? "").trim();   // B
    const startAt = row[3];                      // E (w tym zakresie to index 3)
    const endScanAt = row[4];                    // F (index 4)

    if (!login) continue;

    const hasStart = String(startAt ?? "").trim() !== "";
    const hasEnd = String(endScanAt ?? "").trim() !== "";

    if (hasStart && !hasEnd) running.add(login);
    if (hasEnd) running.delete(login);
  }

  return running;
}

function api_listWorkers(params) {
  params = params || {};
  _authRequireRead_(String(params.token || "").trim());

  const sheet = _getWorkersSheet_();
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= WORKERS_SHEET.headerRow) {
    return { items: [], total: 0, page: 1, pageSize: 50 };
  }

  const data = sheet.getRange(WORKERS_SHEET.headerRow + 1, 1, lastRow - WORKERS_SHEET.headerRow, Math.max(14, lastCol)).getValues();

  const q = String(params.q || "").trim().toLowerCase();
  const type = String(params.type || "").trim().toLowerCase();
  const activeFilter = String(params.active || "").trim().toLowerCase(); // "1"/"0"/"true"/"false"/""
  const onlineFilter = String(params.online || "").trim(); // "1"|"0"| ""

  // Zalogowany = ma rozpoczętą pracę (Workday_Log: StartAt jest, EndScanAt brak)
  const runningSet = _buildRunningWorkersSet_();


  const page = Math.max(1, parseInt(params.page || 1, 10) || 1);
  const pageSize = Math.min(500, Math.max(50, parseInt(params.pageSize || 50, 10) || 50));

  const items = [];

  const metaMap = _znMetaMap_();
  for (let i = 0; i < data.length; i++) {
    const r = data[i];
    const rowNumber = WORKERS_SHEET.headerRow + 1 + i;

    const workerId = String(r[0] ?? "").trim();      // A
    const name = String(r[1] ?? "").trim();          // B
    const login = String(r[2] ?? "").trim();         // C
    const onlineRaw = String(r[5] ?? "").trim();     // F (nie używamy do statusu "Zalogowany")
    const qrText = String(r[6] ?? "").trim();        // G
    const activeRaw = String(r[7] ?? "").trim();     // H
    const email = String(r[8] ?? "").trim();         // I
    const phone = String(r[9] ?? "").trim();         // J
    const workerType = String(r[10] ?? "").trim();   // K
    const addedAt = String(r[11] ?? "").trim();      // L
    const editedAt = String(r[12] ?? "").trim();     // M
    const editedBy = String(r[13] ?? "").trim();     // N

    if (!workerId && !name && !login) continue;

    const active = _parseActive_(activeRaw);
    const online = runningSet.has(login) ? true : _parseOnline_(onlineRaw);

    if (type && workerType.toLowerCase().indexOf(type) === -1) continue;
    if ((activeFilter === "1" || activeFilter === "true") && !active) continue;
    if ((activeFilter === "0" || activeFilter === "false") && active) continue;
    if (onlineFilter === "1" && !online) continue;
    if (onlineFilter === "0" && online) continue;

    if (q) {
      const blob = [workerId, name, login, email, phone, qrText, workerType].join(" ").toLowerCase();
      if (blob.indexOf(q) === -1) continue;
    }

    items.push({
      rowNumber,
      workerId,
      name,
      login,
      type: workerType,
      active,
      online,
      onlineRaw,
      email,
      phone,
      qrText,
      addedAt,
      editedAt,
      editedBy
    });
  }

  items.sort((a,b) => _workerIdNum_(a.workerId) - _workerIdNum_(b.workerId));

  const total = items.length;
  const start = (page - 1) * pageSize;
  const pageItems = items.slice(start, start + pageSize);

  return { items: pageItems, total, page, pageSize };
}

function api_createWorker(payload) {
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || "").trim());

  const sheet = _getWorkersSheet_();
  const nextId = _workersNextId_(sheet);

  const name = _cleanName_(payload.name);
  const login = _cleanLogin_(payload.login);
  const type = String(payload.type || "").trim();
  const active = _parseActive_(payload.active);
  const email = String(payload.email || "").trim();
  const phone = String(payload.phone || "").trim();
  const qrText = String(payload.qrText || "").trim();
  const newPassword = String(payload.newPassword || "").trim();

  if (!name) throw new Error("Imię i nazwisko jest wymagane.");
  if (!login) throw new Error("Login jest wymagany.");
  if (!type) throw new Error("Typ pracownika jest wymagany.");
  if (!newPassword) throw new Error("Hasło jest wymagane.");

  _workersEnsureLoginUnique_(sheet, login, null);

  const passHash = _authSha256Hex_(newPassword);
  const nowIso = new Date().toISOString();

  const row = [
    nextId,                 // A
    name,                   // B
    login,                  // C
    newPassword,           // D
    passHash,               // E
    "",                     // F
    qrText,                 // G
    _formatActive_(active), // H
    email,                  // I
    phone,                  // J
    type,                   // K
    nowIso,                 // L
    "",                     // M
    ""                      // N
  ];

  sheet.appendRow(row);
  return { ok: true, workerId: nextId };
}

function api_updateWorker(payload) {
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || "").trim());

  const workerId = String(payload.workerId || "").trim();
  if (!workerId) throw new Error("Brak WorkerID.");

  const sheet = _getWorkersSheet_();
  const rowNumber = _findRowByWorkerId_(sheet, workerId);
  if (!rowNumber) throw new Error("Nie znaleziono pracownika po ID.");

  const name = _cleanName_(payload.name);
  const login = _cleanLogin_(payload.login);
  const type = String(payload.type || "").trim();
  const active = _parseActive_(payload.active);
  const email = String(payload.email || "").trim();
  const phone = String(payload.phone || "").trim();
  const qrText = String(payload.qrText || "").trim();
  const newPassword = String(payload.newPassword || "").trim();
  const editorName = String(payload.editorName || "").trim() || "Użytkownik";

  if (!name) throw new Error("Imię i nazwisko jest wymagane.");
  if (!login) throw new Error("Login jest wymagany.");
  if (!type) throw new Error("Typ pracownika jest wymagany.");

  _workersEnsureLoginUnique_(sheet, login, workerId);

  const row = sheet.getRange(rowNumber, 1, 1, Math.max(14, sheet.getLastColumn())).getValues()[0];

  row[1] = name;
  row[2] = login;
  if (newPassword) {
    row[3] = newPassword;
    row[4] = _authSha256Hex_(newPassword);
  }
  row[6] = qrText;
  row[7] = _formatActive_(active);
  row[8] = email;
  row[9] = phone;
  row[10] = type;
  row[12] = new Date().toISOString(); // M
  row[13] = editorName; // N

  sheet.getRange(rowNumber, 1, 1, row.length).setValues([row]);
  return { ok: true, workerId };
}

function api_deleteWorker(payload) {
  payload = payload || {};
  _authRequireAdmin_(String(payload.token || "").trim());

  const sheet = _getWorkersSheet_();

  // UI może przekazać workerId oraz/lub rowNumber (dla pewności).
  const workerId = String(payload.workerId || payload.id || "").trim();
  const rowFromUi = parseInt(payload.rowNumber || payload.row || 0, 10) || 0;

  // 1) jeśli UI podało rowNumber — spróbuj usuwać po wierszu (z weryfikacją ID)
  if (rowFromUi > WORKERS_SHEET.headerRow) {
    const idAtRow = String(sheet.getRange(rowFromUi, 1).getValue() || "").trim();
    if (workerId && idAtRow && idAtRow !== workerId) {
      throw new Error("Niezgodność danych: inny WorkerID w wierszu.");
    }
    if (!idAtRow) throw new Error("Brak danych w wskazanym wierszu (rowNumber).");
    sheet.deleteRow(rowFromUi);
    return { ok: true };
  }

  // 2) standardowo — usuń po WorkerID
  if (!workerId) throw new Error("Brak WorkerID.");
  const rowNumber = _findRowByWorkerId_(sheet, workerId);
  if (!rowNumber) throw new Error("Nie znaleziono pracownika po ID.");

  sheet.deleteRow(rowNumber);
  return { ok: true };
}


/** helpers */
function _cleanName_(v) { return String(v ?? "").trim().replace(/\s+/g, " "); }
function _cleanLogin_(v) { return String(v ?? "").trim(); }

function _findRowByWorkerId_(sheet, workerId) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= WORKERS_SHEET.headerRow) return 0;

  const values = sheet.getRange(WORKERS_SHEET.headerRow + 1, 1, lastRow - WORKERS_SHEET.headerRow, 1).getValues();
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0] ?? "").trim() === workerId) return WORKERS_SHEET.headerRow + 1 + i;
  }
  return 0;
}



/**
 * Zwraca obecne hasło pracownika (kol. D) — TYLKO Admin/Kierownik.
 * Uwaga: to jest celowo "plain text", bo w arkuszu trzymasz hasło w D.
 */
function api_getWorkerPassword(payload) {
  payload = payload || {};
  const token = String(payload.token || "").trim();
  const u = _authGetUserByToken_(token);
  if (!u) throw new Error("Sesja wygasła. Zaloguj się ponownie.");
  if (_roleLevel_(u.role) < 2) throw new Error("Brak uprawnień do podglądu hasła.");

  const workerId = String(payload.workerId || "").trim();
  if (!workerId) throw new Error("Brak WorkerID.");

  const sheet = _getWorkersSheet_();
  const rowNumber = _findRowByWorkerId_(sheet, workerId);
  if (!rowNumber) throw new Error("Nie znaleziono pracownika po ID.");

  // kolumna D => index 3
  const row = sheet.getRange(rowNumber, 1, 1, Math.max(5, sheet.getLastColumn())).getValues()[0];
  const password = String(row[3] || "");
  return { ok: true, workerId, password };
}

function _workersNextId_(sheet) {
  const lastRow = sheet.getLastRow();
  // Startujemy od W001
  if (lastRow <= WORKERS_SHEET.headerRow) return "W001";

  // Pobierz istniejące ID z kolumny A
  const values = sheet.getRange(WORKERS_SHEET.headerRow + 1, 1, lastRow - WORKERS_SHEET.headerRow, 1).getValues();

  const used = new Set();
  for (let i = 0; i < values.length; i++) {
    const raw = String(values[i][0] ?? "").trim();
    if (!raw) continue;

    // akceptujemy W001/W1/W0001 -> znormalizujemy do numeru
    const mw = raw.match(/^W(\d+)$/i);
    if (mw) {
      const num = parseInt(mw[1], 10);
      if (isFinite(num) && num > 0) used.add(num);
      continue;
    }

    // legacy: BC0001 itd. (nie wpływa na W001…)
    // const mbc = raw.match(/^BC(\d+)$/i);
    // if (mbc) { ... }
  }

  // Szukamy pierwszego wolnego numeru od 1 wzwyż
  for (let n = 1; n <= 999; n++) {
    if (!used.has(n)) {
      return "W" + String(n).padStart(3, "0");
    }
  }

  // fallback jeśli kiedyś przekroczymy 999
  const maxN = Math.max(0, ...Array.from(used));
  return "W" + String(maxN + 1).padStart(3, "0");
}


function _workerIdNum_(workerId) {
  const raw = String(workerId || "").trim();

  // Preferowane: W001...
  let m = raw.match(/^W(\d+)$/i);
  if (m) {
    const n = parseInt(m[1], 10);
    return isFinite(n) ? n : Number.MAX_SAFE_INTEGER;
  }

  // Legacy: BC0001...
  m = raw.match(/^BC(\d+)$/i);
  if (m) {
    const n = parseInt(m[1], 10);
    // żeby Wxxx było zawsze przed BCxxxx w sortowaniu:
    return isFinite(n) ? (10000 + n) : Number.MAX_SAFE_INTEGER;
  }

  return Number.MAX_SAFE_INTEGER;
}


function _workersEnsureLoginUnique_(sheet, login, currentWorkerId) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= WORKERS_SHEET.headerRow) return true;

  const data = sheet.getRange(WORKERS_SHEET.headerRow + 1, 1, lastRow - WORKERS_SHEET.headerRow, 3).getValues();
  const loginNorm = String(login || "").trim().toLowerCase();

  for (let i = 0; i < data.length; i++) {
    const id = String(data[i][0] ?? "").trim();
    const lg = String(data[i][2] ?? "").trim();
    if (!lg) continue;

    if (lg.trim().toLowerCase() === loginNorm) {
      if (currentWorkerId && id === currentWorkerId) continue;
      throw new Error("Taki login już istnieje.");
    }
  }
  return true;
}

function _parseActive_(raw) {
  const s = String(raw ?? "").trim().toLowerCase();
  if (!s) return false;
  if (["aktywny","tak","true","1","yes","on"].includes(s)) return true;
  if (["nieaktywny","nie","false","0","no","off","inactive"].includes(s)) return false;
  return true;
}
function _formatActive_(isActive) { return isActive ? "TRUE" : "FALSE"; }

function _parseOnline_(raw) {
  // "Zalogowany" = pracownik ma rozpoczętą pracę.
  // W arkuszu (kol. F) zwykle trzymamy znacznik: TAK/TRUE/1 albo timestamp START (czyszczony przy STOP).
  // Jeśli jest jakakolwiek poprawna data/czas → traktujemy jako TAK (bez limitu 60 min).
  const s = String(raw ?? "").trim();
  if (!s) return false;
  const sl = s.toLowerCase();

  if (["tak","true","1","online","on"].includes(sl)) return true;
  if (["nie","false","0","offline","off"].includes(sl)) return false;

  const d = new Date(s);
  if (isFinite(d.getTime())) return true;

  return false;
}

function api_listEvents(params){
  params = params || {};
  _authRequireRead_(String(params.token || "").trim());

  const sh = _evGetSheet_();
  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow <= EVENTS_SHEET.headerRow) return { items: [], total: 0, page: 1, pageSize: 200 };

  const headers = sh.getRange(EVENTS_SHEET.headerRow, 1, 1, lastCol).getValues()[0].map(v => String(v ?? "").trim());
  const idx = _evBuildIdx_(headers);

  const data = sh.getRange(EVENTS_SHEET.headerRow + 1, 1, lastRow - EVENTS_SHEET.headerRow, lastCol).getValues();

  const fromIso = _evNorm_(params.fromIso);
  const toIso = _evNorm_(params.toIso);
  const worker = _evNorm_(params.worker);
  const strefa = _evNorm_(params.strefa);
  const pomieszczenie = _evNorm_(params.pomieszczenie);
  const roomId = _evNorm_(params.roomId);
  const status = String(params.status || "").trim();
  const q = _evNorm_(params.q);

  const page = Math.max(1, parseInt(params.page || 1, 10) || 1);
  const pageSize = Math.min(500, Math.max(50, parseInt(params.pageSize || 200, 10) || 200));

  const items = [];

  const metaMap = _znMetaMap_();

  for (let i = 0; i < data.length; i++){
    const r = data[i];
    const rowNumber = EVENTS_SHEET.headerRow + 1 + i;

    const cycleId = String(r[idx.cycleId] ?? "").trim();
    if (!cycleId) continue;

    const startAt = _evCellToIso_(r[idx.startAt]);
    const startKey = _evDateKey_(startAt);

    if (fromIso && startKey && startKey < fromIso) continue;
    if (toIso && startKey && startKey > toIso) continue;

    const workerName = (idx.workerName >= 0) ? String(r[idx.workerName] ?? "").trim() : "";
    const rrid = (idx.roomId >= 0) ? String(r[idx.roomId] ?? "").trim() : "";
    const st = (idx.strefa >= 0) ? String(r[idx.strefa] ?? "").trim() : "";
    const pm = (idx.pomieszczenie >= 0) ? String(r[idx.pomieszczenie] ?? "").trim() : "";
    const meta = rrid ? (metaMap[_evNorm_(rrid)] || null) : null;
    const klientName = meta ? String(meta.klient || "").trim() : "";
    const lokalizacja = meta ? String(meta.lokalizacja || "").trim() : "";
    const endAt = (idx.endAt >= 0) ? _evCellToIso_(r[idx.endAt]) : "";
    const dur = (idx.durationSec >= 0) ? Number(r[idx.durationSec] ?? 0) : 0;
    const stt = (idx.status >= 0) ? String(r[idx.status] ?? "").trim() : "";
    const cmt = (idx.comment >= 0) ? String(r[idx.comment] ?? "").trim() : "";
    const edBy = (idx.editedBy >= 0) ? String(r[idx.editedBy] ?? "").trim() : "";

    if (worker && _evNorm_(workerName).indexOf(worker) === -1) continue;
    if (strefa && _evNorm_(st).indexOf(strefa) === -1) continue;
    if (pomieszczenie){
      const hitClient = _evNorm_(klientName).indexOf(pomieszczenie) !== -1;
      const hitPm = _evNorm_(pm).indexOf(pomieszczenie) !== -1;
      if (!hitClient && !hitPm) continue;
    }
    if (roomId && _evNorm_(rrid).indexOf(roomId) === -1) continue;
    if (status && stt !== status) continue;

    if (q){
      const blob = [cycleId, workerName, rrid, st, pm, klientName, lokalizacja, startAt, endAt, stt, cmt, edBy].join(" ").toLowerCase();
      if (blob.indexOf(q) === -1) continue;
    }

    items.push({
      rowNumber,
      cycleId,
      workdayId: (idx.workdayId >= 0) ? String(r[idx.workdayId] ?? "").trim() : "",
      workerName,
      roomId: rrid,
      strefa: st,
      pomieszczenie: pm,
      klient: klientName,
      lokalizacja: lokalizacja,
      startAt,
      endAt,
      durationSec: isFinite(dur) ? dur : 0,
      status: stt,
      comment: cmt,
      editedBy: edBy
    });
  }

  items.sort((a,b)=>{
    const ak = String(a.startAt || "");
    const bk = String(b.startAt || "");
    if (ak === bk) return b.rowNumber - a.rowNumber;
    return (ak < bk) ? 1 : -1;
  });

  const total = items.length;
  const start = (page - 1) * pageSize;
  const pageItems = items.slice(start, start + pageSize);

  return { items: pageItems, total, page, pageSize };
}


function api_createEvent(payload){
  payload = payload || {};
  const u = _authRequireWrite_(String(payload.token || "").trim());

  const sh = _evGetSheet_();
  const lastCol = sh.getLastColumn();
  if (lastCol <= 0) throw new Error("Arkusz Zdarzenia nie ma nagłówków.");

  const headers = sh.getRange(EVENTS_SHEET.headerRow, 1, 1, lastCol).getValues()[0].map(v => String(v ?? "").trim());
  const idx = _evBuildIdx_(headers);

  const cycleId = String(payload.cycleId || "").trim() || Utilities.getUuid();

  const workerName = String(payload.workerName || "").trim();
  const workdayId = String(payload.workdayId || "").trim();
  const roomId = String(payload.roomId || "").trim();
  const strefa = String(payload.strefa || "").trim();
  const pomieszczenie = String(payload.pomieszczenie || "").trim();

  const startAt = String(payload.startAt || "").trim();
  const endAt = String(payload.endAt || "").trim();

  if (!startAt) throw new Error("Brak StartAt.");

  // duration
  let durationSec = 0;
  try{
    const ds = new Date(startAt);
    const de = endAt ? new Date(endAt) : null;
    if (isFinite(ds.getTime()) && de && isFinite(de.getTime())){
      durationSec = Math.max(0, Math.round((de.getTime() - ds.getTime()) / 1000));
    }
  }catch(e){ durationSec = 0; }

  const status = String(payload.status || "").trim() || (endAt ? "CLOSED" : "RUNNING");
  const comment = String(payload.comment || "").trim();

  const editorName = (u && (u.name || u.login))
    ? String(u.name || u.login).trim()
    : (String(payload.editorName || "").trim() || "Użytkownik");

  // budujemy wiersz o długości lastCol (nie wstawiamy poza zakres)
  const row = new Array(lastCol).fill("");

  if (idx.cycleId >= 0) row[idx.cycleId] = cycleId;
  if (idx.workdayId >= 0) row[idx.workdayId] = workdayId;
  if (idx.workerName >= 0) row[idx.workerName] = workerName;
  if (idx.roomId >= 0) row[idx.roomId] = roomId;
  if (idx.strefa >= 0) row[idx.strefa] = strefa;
  if (idx.pomieszczenie >= 0) row[idx.pomieszczenie] = pomieszczenie;

  if (idx.startAt >= 0) row[idx.startAt] = startAt;
  if (idx.endAt >= 0) row[idx.endAt] = endAt;

  if (idx.durationSec >= 0) row[idx.durationSec] = durationSec;
  if (idx.status >= 0) row[idx.status] = status;

  if (idx.comment >= 0) row[idx.comment] = comment;

  if (idx.editedBy >= 0) row[idx.editedBy] = editorName;
  if (idx.editedAt >= 0) row[idx.editedAt] = new Date().toISOString();

  sh.appendRow(row);
  return { ok:true, cycleId: cycleId };
}

function api_listWorkerTimeEvents(params){
  params = params || {};
  _authRequireRead_(String(params.token || '').trim());

  const workerName = String(params.workerName || '').trim();
  if (!workerName) throw new Error('Brak workerName.');

  // Bazowo bierzemy paginowane zdarzenia.
  const res = api_listEvents({
    token: String(params.token || '').trim(),
    fromIso: params.fromIso,
    toIso: params.toIso,
    worker: workerName,
    strefa: params.strefa,
    pomieszczenie: params.pomieszczenie,
    roomId: params.roomId,
    status: params.status,
    q: params.q,
    page: params.page,
    pageSize: Math.min(500, Math.max(50, parseInt(params.pageSize || 50, 10) || 50))
  });

  const items = (res && res.items) ? res.items : [];
  if (!items.length){
    return { ...res, items: [] };
  }

  // Zbieramy klucze do mapowania przerw.
  const workdayIds = {};
  const dateKeys = {};
  for (const it of items){
    if (it.workdayId) workdayIds[String(it.workdayId)] = 1;
    const dk = _dateKey_(it.startAt);
    if (dk) dateKeys[dk] = 1;
  }

  // Wczytujemy przerwy z zakresu dat (po StartAt przerwy).
  const ps = _psGetSheet_();
  const psLastRow = ps.getLastRow();
  const psLastCol = ps.getLastColumn();
  if (psLastRow <= PAUSES_SHEET.headerRow){
    return { ...res, items: items.map(it => ({...it, breakSec: 0})) };
  }

  const psHeaders = ps.getRange(PAUSES_SHEET.headerRow, 1, 1, psLastCol).getValues()[0].map(v => String(v ?? '').trim());
  const psIdx = _psBuildIdx_(psHeaders);
  const psData = ps.getRange(PAUSES_SHEET.headerRow + 1, 1, psLastRow - PAUSES_SHEET.headerRow, psLastCol).getValues();

  const fromIso = String(params.fromIso || '').trim();
  const toIso = String(params.toIso || '').trim();
  const wNorm = _psNorm_(workerName);

  const sumByWorkday = {};
  const sumByDate = {};

  for (let i = 0; i < psData.length; i++){
    const r = psData[i];
    const wname = (psIdx.workerName >= 0) ? String(r[psIdx.workerName] ?? '').trim() : '';
    if (!wname) continue;
    if (_psNorm_(wname).indexOf(wNorm) < 0) continue;

    const startAt = _psCellToIso_(r[psIdx.startAt]);
    const dk = _dateKey_(startAt);
    if (fromIso && dk && dk < fromIso) continue;
    if (toIso && dk && dk > toIso) continue;

    const dur = (psIdx.durationSec >= 0) ? Number(r[psIdx.durationSec] ?? 0) : 0;
    const dsec = (isFinite(dur) && dur >= 0) ? dur : 0;

    const wd = (psIdx.workdayId >= 0) ? String(r[psIdx.workdayId] ?? '').trim() : '';

    if (wd && workdayIds[wd]){
      sumByWorkday[wd] = (sumByWorkday[wd] || 0) + dsec;
    }

    if (dk && dateKeys[dk]){
      sumByDate[dk] = (sumByDate[dk] || 0) + dsec;
    }
  }

  const mapped = items.map(it => {
    let bsec = 0;
    if (it.workdayId && sumByWorkday[String(it.workdayId)]){
      bsec = sumByWorkday[String(it.workdayId)] || 0;
    } else {
      const dk = _dateKey_(it.startAt);
      bsec = dk ? (sumByDate[dk] || 0) : 0;
    }
    return { ...it, breakSec: bsec };
  });

  return { ...res, items: mapped };
}

function api_listPauses(params){
  params = params || {};
  _authRequireRead_(String(params.token || '').trim());

  const sh = _psGetSheet_();
  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow <= PAUSES_SHEET.headerRow) return { items: [], total: 0, page: 1, pageSize: 50 };

  const headers = sh.getRange(PAUSES_SHEET.headerRow, 1, 1, lastCol).getValues()[0].map(v => String(v ?? '').trim());
  const idx = _psBuildIdx_(headers);
  const data = sh.getRange(PAUSES_SHEET.headerRow + 1, 1, lastRow - PAUSES_SHEET.headerRow, lastCol).getValues();

  const fromIso = _psNorm_(params.fromIso);
  const toIso = _psNorm_(params.toIso);
  const worker = _psNorm_(params.worker);
  const status = String(params.status || '').trim();
  const q = _psNorm_(params.q);

  const page = Math.max(1, parseInt(params.page || 1, 10) || 1);
  const pageSize = Math.min(500, Math.max(50, parseInt(params.pageSize || 50, 10) || 50));

  const items = [];

  for (let i = 0; i < data.length; i++){
    const r = data[i];
    const rowNumber = PAUSES_SHEET.headerRow + 1 + i;

    const pauseId = String(r[idx.pauseId] ?? '').trim();
    if (!pauseId) continue;

    const startAt = _psCellToIso_(r[idx.startAt]);
    const startKey = _dateKey_(startAt);

    if (fromIso && startKey && startKey < fromIso) continue;
    if (toIso && startKey && startKey > toIso) continue;

    const workerName = (idx.workerName >= 0) ? String(r[idx.workerName] ?? '').trim() : '';
    const workerLogin = (idx.workerLogin >= 0) ? String(r[idx.workerLogin] ?? '').trim() : '';
    const workdayId = (idx.workdayId >= 0) ? String(r[idx.workdayId] ?? '').trim() : '';
    const stopAt = (idx.stopAt >= 0) ? _psCellToIso_(r[idx.stopAt]) : '';
    const dur = (idx.durationSec >= 0) ? Number(r[idx.durationSec] ?? 0) : 0;
    const stt = (idx.status >= 0) ? String(r[idx.status] ?? '').trim() : '';
    const createdAt = (idx.createdAt >= 0) ? _psCellToIso_(r[idx.createdAt]) : '';
    const updatedAt = (idx.updatedAt >= 0) ? _psCellToIso_(r[idx.updatedAt]) : '';

    if (worker){
      const blobW = (workerName + ' ' + workerLogin).toLowerCase();
      if (blobW.indexOf(worker) < 0) continue;
    }
    if (status && stt !== status) continue;

    if (q){
      const blob = [pauseId, workdayId, workerLogin, workerName, startAt, stopAt, stt].join(' ').toLowerCase();
      if (blob.indexOf(q) < 0) continue;
    }

    items.push({
      rowNumber,
      pauseId,
      workdayId,
      workerLogin,
      workerName,
      startAt,
      stopAt,
      durationSec: isFinite(dur) ? dur : 0,
      status: stt,
      createdAt,
      updatedAt
    });
  }

  items.sort((a,b)=>{
    const ak = String(a.startAt || '');
    const bk = String(b.startAt || '');
    if (ak === bk) return b.rowNumber - a.rowNumber;
    return (ak < bk) ? 1 : -1;
  });

  const total = items.length;
  const start = (page - 1) * pageSize;
  const pageItems = items.slice(start, start + pageSize);

  return { items: pageItems, total, page, pageSize };
}

function _ioToIsoDate_(v){
  // Accepts Date or string; returns yyyy-MM-dd or empty.
  if (v === null || v === undefined || v === "") return "";
  if (Object.prototype.toString.call(v) === "[object Date]" && !isNaN(v.getTime())){
    return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  const s = String(v).trim();
  if (!s) return "";
  // already ISO
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
  // dd.MM.yyyy
  const m1 = s.match(/^(\d{2})\.(\d{2})\.(\d{4})$/);
  if (m1) return `${m1[3]}-${m1[2]}-${m1[1]}`;
  // dd/MM/yyyy or d/M/yyyy
  const m2 = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if (m2){
    const dd = String(m2[1]).padStart(2,"0");
    const mm = String(m2[2]).padStart(2,"0");
    return `${m2[3]}-${mm}-${dd}`;
  }
  // fallback: try Date parse
  const d = new Date(s);
  if (!isNaN(d.getTime())){
    return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  return s; // last resort
}

function _ioFmtDatePL_(isoOrAny){
  const iso = _ioToIsoDate_(isoOrAny);
  if (!iso) return "";
  const m = iso.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if (!m) return iso;
  return `${m[3]}.${m[2]}.${m[1]}`;
}

    if (isFinite(n) && n > maxNum) maxNum = n;
  }

}

function _znNextQr_(sheet){
  const lastRow = sheet.getLastRow();
  if (lastRow <= ZONES_SHEET.headerRow) return "BC0001";

  const values = sheet.getRange(ZONES_SHEET.headerRow + 1, 1, lastRow - ZONES_SHEET.headerRow, 1).getValues();

  let maxBC = 0;
  let maxNum = 0;
  let hasBC = false;
  let hasNum = false;

  for (let i = 0; i < values.length; i++){
    const raw = String(values[i][0] ?? "").trim();
    if (!raw) continue;

    const mBC = raw.match(/^BC(\d+)$/i);
    if (mBC){
      hasBC = true;
      const n = parseInt(mBC[1], 10);
      if (isFinite(n) && n > maxBC) maxBC = n;
      continue;
    }

    const mNum = raw.match(/^(\d+)$/);
    if (mNum){
      hasNum = true;
      const n2 = parseInt(mNum[1], 10);
      if (isFinite(n2) && n2 > maxNum) maxNum = n2;
    }
  }

  if (hasBC){
    const next = maxBC + 1;
    return "BC" + String(next).padStart(4, "0");
  }

  if (hasNum){
    return String(maxNum + 1);
  }

  return "BC0001";
}



function _znFmtDateTime_(v){
  if (v === null || v === undefined || v === "") return "";
  if (Object.prototype.toString.call(v) === "[object Date]" && !isNaN(v.getTime())){
    return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
  }
  return String(v).trim();
}

function api_saveClient(payload) {
  payload = payload || {};
  const ss = SpreadsheetApp.openById(CLIENTS_SHEET.spreadsheetId);
  const sheet = ss.getSheetByName("Klienci") || ss.getSheets()[0];
  
  const id = String(payload.id || "").trim();
  const lock = LockService.getScriptLock();
  
  try {
    lock.waitLock(10000);
    const data = sheet.getDataRange().getValues();

    // --- LOGIKA GENEROWANIA ID (LK001, LK002...) ---
    let finalId = id;
    if (!finalId) {
      let maxNum = 0;
      // Przeszukujemy kolumnę A (od drugiego wiersza), by znaleźć najwyższy numer
      for (let i = 1; i < data.length; i++) {
        const currentId = String(data[i][0]); // Kolumna A
        if (currentId.startsWith("LK")) {
          const numPart = parseInt(currentId.replace("LK", ""), 10);
          if (!isNaN(numPart) && numPart > maxNum) maxNum = numPart;
        }
      }
      finalId = "LK" + String(maxNum + 1).padStart(3, '0');
    }

    // --- MAPOWANIE DANYCH (Zgodnie z Twoim układem A-H) ---
    // A:ID, B:Nazwa, C:NIP, D:Miasto, E:Adres, F:Kontakt, G:Status, H:Koordynator
    const rowValues = [
      payload.nazwa || "",       // B
      payload.nip || "",         // C
      payload.miasto || "",      // D
      payload.adres || "",       // E
      payload.kontakt || "",     // F (NOWA KOLUMNA)
      payload.status || "Aktywny", // G
      payload.koordynator || ""  // H
    ];

    if (id) {
      // EDYCJA: Szukamy wiersza po ID
      let rowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === id) {
          rowIndex = i + 1;
          break;
        }
      }
      if (rowIndex === -1) throw new Error("Nie znaleziono klienta o ID " + id);
      
      // Zapisujemy od kolumny B (2) do H (8) -> 7 kolumn
      sheet.getRange(rowIndex, 2, 1, 7).setValues([rowValues]);
    } else {
      // NOWY KLIENT
      const newRow = [finalId].concat(rowValues);
      sheet.appendRow(newRow);
    }

    return { success: true, newId: finalId };
  } catch (e) {
    throw new Error(e.message);
  } finally {
    lock.releaseLock();
  }
}










/** ---------------------------
 * Reports (Zestawienia)
 * - Strefy per klient (z zakończonych zdarzeń)
 * - Lista zakończonych zdarzeń dla porównywarki (A/B)
 * --------------------------- */
function api_reports_listStrefy(params){
  // Zestawienia → Zdarzenia: lista stref przypisana do klienta z arkusza "Strefy"
  // (B=Klient, C=Strefa). Nie opieramy się o zakończone zdarzenia, bo wtedy lista potrafi być pusta.
  params = params || {};
  _authRequireRead_(String(params.token || "").trim());

  const klientRaw = String(params.klient || "").trim();
  const klientF = _evNorm_(klientRaw);

  const sh = _znGetSheet_(); // Strefy (gid=0)
  const lastRow = sh.getLastRow();
  if (lastRow <= ZONES_SHEET.headerRow) return { items: [] };

  const data = sh.getRange(ZONES_SHEET.headerRow + 1, 1, lastRow - ZONES_SHEET.headerRow, ZONES_SHEET.maxCols).getValues();

  const set = {};

  function addIfMatch(matchFn){
    for (let i = 0; i < data.length; i++){
      const r = data[i];

      const klient = String(r[1] ?? "").trim(); // B
      const strefa = String(r[2] ?? "").trim(); // C
      if (!strefa) continue;

      if (klientF){
        const kn = _evNorm_(klient);
        if (!matchFn(kn, klientF)) continue;
      }
      set[strefa] = true;
    }
  }

  // 1) dopasowanie dokładne
  addIfMatch((kn, kf)=> kn === kf);

  // 2) jeśli nic nie znaleziono i mamy filtr klienta – dopasowanie "zawiera" (na wypadek spacji/variantów)
  if (klientF && Object.keys(set).length === 0){
    addIfMatch((kn, kf)=> kn.indexOf(kf) !== -1 || kf.indexOf(kn) !== -1);
  }

  const items = Object.keys(set).sort((a,b)=>{
    const aa = _evNorm_(a);
    const bb = _evNorm_(b);
    if (aa === bb) return 0;
    return aa < bb ? -1 : 1;
  });

  return { items };
}

    const startKey = _evDateKey_(startAt);

    if (fromIso && startKey && startKey < fromIso) continue;
    if (toIso && startKey && startKey > toIso) continue;

    const rrid = (idx.roomId >= 0) ? String(r[idx.roomId] ?? "").trim() : "";
    const meta = rrid ? (metaMap[_evNorm_(rrid)] || null) : null;
    const klientName = meta ? String(meta.klient || "").trim() : "";
    const lokalizacja = meta ? String(meta.lokalizacja || "").trim() : "";
    if (klientF && _evNorm_(klientName) !== klientF) continue;

    const st = (idx.strefa >= 0) ? String(r[idx.strefa] ?? "").trim() : "";
    if (strefaF && _evNorm_(st) !== strefaF) continue;

    const endAt = (idx.endAt >= 0) ? _evCellToIso_(r[idx.endAt]) : "";
    if (!endAt) continue; // tylko zakończone

    const dur = (idx.durationSec >= 0) ? Number(r[idx.durationSec] ?? 0) : 0;
    const workerName = (idx.workerName >= 0) ? String(r[idx.workerName] ?? "").trim() : "";
    const pm = (idx.pomieszczenie >= 0) ? String(r[idx.pomieszczenie] ?? "").trim() : "";
    const stt = (idx.status >= 0) ? String(r[idx.status] ?? "").trim() : "";
    const cmt = (idx.comment >= 0) ? String(r[idx.comment] ?? "").trim() : "";

    items.push({
      cycleId: cycleId,
      roomId: rrid,
      klient: klientName,
      strefa: st,
      pomieszczenie: pm,
      workerName: workerName,
      startAt: startAt,
      endAt: endAt,
      durationSec: isFinite(dur) ? dur : 0,
      status: stt,
      comment: cmt
    });
  }

  // sort newest first
  items.sort((a,b)=> String(b.startAt||"").localeCompare(String(a.startAt||"")));
  return { items: items };
}

/**
 * Dashboard: Podsumowanie (dzisiaj).
 * - openNoStop: unikalni pracownicy, którzy mają dzisiejszy START i brak STOP (w Zdarzeniach).
 * - over9h: ci z openNoStop, u których (teraz - START) > 9h.
 *
 * Zwraca dane JSON-serializable (bez obiektów Date).
 */
function api_getDashboardSummary(params){
  params = params || {};
  _authRequireRead_(String(params.token || "").trim());

  const sh = _evGetSheet_();
  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow <= EVENTS_SHEET.headerRow) {
    return { ok:true, data:{ openNoStop:{count:0, workers:[]}, over9h:{count:0, workers:[]} } };
  }

  const headers = sh.getRange(EVENTS_SHEET.headerRow, 1, 1, lastCol).getValues()[0].map(v => String(v ?? "").trim());
  const idx = _evBuildIdx_(headers);

  // Dodatkowe kolumny (często są osobno): Data + Start (czas) + Stop (czas)
  const norm = headers.map(h => _evNorm_(h));
  const idxDate = _evFindCol_(norm, ["data","date","dzień","dzien"], ["data","date","dzien"]);
  // idx.startAt/idx.endAt wskazują zwykle na kolumny Start/Stop (czas). Data może być osobno.

  const data = sh.getRange(EVENTS_SHEET.headerRow + 1, 1, lastRow - EVENTS_SHEET.headerRow, lastCol).getValues();

  const tz = Session.getScriptTimeZone() || "Europe/Warsaw";
  const now = new Date();

  function _toDate_(v){
    if (v === null || v === undefined || v === "") return null;
    if (Object.prototype.toString.call(v) === "[object Date]" && !isNaN(v.getTime())) return v;

    const s = String(v).trim();
    if (!s) return null;

    // dd.mm.yyyy / dd-mm-yyyy / dd/mm/yyyy
    const m = s.match(/^(\d{1,2})[.\-\/](\d{1,2})[.\-\/](\d{4})/);
    if (m){
      const dd = Number(m[1]), mm = Number(m[2]) - 1, yy = Number(m[3]);
      const d = new Date(yy, mm, dd);
      if (!isNaN(d.getTime())) return d;
    }

    // time-only "HH:MM" or "HH:MM:SS" -> zwracamy Date z rokiem 1900 (wykryjemy to w _combineDateTime_)
    const mt = s.match(/^(\d{1,2}):(\d{2})(?::(\d{2}))?$/);
    if (mt){
      const hh = Number(mt[1]), mi = Number(mt[2]), ss = Number(mt[3] || 0);
      const d = new Date(1900, 0, 1, hh, mi, ss);
      if (!isNaN(d.getTime())) return d;
    }

    const d2 = new Date(s);
    if (!isNaN(d2.getTime())) return d2;
    return null;
  }

  function _combineDateTime_(dateCell, timeCell){
    // jeśli nie ma daty — zwracamy timeCell (może być pełna data)
    const d = _toDate_(dateCell);
    const t = _toDate_(timeCell);

    if (!t) return null;

    // jeśli timeCell ma pełną datę (rok sensowny), bierzemy go wprost
    const y = t.getFullYear();
    if (y >= 2000 && y <= 2100) return t;

    // jeśli timeCell wygląda na "czas" (rok 1899/1900), to dołóż do daty z kolumny Data
    if (!d) return null;

    return new Date(d.getFullYear(), d.getMonth(), d.getDate(), t.getHours(), t.getMinutes(), t.getSeconds());
  }

  function _isClosedByStatus_(st){
    const s = String(st || "").trim().toLowerCase();
    if (!s) return false;
    if (s === "closed" || s === "zamkniete" || s === "zamknięte") return true;
    if (s.indexOf("clos") >= 0) return true;
    if (s.indexOf("zamk") >= 0) return true;
    return false;
  }

  // Map: 1 aktywny (otwarty) wpis per pracownik (najwcześniejszy START)
  const openMap = {}; // key -> worker object

  // wykryj opcjonalne workerId/login raz
  let widIdx = -1;
  try {
    widIdx = _evFindCol_(norm,
      ["workerid","worker id","login","idpracownika","id pracownika","pracownikid","pracownik id"],
      ["workerid","login","idprac","pracownikid"]
    );
  } catch (e) {}

  for (let i = 0; i < data.length; i++){
    const r = data[i];

    const workerName = (idx.workerName >= 0) ? String(r[idx.workerName] ?? "").trim() : "";
    if (!workerName) continue;

    const workerId = (widIdx >= 0) ? String(r[widIdx] ?? "").trim() : "";

    // STATUS: jeśli już CLOSED/Zamknięte -> pomijamy (nawet jeśli Stop nie jest wpisany)
    const statusVal = (idx.status >= 0) ? String(r[idx.status] ?? "").trim() : "";
    if (_isClosedByStatus_(statusVal)) continue;

    // START/STOP jako datetime (Data + Start/Stop)
    const dateCell = (idxDate >= 0) ? r[idxDate] : null;

    const startCell = (idx.startAt >= 0) ? r[idx.startAt] : null;
    const startDt = _combineDateTime_(dateCell, startCell);
    if (!startDt) continue;

    const stopCell = (idx.endAt >= 0) ? r[idx.endAt] : null;
    const stopRaw = String(stopCell ?? "").trim();
    const endDt = _combineDateTime_(dateCell, stopCell);

    // zamknięte, jeśli:
    // - da się zbudować endDt
    // - LUB cokolwiek jest wpisane w komórce Stop (nawet jeśli to tekst "15:00")
    if (endDt || stopRaw) continue;

    const strefa = (idx.strefa >= 0) ? String(r[idx.strefa] ?? "").trim() : "";
    const klient = (idx.pomieszczenie >= 0) ? String(r[idx.pomieszczenie] ?? "").trim() : "";

    const key = workerId ? ("ID:" + workerId) : ("NM:" + workerName.toLowerCase());

    const startT = startDt.getTime();
    if (!isFinite(startT)) continue;

    const durH = (now.getTime() - startT) / 3600000;

    const cur = openMap[key];
    // wybieramy NAJWCZEŚNIEJSZY START (żeby jak ktoś ma kilka otwartych wpisów, liczyć od najstarszego)
    if (!cur || new Date(cur.startAt).getTime() > startT){
      openMap[key] = {
        workerId: workerId,
        workerName: workerName,
        strefa: strefa,
        klient: klient,
        // spójność z frontendem: w JS jest używane w.startAt
        startAt: Utilities.formatDate(startDt, tz, "yyyy-MM-dd'T'HH:mm:ss"),
        durationHours: Math.round(durH * 10) / 10
      };
    }
  }

  const openArr = Object.keys(openMap).map(k => openMap[k]).filter(Boolean);
  openArr.sort((a,b)=> (Number(b.durationHours||0) - Number(a.durationHours||0)));

  // ------------------------------------------------------------
  // over9h (zakończone dni) — MUSI odpowiadać czerwonym wierszom
  // na liście "Czas pracy pracownika"
  // Źródło: Workday_Log (WorkerName/StartAt/EndAt) + opcjonalnie WT_AlertAck
  // Zakres: bieżący miesiąc
  // ------------------------------------------------------------
  let overArr = [];
  try {
    // Map WorkerName -> WorkerID (z arkusza Pracownicy), żeby klik w modalu mógł przejść do czasu pracy
    const nameToId = {};
    try {
      const ws = _getWorkersSheet_();
      const lr = ws.getLastRow();
      if (lr > WORKERS_SHEET.headerRow) {
        const vals = ws.getRange(WORKERS_SHEET.headerRow + 1, 1, lr - WORKERS_SHEET.headerRow, 3).getValues(); // A..C
        for (let i=0;i<vals.length;i++){
          const id = String(vals[i][0] ?? "").trim();
          const nm = String(vals[i][1] ?? "").trim();
          if (!nm) continue;
          const key = nm.replace(/\s+/g," ").trim().toLowerCase();
          if (key && id && !nameToId[key]) nameToId[key] = id;
        }
      }
    } catch(e){}

    const shW = _getWorkdayLogSheet_();
    const lrW = shW.getLastRow();
    const lcW = shW.getLastColumn();

    if (lrW > WORKDAY_LOG_SHEET.headerRow) {
      const headersW = shW.getRange(WORKDAY_LOG_SHEET.headerRow, 1, 1, lcW).getValues()[0].map(v => String(v ?? "").trim());
      const map = {};
      for (let c=0;c<headersW.length;c++){
        const h = String(headersW[c]||"").trim();
        if (h) map[h] = c;
      }

      // Preferuj dokładne nazwy z Workday_Log (jak w module WorkerTime)
      let idxWorkerName = (map["WorkerName"] != null) ? map["WorkerName"] : -1;
      let idxStartAt    = (map["StartAt"]    != null) ? map["StartAt"]    : -1;
      let idxEndAt      = (map["EndAt"]      != null) ? map["EndAt"]      : -1;
      let idxAck        = (map["WT_AlertAck"]!= null) ? map["WT_AlertAck"]: -1;

      // Fallback (gdyby ktoś zmienił nagłówki)
      const normW = headersW.map(h => String(h||"").trim().toLowerCase());
      const findAny = (exact, contains) => {
        for (let i=0;i<normW.length;i++){
          const h = normW[i];
          for (const k of exact){ if (h === k) return i; }
        }
        for (let i=0;i<normW.length;i++){
          const h = normW[i];
          for (const k of contains){ if (h.indexOf(k) >= 0) return i; }
        }
        return -1;
      };
      if (idxWorkerName < 0) idxWorkerName = findAny(["workername","pracownik","imię i nazwisko","imie i nazwisko","imię","imie"], ["workername","prac","nazw"]);
      if (idxStartAt < 0)    idxStartAt    = findAny(["startat","start","początek","poczatek"], ["start","pocz"]);
      if (idxEndAt < 0)      idxEndAt      = findAny(["endat","end","stop","koniec"], ["end","stop","koni"]);
      if (idxAck < 0)        idxAck        = findAny(["wt_alertack","alertack"], ["alertack","ack"]);

      if (idxWorkerName < 0 || idxStartAt < 0 || idxEndAt < 0) {
        throw new Error("Workday_Log: brakuje wymaganych kolumn (WorkerName/StartAt/EndAt).");
      }

      function isTrue(v){
        if (v === true) return true;
        const s = String(v ?? "").trim().toLowerCase();
        return (s === "true" || s === "1" || s === "tak" || s === "yes" || s === "y");
      }
      function toDate(v){
        if (v === null || v === undefined || v === "") return null;
        if (Object.prototype.toString.call(v) === "[object Date]" && !isNaN(v.getTime())) return v;
        const s = String(v).trim();
        if (!s) return null;
        const d = new Date(s);
        return isNaN(d.getTime()) ? null : d;
      }
      function dateKey(d){
        try { return Utilities.formatDate(d, tz, "yyyy-MM-dd"); } catch(e){ return ""; }
      }
      function iso(d){
        try { return Utilities.formatDate(d, tz, "yyyy-MM-dd'T'HH:mm:ss"); } catch(e){ return ""; }
      }
      function lvlFromSec(sec){
        const s = Number(sec||0);
        if (!isFinite(s) || s <= 0) return 0;
        if (s > 10*3600) return 2;
        if (s > 9*3600) return 1;
        return 0;
      }

      // Bieżący miesiąc
      const monthLabel = Utilities.formatDate(now, tz, "yyyy-MM");
      const y = parseInt(monthLabel.slice(0,4),10);
      const m = parseInt(monthLabel.slice(5,7),10); // 01..12
      const fromYmd = monthLabel + "-01";
      const toYmd = Utilities.formatDate(new Date(y, m, 0), tz, "yyyy-MM-dd");

      const dataW = shW.getRange(WORKDAY_LOG_SHEET.headerRow + 1, 1, lrW - WORKDAY_LOG_SHEET.headerRow, lcW).getValues();

      // 1 wpis na pracownika (jak czerwone podświetlenie): max level w miesiącu
      const byWorker = {}; // workerNorm -> item

      for (let i=0;i<dataW.length;i++){
        const r = dataW[i];
        const wn = String(r[idxWorkerName] ?? "").trim();
        if (!wn) continue;

        if (idxAck >= 0 && isTrue(r[idxAck])) continue;

        const ds = toDate(r[idxStartAt]);
        const de = toDate(r[idxEndAt]);
        if (!ds || !de) continue; // tylko zakończone dni

        const dk = dateKey(ds);
        if (!dk) continue;
        if (dk < fromYmd || dk > toYmd) continue;

        const workSec = Math.max(0, Math.round((de.getTime() - ds.getTime())/1000));
        const lvl = lvlFromSec(workSec);
        if (lvl <= 0) continue;

        const wnNorm = wn.replace(/\s+/g," ").trim().toLowerCase();
        const item = {
          workerId: nameToId[wnNorm] || "",
          workerName: wn,
          dateKey: dk,
          startAt: iso(ds),
          endAt: iso(de),
          workSec: workSec,
          level: lvl
        };

        const cur = byWorker[wnNorm];
        if (!cur) {
          byWorker[wnNorm] = item;
        } else {
          if (item.level > (cur.level||0)) byWorker[wnNorm] = item;
          else if (item.level === (cur.level||0) && String(item.dateKey) > String(cur.dateKey)) byWorker[wnNorm] = item;
          else if (item.level === (cur.level||0) && String(item.dateKey) === String(cur.dateKey) && item.workSec > (cur.workSec||0)) byWorker[wnNorm] = item;
        }
      }

      overArr = Object.keys(byWorker).map(k => byWorker[k]).filter(Boolean);
      overArr.sort((a,b)=>{
        if ((b.level||0) !== (a.level||0)) return (b.level||0) - (a.level||0);
        if (a.dateKey !== b.dateKey) return (a.dateKey < b.dateKey) ? 1 : -1;
        return Number(b.workSec||0) - Number(a.workSec||0);
      });
    }
  } catch(e) {
    overArr = [];
  }
return {
    ok: true,
    data: {
      openNoStop: { count: openArr.length, workers: openArr },
      over9h: { count: overArr.length, workers: overArr }
    }
  };
}

function _clEnsureSheet_(name, headers){
  const ss = _clDb_();
  let sh = ss.getSheetByName(name);

  if (!sh){
    sh = ss.insertSheet(name);
    sh.getRange(1,1,1,headers.length).setValues([headers]);
    sh.setFrozenRows(1);
    return sh;
  }

  const lastCol = Math.max(sh.getLastColumn(), headers.length);
  const row = sh.getRange(1,1,1,lastCol).getValues()[0].map(v=>String(v??"").trim());

  const allEmpty = row.every(v=>!v);
  if (allEmpty){
    sh.getRange(1,1,1,headers.length).setValues([headers]);
    sh.setFrozenRows(1);
    return sh;
  }

  for (let i=0;i<headers.length;i++){
    const got = String(row[i]??"").trim();
    const exp = headers[i];
    if (got !== exp){
      throw new Error("Nieprawidłowe nagłówki w arkuszu '"+name+"' (kol "+(i+1)+"). Oczekuję: '"+exp+"', mam: '"+got+"'.");
    }
  }
  return sh;
}

function _clCsvEscape_(v){
  const s = String(v ?? "");
  if (/[",\n\r]/.test(s)) return '"' + s.replace(/"/g,'""') + '"';
  return s;
}
function _clToCsv_(headers, rows){
  const lines = [];
  lines.push(headers.map(_clCsvEscape_).join(","));
  rows.forEach(r=>{
    lines.push(headers.map(h=>_clCsvEscape_(r[h])).join(","));
  });
  return lines.join("\n");
}

function api_checklists_daily_create(payload){
  payload=payload||{};
  _clEnsureAll_();
  const token=String(payload.token||"").trim();

function api_checklists_daily_move(payload){
  payload=payload||{};
  _clEnsureAll_();
  const token=String(payload.token||"").trim();
  const u=_authRequireWrite_(token);

  const qr=String(payload.qr||"").trim();
  const taskId=String(payload.taskId||"").trim();
  const dir=String(payload.dir||"").trim();

  _clReqNonEmpty_(qr, "Brak QrId.");
  _clReqNonEmpty_(taskId, "Brak TaskId.");
  if(["UP","DOWN"].indexOf(dir)<0) throw new Error("dir tylko UP/DOWN.");

  const all=_clDailyReadAll_().filter(x=>x.QrId===qr && String(x.Active).trim()==="TAK")
                             .sort((a,b)=>(Number(a.Order)||0)-(Number(b.Order)||0));
  const idx=all.findIndex(x=>x.TaskId===taskId);
  if(idx<0) throw new Error("TaskId nie należy do strefy.");
  const swapIdx=(dir==="UP") ? idx-1 : idx+1;
  if(swapIdx<0 || swapIdx>=all.length) return { ok:true };

  const a=all[idx], b=all[swapIdx];
  const oa=Number(a.Order)||0, ob=Number(b.Order)||0;

  const sh=_clDailySheet_();
  const ra=_clDailyFindRowByTaskId_(a.TaskId);
  const rb=_clDailyFindRowByTaskId_(b.TaskId);
  if(!ra || !rb) throw new Error("Nie mogę znaleźć wierszy do zmiany kolejności.");

  const now=_clNowIso_();
  const by=String((u && (u.name || u.login)) || "").trim();

  sh.getRange(ra,7).setValue(ob);
  sh.getRange(ra,9).setValue(now);
  sh.getRange(ra,10).setValue(by);

  sh.getRange(rb,7).setValue(oa);
  sh.getRange(rb,9).setValue(now);
  sh.getRange(rb,10).setValue(by);

  return { ok:true };
}

function api_checklists_daily_copy(payload){
  payload=payload||{};
  _clEnsureAll_();
  const token=String(payload.token||"").trim();
  const u=_authRequireWrite_(token);

  const fromQr=String(payload.fromQr||"").trim();
  const toQr=String(payload.toQr||"").trim();
  const mode=String(payload.mode||"MERGE").trim();

  _clReqNonEmpty_(fromQr, "Brak fromQr.");
  _clReqNonEmpty_(toQr, "Brak toQr.");
  if(fromQr===toQr) throw new Error("Źródło i cel nie mogą być takie same.");
  if(["MERGE","REPLACE"].indexOf(mode)<0) throw new Error("mode tylko MERGE/REPLACE.");

  const sh=_clDailySheet_();
  const all=_clDailyReadAll_();

  const src=all.filter(x=>x.QrId===fromQr && String(x.Active).trim()==="TAK")
               .sort((a,b)=>(Number(a.Order)||0)-(Number(b.Order)||0));
  if(!src.length) return { ok:true, copied: 0 };

  const destActive=all.filter(x=>x.QrId===toQr && String(x.Active).trim()==="TAK");

  const now=_clNowIso_();
  const by=String((u && (u.name || u.login)) || "").trim();

  if(mode==="REPLACE" && destActive.length){
    destActive.forEach(it=>{
      const rn=_clDailyFindRowByTaskId_(it.TaskId);
      if(!rn) return;
      sh.getRange(rn,8).setValue("NIE");
      sh.getRange(rn,9).setValue(now);
      sh.getRange(rn,10).setValue(by);
    });
  }

  const existingNames={};
  if(mode==="MERGE"){
    destActive.forEach(it=>{
      const nm=String(it.TaskName||"").trim().toLowerCase();
      if(nm) existingNames[nm]=true;
    });
  }

  const meta=_clGetZoneMetaByQr_(toQr);
  let baseOrder=_clDailyNextOrder_(destActive);
  let copied=0;

  src.forEach(s=>{
    const nm=String(s.TaskName||"").trim().toLowerCase();
    if(mode==="MERGE" && nm && existingNames[nm]) return;

    const taskId="T_"+Utilities.getUuid();
    const order=baseOrder; baseOrder+=10;
    sh.appendRow([meta.client, meta.object, toQr, taskId, s.TaskName, s.Required, order, "TAK", now, by]);
    copied++;
  });

  return { ok:true, copied: copied };
}

/** ======= CHECKLISTS: HISTORIA KLIENTA (wszystkie strefy) ======= */
function _clZonesForClient_(clientName){
  const cfg = _clCfg_();
  const ss = SpreadsheetApp.openById(cfg.spreadsheetId);
  const sh = ss.getSheetById(cfg.zonesGid);
  if (!sh) return [];

  const client = String(clientName || "").trim();
  if (!client) return [];

  const lastRow = sh.getLastRow();
  const hr = cfg.zonesHeaderRow;
  if (lastRow <= hr) return [];

  // A..G: ID, Klient, Strefa, Funkcja, Edytował, Data, Lokalizacja
  const data = sh.getRange(hr + 1, 1, lastRow - hr, 7).getValues();
  const out = [];
  for (let i=0;i<data.length;i++){
    const r = data[i];
    const qr = String(r[0] || "").trim();
    const c = String(r[1] || "").trim();
    const strefa = String(r[2] || "").trim();
    const lok = String(r[6] || "").trim();
    if (!qr) continue;
    if (c !== client) continue;
    out.push({ qr: qr, strefa: strefa, lokalizacja: lok });
  }
  return out;
}

function _clZoneMapForClient_(clientName){
  const arr = _clZonesForClient_(clientName);
  const map = {};
  arr.forEach(z=>{
    map[z.qr] = { Zone: z.strefa || "", Location: z.lokalizacja || "" };
  });
  return map;
}
