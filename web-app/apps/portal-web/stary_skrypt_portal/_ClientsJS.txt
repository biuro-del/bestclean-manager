<script>
(function(){
  const $ = (sel) => document.querySelector(sel);
  let allClients = [];
  let coordinators = [];

  window.ClientsModule = {
    init: function() {
      google.script.run.withSuccessHandler(list => {
        coordinators = list;
        fillCoordinatorsSelect();
      }).api_getCoordinators();
      this.fetch();
    },

    fetch: function() {
      $("#clientsBody").innerHTML = '<tr><td colspan="7" style="text-align:center; padding:40px;">Pobieranie bazy klientów...</td></tr>';
      google.script.run.withSuccessHandler(res => {
        allClients = res.items;
        renderTable(allClients);
      }).api_listClients();
    }
  };

  window.renderTable = function(data) {
    const tbody = $("#clientsBody");
    tbody.innerHTML = "";
    
    const userRole = sessionStorage.getItem("userRole"); 
    const isAdmin = (userRole === "ADMIN");

    if(!data || data.length === 0) {
      tbody.innerHTML = '<tr><td colspan="7" style="text-align:center; padding:40px; color:#94a3b8;">Brak wyników do wyświetlenia.</td></tr>';
      return;
    }

    data.forEach(item => {
      const tr = document.createElement("tr");
      const statusClass = item.status === 'Aktywny' ? 'badge-active' : 'badge-inactive';
      
      let actionsHtml = `
        <button class="btn-fido btn-outline" style="padding: 6px 12px; font-size:12px;" onclick='openEditModal(${JSON.stringify(item)})'>Edytuj</button>
      `;
      
      if(isAdmin) {
        actionsHtml += `<button class="btn-danger" onclick="deleteClient(${item.rowNumber})">Usuń</button>`;
      }

      tr.innerHTML = `
        <td style="color:#94a3b8; font-family:monospace; font-size:12px;">${item.id || '—'}</td>
        <td style="font-weight:600; color:#1e293b;">${item.nazwa}</td>
        <td>${item.nip || '—'}</td>
        <td>${item.miasto || '—'}</td>
        <td><span class="badge ${statusClass}">${item.status}</span></td>
        <td style="color:#475569;">${item.koordynator || 'Nieprzypisany'}</td>
        <td style="text-align:right; white-space:nowrap;">${actionsHtml}</td>
      `;
      tbody.appendChild(tr);
    });
  };

  window.filterClientsTable = function() {
    const txt = $("#clSearchInput").value.toLowerCase();
    const stat = $("#clSearchStatus").value;

    const filtered = allClients.filter(c => {
      const matchTxt = (c.nazwa.toLowerCase().includes(txt) || String(c.nip).includes(txt) || c.miasto.toLowerCase().includes(txt));
      const matchStat = stat ? (c.status === stat) : true;
      return matchTxt && matchStat;
    });
    renderTable(filtered);
  };

  function fillCoordinatorsSelect() {
    const sel = $("#clKoordynator");
    sel.innerHTML = '<option value="">-- wybierz --</option>';
    coordinators.forEach(c => {
      const opt = document.createElement("option");
      opt.value = c; opt.textContent = c;
      sel.appendChild(opt);
    });
  }

  window.openClientModal = function() {
    $("#clModalTitle").textContent = "Nowy Klient";
    $("#clRowNumber").value = "";
    $("#clId").value = "(automatyczne)";
    $("#clNazwa").value = "";
    $("#clNip").value = "";
    $("#clMiasto").value = "";
    $("#clAdres").value = "";
    $("#clStatus").value = "Aktywny";
    $("#clKoordynator").value = "";
    $("#clModal").style.display = "flex";
  };

  window.openEditModal = function(item) {
    $("#clModalTitle").textContent = "Edycja Klienta";
    $("#clRowNumber").value = item.rowNumber;
    $("#clId").value = item.id;
    $("#clNazwa").value = item.nazwa;
    $("#clNip").value = item.nip;
    $("#clMiasto").value = item.miasto;
    $("#clAdres").value = item.adres;
    $("#clStatus").value = item.status;
    $("#clKoordynator").value = item.koordynator;
    $("#clModal").style.display = "flex";
  };

  window.closeClModal = function() { $("#clModal").style.display = "none"; };

  window.saveClientData = function() {
    const payload = {
      rowNumber: $("#clRowNumber").value,
      id: $("#clId").value === "(automatyczne)" ? "" : $("#clId").value,
      nazwa: $("#clNazwa").value,
      nip: $("#clNip").value,
      miasto: $("#clMiasto").value,
      adres: $("#clAdres").value,
      status: $("#clStatus").value,
      koordynator: $("#clKoordynator").value
    };
    if(!payload.nazwa) return alert("Nazwa klienta jest wymagana!");
    google.script.run.withSuccessHandler(() => {
      closeClModal();
      window.ClientsModule.fetch();
    }).api_saveClient(payload);
  };

  window.deleteClient = function(rowNum) {
    if(confirm("Czy na pewno chcesz usunąć tego klienta z bazy?")) {
      google.script.run.withSuccessHandler(() => {
        window.ClientsModule.fetch();
      }).api_deleteClient({rowNumber: rowNum});
    }
  };
})();
</script>
