<script>
/**
 * Audyty — etap 1 (UI)
 * - widok + toolbar (wyszukiwanie, status, odśwież)
 * - bez backendu (listę + API dodamy w kolejnym etapie)
 */
(function(){
  const $ = (sel, root=document) => root.querySelector(sel);

  function _getEls_(){
    const view = document.getElementById("view-audits");
    if (!view) return null;
    return {
      view,
      q: $("#auditSearch", view),
      status: $("#auditStatus", view),
      refresh: $("#auditRefresh", view),
      hint: $("#auditHint", view),
    };
  }

  function _renderHint_(){
    const els = _getEls_();
    if (!els || !els.hint) return;

    const q = String((els.q && els.q.value) || "").trim();
    const st = String((els.status && els.status.value) || "").trim();

    const parts = [];
    if (q) parts.push("Szukasz: \"" + q + "\"");
    if (st) parts.push("Status: " + st);

    els.hint.textContent = parts.length
      ? (parts.join(" • ") + " • (UI gotowe — dane podepniemy w kolejnym etapie)")
      : "Na razie dodaliśmy tylko UI. W kolejnym etapie podepniemy pobieranie listy audytów z Google Sheets.";
  }

  function init(){
    const els = _getEls_();
    if (!els) return;

    if (els.view.__auditsBound) {
      _renderHint_();
      return;
    }
    els.view.__auditsBound = true;

    if (els.q) els.q.addEventListener("input", _renderHint_);
    if (els.status) els.status.addEventListener("change", _renderHint_);
    if (els.refresh) els.refresh.addEventListener("click", _renderHint_);

    _renderHint_();
  }

  window.AuditsModule = { init };
})();
</script>
