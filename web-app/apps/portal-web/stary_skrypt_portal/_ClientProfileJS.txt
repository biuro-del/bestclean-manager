<script>
(function(){
  const $ = (sel) => document.querySelector(sel);

  let allClientsData = [];
  let cpOriginal = null;   // snapshot z listy
  let cpCurrent = null;    // aktualnie w modalu
  let cpIsEdit = false;
  let cpKoordsLoaded = false;

  function getToken(){
    return (window.FIDO_AUTH ? window.FIDO_AUTH.getToken() : "");
  }
  function getUserName(){
    return (window.FIDO_AUTH ? window.FIDO_AUTH.getUserName() : "Użytkownik");
  }
  function canEdit(){
    try {
      return window.FIDO_AUTH && window.FIDO_AUTH.roleLevel(window.FIDO_AUTH.getUserRole()) >= 2;
    } catch(e){
      return false;
    }
  }
  function txt(v, dash=true){
    const s = (v === null || v === undefined) ? "" : String(v);
    const t = s.trim();
    return (t || !dash) ? t : "—";
  }

  function setText(id, value){
    const el = document.getElementById(id);
    if (el) el.textContent = value;
  }

  function setInput(id, value){
    const el = document.getElementById(id);
    if (!el) return;
    el.value = value === null || value === undefined ? "" : String(value);
  }

  function togglePair(viewId, editId, isEdit){
    const v = document.getElementById(viewId);
    const e = document.getElementById(editId);
    if (v) v.style.display = isEdit ? "none" : "";
    if (e) e.style.display = isEdit ? "" : "none";
  }

  function ensureStatusOption(value){
    const sel = document.getElementById('edit-status');
    if (!sel) return;
    const val = String(value || '').trim();
    if (!val) return;
    const exists = Array.from(sel.options).some(o => String(o.value).toLowerCase() === val.toLowerCase());
    if (!exists){
      const opt = document.createElement('option');
      opt.value = val;
      opt.textContent = val;
      sel.appendChild(opt);
    }
  }

  function fillView(item){
    setText('view-id', txt(item.id));
    setText('view-status', txt(item.status));
    setText('view-nazwa', txt(item.nazwa));
    setText('view-nip', txt(item.nip));
    setText('view-koordynator', txt(item.koordynator));
    setText('view-kontakt', txt(item.kontakt));
    setText('view-miasto', txt(item.miasto));
    setText('view-adres', txt(item.adres));
    setText('view-czestotliwosc', txt(item.czestotliwosc));
    setText('view-osoby', txt(item.osoby));
    setText('view-chemia', txt(item.chemia));
    setText('view-sprzet', txt(item.sprzet));
    setText('view-informacje', txt(item.informacje));

    const title = document.getElementById('cpModalTitle');
    if (title) title.textContent = 'Szczegóły Klienta: ' + (item.nazwa ? String(item.nazwa) : '—');
  }

  function fillEdit(item){
    setInput('edit-nazwa', item.nazwa);
    setInput('edit-nip', item.nip);
    setInput('edit-miasto', item.miasto);
    setInput('edit-adres', item.adres);
    setInput('edit-kontakt', item.kontakt);
    ensureStatusOption(item.status);
    setInput('edit-status', item.status);
    setInput('edit-koordynator', item.koordynator);
    setInput('edit-czestotliwosc', item.czestotliwosc);
    setInput('edit-osoby', item.osoby);
    setInput('edit-chemia', item.chemia);
    setInput('edit-sprzet', item.sprzet);
    setInput('edit-informacje', item.informacje);
  }

  function collectEdit(){
    return {
      nazwa: (document.getElementById('edit-nazwa') || {}).value || '',
      nip: (document.getElementById('edit-nip') || {}).value || '',
      miasto: (document.getElementById('edit-miasto') || {}).value || '',
      adres: (document.getElementById('edit-adres') || {}).value || '',
      kontakt: (document.getElementById('edit-kontakt') || {}).value || '',
      status: (document.getElementById('edit-status') || {}).value || '',
      koordynator: (document.getElementById('edit-koordynator') || {}).value || '',
      czestotliwosc: (document.getElementById('edit-czestotliwosc') || {}).value || '',
      osoby: (document.getElementById('edit-osoby') || {}).value || '',
      chemia: (document.getElementById('edit-chemia') || {}).value || '',
      sprzet: (document.getElementById('edit-sprzet') || {}).value || '',
      informacje: (document.getElementById('edit-informacje') || {}).value || ''
    };
  }

  function setMode(isEdit){
    cpIsEdit = !!isEdit;

    togglePair('view-status', 'edit-status', cpIsEdit);
    togglePair('view-nazwa', 'edit-nazwa', cpIsEdit);
    togglePair('view-nip', 'edit-nip', cpIsEdit);
    togglePair('view-koordynator', 'edit-koordynator', cpIsEdit);
    togglePair('view-kontakt', 'edit-kontakt', cpIsEdit);
    togglePair('view-miasto', 'edit-miasto', cpIsEdit);
    togglePair('view-adres', 'edit-adres', cpIsEdit);
    togglePair('view-czestotliwosc', 'edit-czestotliwosc', cpIsEdit);
    togglePair('view-osoby', 'edit-osoby', cpIsEdit);
    togglePair('view-chemia', 'edit-chemia', cpIsEdit);
    togglePair('view-sprzet', 'edit-sprzet', cpIsEdit);
    togglePair('view-informacje', 'edit-informacje', cpIsEdit);

    const btnEdit = document.getElementById('cpBtnEdit');
    const btnSave = document.getElementById('cpBtnSave');
    const btnCancel = document.getElementById('cpBtnCancel');

    if (btnEdit) btnEdit.style.display = (!cpIsEdit && canEdit()) ? '' : 'none';
    if (btnSave) btnSave.style.display = cpIsEdit ? '' : 'none';
    if (btnCancel) btnCancel.style.display = cpIsEdit ? '' : 'none';
  }

  function loadCoordinatorsIfNeeded(){
    if (cpKoordsLoaded) return;
    if (!window.google || !google.script || !google.script.run) return;

    google.script.run
      .withSuccessHandler(res => {
        const list = document.getElementById('cpKoordList');
        if (!list) return;
        list.innerHTML = '';

        const items = (res && res.items) ? res.items : [];
        items.forEach(name => {
          const opt = document.createElement('option');
          opt.value = String(name || '').trim();
          if (opt.value) list.appendChild(opt);
        });

        cpKoordsLoaded = true;
      })
      .withFailureHandler(_ => {})
      .api_getCoordinators({ token: getToken() });
  }

  window.ClientProfileModule = {
    init: function() {
      console.log('Inicjalizacja modułu Profilu Klienta...');
      this.fetch();
      this.bindEvents();
    },

    fetch: function() {
      const tbody = $('#clientProfileBody');
      if (tbody) tbody.innerHTML = '<tr><td colspan="4" style="text-align:center; padding:40px;">Pobieranie danych...</td></tr>';

      google.script.run
        .withSuccessHandler(res => {
          console.log('Dane odebrane z serwera:', res);
          allClientsData = (res && res.items) ? res.items : [];
          this.render(allClientsData);
        })
        .withFailureHandler(err => {
          console.error('Błąd serwera:', err);
          if (tbody) tbody.innerHTML = '<tr><td colspan="4" style="color:red; text-align:center;">Błąd pobierania danych.</td></tr>';
        })
        .api_listClients({ token: getToken() });
    },

    render: function(data) {
      const tbody = $('#clientProfileBody');
      if (!tbody) return;
      tbody.innerHTML = '';

      if (!data || data.length === 0) {
        tbody.innerHTML = '<tr><td colspan="4" style="text-align:center; padding:40px;">Brak danych w bazie (Arkusz gid=426771989).</td></tr>';
        return;
      }

      data.forEach(item => {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td style="color:#94a3b8; font-family:monospace;">${txt(item.id)}</td>
          <td style="font-weight:600;">${txt(item.nazwa, false) || '—'}</td>
          <td>${txt(item.nip)}</td>
          <td style="text-align:right;">
            <button class="btn-fido btn-outline" onclick='openCpModal(${JSON.stringify(item)})'>Podgląd</button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    },

    bindEvents: function() {
      const filter = () => {
        const n = ($('#cpSearchName').value || '').toLowerCase();
        const p = ($('#cpSearchNip').value || '').toLowerCase();
        const filtered = allClientsData.filter(c =>
          (c.nazwa || '').toLowerCase().includes(n) && (c.nip || '').toLowerCase().includes(p)
        );
        this.render(filtered);
      };
      if ($('#cpSearchName')) $('#cpSearchName').onkeyup = filter;
      if ($('#cpSearchNip')) $('#cpSearchNip').onkeyup = filter;
    }
  };

  window.openCpModal = function(item){
    cpOriginal = JSON.parse(JSON.stringify(item || {}));
    cpCurrent = JSON.parse(JSON.stringify(item || {}));

    fillView(cpCurrent);
    fillEdit(cpCurrent);
    setMode(false);
    loadCoordinatorsIfNeeded();

    const modal = $('#cpModal');
    if (modal) modal.style.display = 'flex';
  };

  window.closeCpModal = function(){
    const modal = $('#cpModal');
    if (modal) modal.style.display = 'none';
    cpIsEdit = false;
    cpOriginal = null;
    cpCurrent = null;
  };

  window.cpEnterEdit = function(){
    if (!canEdit()){
      alert('Brak uprawnień do edycji (wymagany Kierownik/Admin).');
      return;
    }
    if (!cpCurrent) return;
    fillEdit(cpCurrent);
    setMode(true);
    loadCoordinatorsIfNeeded();
  };

  window.cpCancelEdit = function(){
    if (!cpOriginal) return;
    cpCurrent = JSON.parse(JSON.stringify(cpOriginal));
    fillView(cpCurrent);
    fillEdit(cpCurrent);
    setMode(false);
  };

  window.cpSaveEdit = function(){
    if (!canEdit()){
      alert('Brak uprawnień do edycji (wymagany Kierownik/Admin).');
      return;
    }
    if (!cpCurrent || !cpCurrent.id){
      alert('Brak ID klienta.');
      return;
    }

    const data = collectEdit();
    if (!String(data.nazwa || '').trim()){
      alert('Nazwa klienta jest wymagana.');
      return;
    }

    const btnSave = document.getElementById('cpBtnSave');
    const btnCancel = document.getElementById('cpBtnCancel');
    if (btnSave) { btnSave.disabled = true; btnSave.textContent = 'Zapisywanie...'; }
    if (btnCancel) btnCancel.disabled = true;

    google.script.run
      .withSuccessHandler(_res => {
        // aktualizujemy lokalny stan
        cpCurrent = { ...cpCurrent, ...data };
        cpOriginal = JSON.parse(JSON.stringify(cpCurrent));

        fillView(cpCurrent);
        setMode(false);

        // Odśwież listę, żeby tabela też miała nowe dane
        if (window.ClientProfileModule) window.ClientProfileModule.fetch();

        if (btnSave) { btnSave.disabled = false; btnSave.textContent = 'Zapisz'; }
        if (btnCancel) btnCancel.disabled = false;
      })
      .withFailureHandler(err => {
        const msg = (err && err.message) ? err.message : 'Błąd zapisu.';
        alert(msg);
        if (btnSave) { btnSave.disabled = false; btnSave.textContent = 'Zapisz'; }
        if (btnCancel) btnCancel.disabled = false;
      })
      .api_updateClientProfile({
        token: getToken(),
        id: cpCurrent.id,
        nazwa: data.nazwa,
        nip: data.nip,
        miasto: data.miasto,
        adres: data.adres,
        kontakt: data.kontakt,
        status: data.status,
        koordynator: data.koordynator,
        czestotliwosc: data.czestotliwosc,
        osoby: data.osoby,
        chemia: data.chemia,
        sprzet: data.sprzet,
        informacje: data.informacje,
        editorName: getUserName()
      });
  };
})();
</script>
