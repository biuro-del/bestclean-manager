<script>
/**
 * ReportsModule (Zestawienia) — stable binding + safe delegation
 * Key goals:
 * - tiles always clickable (even if other router handlers are buggy)
 * - open builder immediately (API load errors only affect data, not navigation)
 * - only CLOSED events (endAt present) used for time
 */
(function(){
  if (window.ReportsModule && window.ReportsModule.__v === "2026-01-28") return;

  const $ = (s, r) => (r || document).querySelector(s);
  const $$ = (s, r) => Array.from((r || document).querySelectorAll(s));

  let _bound = false;
  let _lastCsv = null;

  function _token(){
    try{
      if (window.FIDO_AUTH && typeof window.FIDO_AUTH.getToken === "function") return String(window.FIDO_AUTH.getToken() || "");
      if (typeof window.getToken === "function") return String(window.getToken() || "");
      // fallback (na wypadek zmian w routerze / widokach)
      try{
        const a = JSON.parse(sessionStorage.getItem("fido_auth") || "null");
        if (a && a.token) return String(a.token);
      }catch(e){}
      try{
        const a = JSON.parse(localStorage.getItem("fido_auth") || "null");
        if (a && a.token) return String(a.token);
      }catch(e){}
      return "";
    }catch(e){
      return "";
    }
  }

  function _setStatus(msg){
    const el = $("#repStatus");
    if (el) el.textContent = msg || "";
  }

  function _show(id){
    const el = document.getElementById(id);
    if (el) el.style.display = "";
  }
  function _hide(id){
    const el = document.getElementById(id);
    if (el) el.style.display = "none";
  }

  function _today(){
    const d = new Date();
    const y = d.getFullYear();
    const m = String(d.getMonth()+1).padStart(2,"0");
    const dd = String(d.getDate()).padStart(2,"0");
    return `${y}-${m}-${dd}`;
  }

  function _esc(v){
    return String(v ?? "")
      .replaceAll("&","&amp;")
      .replaceAll("<","&lt;")
      .replaceAll(">","&gt;")
      .replaceAll('"',"&quot;")
      .replaceAll("'","&#39;");
  }

  function _fmtSec(sec){
    sec = Number(sec || 0);
    if (!isFinite(sec) || sec < 0) sec = 0;
    const h = Math.floor(sec / 3600);
    const m = Math.floor((sec - h*3600)/60);
    const s = Math.floor(sec - h*3600 - m*60);
    const pad = (n)=> String(n).padStart(2,"0");
    return `${h}:${pad(m)}:${pad(s)}`;
  }

  function _dateKey(iso){
    const s = String(iso || "");
    return s ? s.slice(0,10) : "";
  }

  function _call(fn, params){
    return new Promise((resolve, reject)=>{
      try{
        google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(reject)[fn](params || {});
      }catch(err){
        reject(err);
      }

  // Bind select handlers even if views are hydrated later
  const _selectBinds = new Set();
  function _bindClientZonesOnce(selClientId, selZoneId){
    const key = selClientId + "->" + selZoneId;
    if (_selectBinds.has(key)) return;

    const selClient = document.getElementById(selClientId);
    const selZone = document.getElementById(selZoneId);
    if (!selClient || !selZone) return; // view may not be in DOM yet

    _selectBinds.add(key);

    selClient.addEventListener("change", async ()=>{
      const c = String(selClient.value || "").trim();
      _fillSelect(selZone, [], "— wybierz strefę —");
      if (!c) return;
      try{
        _setStatus(`Ładowanie stref dla: ${c}…`);
        const zones = await _loadZones(c);
        _fillSelect(selZone, zones, "— wybierz strefę —");
        _setStatus(zones && zones.length ? "" : "Brak stref dla tego klienta w zakładce Strefy.");
      }catch(err){
        console.error(err);
        _setStatus("Nie udało się wczytać stref (sprawdź logowanie/token lub dane w zakładce Strefy).");
      }
    });
  }
    });
  }

  async function _loadClients(){
    // api_listClients may ignore paging; keep payload minimal
    const res = await _call("api_listClients", { token: _token() });
    const items = (res && res.items) ? res.items : [];
    return items
      .map(it => String(it.nazwa || it.name || it.klient || "").trim())
      .filter(Boolean)
      .map(name => ({ value:name, label:name }));
  }

  async function _loadZones(clientName){
    const res = await _call("api_reports_listStrefy", { token:_token(), klient: String(clientName||"") });
    const items = (res && res.items) ? res.items : [];
    return items.map(z => ({ value:String(z), label:String(z) }));
  }

  function _fillSelect(sel, opts, ph){
    if (!sel) return;
    const cur = sel.value;
    sel.innerHTML = "";
    const o0 = document.createElement("option");
    o0.value = "";
    o0.textContent = ph || "— wybierz —";
    sel.appendChild(o0);
    (opts||[]).forEach(o=>{
      const op = document.createElement("option");
      op.value = o.value;
      op.textContent = o.label;
      sel.appendChild(op);
    });
    if (cur && (opts||[]).some(o=>o.value===cur)) sel.value = cur;
    else sel.value = "";
  }

  function _getMode(){
    const day = $("#repMode_day");
    return (day && day.checked) ? "day" : "single";
  }

  function _getChecks(){
    const chk = (id)=> {
      const el = document.getElementById(id);
      return !!(el && el.checked);
    };
    return {
      time: chk("repChk_time"),
      count: chk("repChk_count"),
      avg: chk("repChk_avg"),
      minmax: chk("repChk_minmax")
    };
  }

  function _aggByDay(items){
    const m = new Map();
    (items||[]).forEach(it=>{
      const day = _dateKey(it.startAt) || _dateKey(it.endAt);
      if (!day) return;
      const dur = Number(it.durationSec || 0);
      if (!isFinite(dur) || dur <= 0) return;

      const cur = m.get(day) || { sumSec:0, count:0, minSec:null, maxSec:null };
      cur.sumSec += dur;
      cur.count += 1;
      cur.minSec = (cur.minSec==null) ? dur : Math.min(cur.minSec, dur);
      cur.maxSec = (cur.maxSec==null) ? dur : Math.max(cur.maxSec, dur);
      m.set(day, cur);
    });
    return m;
  }

  function _buildSummary(sumA, cntA, sumB, cntB){
    const el = $("#repSummary");
    if (!el) return;
    el.innerHTML = "";

    const mk = (k,v)=>{
      const d = document.createElement("div");
      d.className = "rep-stat";
      d.innerHTML = `<div class="k">${_esc(k)}</div><div class="v">${_esc(v)}</div>`;
      return d;
    };

    el.appendChild(mk("A: czas", _fmtSec(sumA)));
    el.appendChild(mk("A: zdarzenia", String(cntA)));
    el.appendChild(mk("B: czas", _fmtSec(sumB)));
    el.appendChild(mk("B: zdarzenia", String(cntB)));

    el.style.display = "";
  }

  function _tableSingle(itemsA, itemsB, checks){
    const tbl = $("#repTable");
    const wrap = $("#repTableWrap");
    if (!tbl || !wrap) return;

    const cols = [
      { key:"side", label:"Strona" },
      { key:"startAt", label:"Start" },
      { key:"endAt", label:"Stop" },
      { key:"strefa", label:"Strefa" },
      { key:"workerName", label:"Pracownik" }
    ];
    if (checks.time) cols.push({ key:"durationSec", label:"Czas" });

    const rows = [];
    (itemsA||[]).forEach(it=> rows.push({ side:"A", ...it }));
    (itemsB||[]).forEach(it=> rows.push({ side:"B", ...it }));
    rows.sort((a,b)=> String(b.startAt||"").localeCompare(String(a.startAt||"")));

    let html = "<thead><tr>" + cols.map(c=>`<th>${_esc(c.label)}</th>`).join("") + "</tr></thead><tbody>";
    rows.forEach(r=>{
      html += "<tr>";
      cols.forEach(c=>{
        let v = r[c.key];
        if (c.key==="durationSec") v = _fmtSec(v);
        html += `<td>${_esc(v)}</td>`;
      });
      html += "</tr>";
    });
    html += "</tbody>";
    tbl.innerHTML = html;
    wrap.style.display = "";
  }

  function _toCsv(rows){
    const esc = (s)=>{
      const v = String(s ?? "");
      if (/[,"\n]/.test(v)) return '"' + v.replaceAll('"','""') + '"';
      return v;
    };
    return rows.map(r=>r.map(esc).join(",")).join("\n");
  }

  function _downloadCsv(filename, content){
    const blob = new Blob([content], { type:"text/csv;charset=utf-8" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = filename || "zestawienie.csv";
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(()=>URL.revokeObjectURL(url), 500);
  }

  function _tableDay(aggA, aggB, checks){
    const tbl = $("#repTable");
    const wrap = $("#repTableWrap");
    if (!tbl || !wrap) return;

    const days = Array.from(new Set([...(aggA.keys()), ...(aggB.keys())])).sort();
    const cols = [{key:"day", label:"Dzień"}];

    if (checks.time) cols.push({key:"a_time", label:"A: czas"}, {key:"b_time", label:"B: czas"}, {key:"d_time", label:"Δ czas"});
    if (checks.count) cols.push({key:"a_cnt", label:"A: zdarzenia"}, {key:"b_cnt", label:"B: zdarzenia"}, {key:"d_cnt", label:"Δ zdarzeń"});
    if (checks.avg) cols.push({key:"a_avg", label:"A: średni"}, {key:"b_avg", label:"B: średni"});
    if (checks.minmax) cols.push({key:"a_minmax", label:"A: min/max"}, {key:"b_minmax", label:"B: min/max"});

    let html = "<thead><tr>" + cols.map(c=>`<th>${_esc(c.label)}</th>`).join("") + "</tr></thead><tbody>";

    const csvRows = [];
    csvRows.push(cols.map(c=>c.label));

    days.forEach(day=>{
      const A = aggA.get(day) || {sumSec:0,count:0,minSec:null,maxSec:null};
      const B = aggB.get(day) || {sumSec:0,count:0,minSec:null,maxSec:null};

      const row = { day };

      if (checks.time){
        row.a_time = _fmtSec(A.sumSec);
        row.b_time = _fmtSec(B.sumSec);
        row.d_time = _fmtSec(B.sumSec - A.sumSec);
      }
      if (checks.count){
        row.a_cnt = A.count;
        row.b_cnt = B.count;
        row.d_cnt = (B.count - A.count);
      }
      if (checks.avg){
        row.a_avg = A.count ? _fmtSec(Math.round(A.sumSec / A.count)) : "—";
        row.b_avg = B.count ? _fmtSec(Math.round(B.sumSec / B.count)) : "—";
      }
      if (checks.minmax){
        row.a_minmax = (A.minSec==null) ? "—" : `${_fmtSec(A.minSec)} / ${_fmtSec(A.maxSec)}`;
        row.b_minmax = (B.minSec==null) ? "—" : `${_fmtSec(B.minSec)} / ${_fmtSec(B.maxSec)}`;
      }

      html += "<tr>";
      cols.forEach(c=> html += `<td>${_esc(row[c.key])}</td>`);
      html += "</tr>";

      csvRows.push(cols.map(c=>row[c.key]));
    });

    html += "</tbody>";
    tbl.innerHTML = html;
    wrap.style.display = "";

    _lastCsv = {
      filename: `zestawienie_day_${new Date().toISOString().slice(0,10)}.csv`,
      content: _toCsv(csvRows)
    };

    const dl = $("#repDownloadCsv");
    if (dl) dl.disabled = false;
  }

  async function _runEvents(){
    const btn = $("#repRun");
    const dl = $("#repDownloadCsv");
    if (dl) dl.disabled = true;
    _lastCsv = null;

    const checks = _getChecks();
    if (!checks.time && !checks.count && !checks.avg && !checks.minmax){
      _setStatus("Zaznacz przynajmniej jedną metrykę do porównania.");
      return;
    }

    const A = {
      klient: $("#repA_client")?.value || "",
      strefa: $("#repA_zone")?.value || "",
      fromIso: $("#repA_from")?.value || "",
      toIso: $("#repA_to")?.value || ""
    };
    const B = {
      klient: $("#repB_client")?.value || "",
      strefa: $("#repB_zone")?.value || "",
      fromIso: $("#repB_from")?.value || "",
      toIso: $("#repB_to")?.value || ""
    };

    if (!A.klient || !A.strefa || !A.fromIso || !A.toIso || !B.klient || !B.strefa || !B.fromIso || !B.toIso){
      _setStatus("Uzupełnij klienta, strefę oraz zakres dat w panelu A i B.");
      return;
    }

    try{
      if (btn) btn.disabled = true;
      _setStatus("Pobieranie danych (tylko zakończone zdarzenia)…");
      _hide("repTableWrap");
      _hide("repSummary");

      const [resA, resB] = await Promise.all([
        _call("api_reports_listClosedEvents", { token:_token(), ...A }),
        _call("api_reports_listClosedEvents", { token:_token(), ...B })
      ]);

      const itemsA = (resA && resA.items) ? resA.items : [];
      const itemsB = (resB && resB.items) ? resB.items : [];

      const sumA = itemsA.reduce((a,it)=> a + Number(it.durationSec||0), 0);
      const sumB = itemsB.reduce((a,it)=> a + Number(it.durationSec||0), 0);

      _buildSummary(sumA, itemsA.length, sumB, itemsB.length);

      const mode = _getMode();
      if (mode === "day"){
        const aggA = _aggByDay(itemsA);
        const aggB = _aggByDay(itemsB);
        _tableDay(aggA, aggB, checks);
        _setStatus("Gotowe (dzień-do-dnia).");
      } else {
        _tableSingle(itemsA, itemsB, checks);
        _setStatus("Gotowe (pojedyncze zdarzenia).");
      }

    }catch(err){
      console.error(err);
      _setStatus("Błąd pobierania danych. Otwórz konsolę (F12) po szczegóły.");
    }finally{
      if (btn) btn.disabled = false;
    }
  }

  async function open(kind){
    // show builder immediately
    _setStatus("");
    _hide("repHome");
    _show("repBuilder");

    _hide("repEvents");
    _hide("repSoon");
    _hide("repSummary");
    _hide("repTableWrap");
    const dl = $("#repDownloadCsv"); if (dl) dl.disabled = true;
    _lastCsv = null;

    const title = $("#repTitle");
    const sub = $("#repSubtitle");

    if (kind === "events"){
      if (title) title.textContent = "Zdarzenia";
      if (sub) sub.textContent = "Porównanie A vs B (Klient + Strefa) — czas liczony z zakończonych zdarzeń.";
      _show("repEvents");

      // defaults
      const t = _today();
      const af = $("#repA_from"), at = $("#repA_to"), bf = $("#repB_from"), bt = $("#repB_to");
      if (af) af.value = t; if (at) at.value = t;
      if (bf) bf.value = t; if (bt) bt.value = t;

      // load clients + bind zones (safe)
      try{
        _setStatus("Ładowanie listy klientów…");
        const opts = await _loadClients();
        _fillSelect($("#repA_client"), opts, "— wybierz klienta —");
        _fillSelect($("#repB_client"), opts, "— wybierz klienta —");
        _fillSelect($("#repA_zone"), [], "— wybierz strefę —");
        _fillSelect($("#repB_zone"), [], "— wybierz strefę —");

        // bind (na pewno po renderze widoku)
        _bindClientZonesOnce("repA_client", "repA_zone");
        _bindClientZonesOnce("repB_client", "repB_zone");

        _setStatus("Wybierz klienta w panelu A i B (wtedy pojawią się strefy).");
      }catch(err){
        console.error(err);
        _setStatus("Nie udało się wczytać listy klientów (sprawdź token/logowanie).");
      }

      return;
    }

    const titles = { clients:"Klienci", audits:"Audyty", workerTime:"Czasy pracowników" };
    if (title) title.textContent = titles[kind] || "Zestawienie";
    if (sub) sub.textContent = "Ten moduł dodamy w kolejnym kroku.";
    _show("repSoon");
  }

  function back(){
    _setStatus("");
    _hide("repBuilder");
    _show("repHome");
    _lastCsv = null;
  }

  function init(){
    // When entering the reports route, ensure HOME is visible
    try{
      _hide("repBuilder");
      _show("repHome");
      _setStatus("");
      const dl = $("#repDownloadCsv"); if (dl) dl.disabled = true;
      _lastCsv = null;
    }catch(e){
      console.error(e);
    }
  }

  function _bind(){
    if (_bound) return;
    _bound = true;

    // Back
    const backBtn = $("#repBack");
    if (backBtn) backBtn.addEventListener("click", back);

    // Run
    const runBtn = $("#repRun");
    if (runBtn) runBtn.addEventListener("click", _runEvents);

    // CSV
    const dl = $("#repDownloadCsv");
    if (dl) dl.addEventListener("click", ()=>{
      if (!_lastCsv) return;
      _downloadCsv(_lastCsv.filename, _lastCsv.content);
    });

    // Change client => load zones (bind even if view hydrates later)
    _bindClientZonesOnce("repA_client", "repA_zone");
    _bindClientZonesOnce("repB_client", "repB_zone");

    // Direct tile binding (if nothing intercepts)
    $$("[data-rep-open]").forEach(btn=>{
      btn.addEventListener("click", (e)=>{
        e.preventDefault();
        e.stopPropagation();
        open(btn.getAttribute("data-rep-open"));
      });
    });

    // Delegated capture — protects against buggy global click handlers
    document.addEventListener("click", (e)=>{
      const btn = e.target && e.target.closest ? e.target.closest("[data-rep-open]") : null;
      if (!btn) return;
      e.preventDefault();
      e.stopPropagation();
      open(btn.getAttribute("data-rep-open"));
    }, true);
  }

  // Bind ASAP
  document.addEventListener("DOMContentLoaded", ()=>{
    _bind();
    // If user already on reports view, ensure init
    if (document.getElementById("view-reports")) init();
  });

  window.ReportsModule = {
    __v: "2026-02-12-strefy-fix",
    init,
    open
  };
})();
</script>
