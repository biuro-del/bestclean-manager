/**
 * CoordinatorLog repository (Coordinator_Log).
 * Router expects: CoordinatorLog_append_(record)
 */

function CoordinatorLog_sheet_() {
  const H = CONFIG.HEADERS.COORDINATOR_LOG;
  const headers = [
    H.CREATED_AT, H.DATE, H.WORKER_ID, H.ROLE, H.FUNCTION, H.MODULE,
    H.SESSION_ID, H.START_AT, H.STOP_AT, H.DURATION_SEC, H.AUDIT_DEDUCT_SEC,
    H.NOTE, H.META_JSON
  ];

  // Prefer sheetId from config; fallback to create by name if missing.
  try {
    return Sheets_getBySheetId_(CONFIG.SHEET_ID_COORDINATOR_LOG);
  } catch (_) {
    return Sheets_getOrCreateByName_('Coordinator_Log', headers);
  }
}

/**
 * Append one coordinator record into Coordinator_Log.
 * Expected input keys (camelCase):
 * { createdAt, date, workerId, role, functionName, module, sessionId,
 *   startAt, stopAt, durationSec, auditDeductSec, note, metaJson }
 */
function CoordinatorLog_append_(r) {
  const rec = r || {};
  const H = CONFIG.HEADERS.COORDINATOR_LOG;

  const record = {};
  record[H.CREATED_AT] = rec.createdAt || nowIso_();
  record[H.DATE] = rec.date || '';
  record[H.WORKER_ID] = rec.workerId || '';
  record[H.ROLE] = rec.role || '';
  record[H.FUNCTION] = rec.functionName || '';
  record[H.MODULE] = rec.module || '';
  record[H.SESSION_ID] = rec.sessionId || '';
  record[H.START_AT] = rec.startAt || '';
  record[H.STOP_AT] = rec.stopAt || '';
  record[H.DURATION_SEC] = Number(rec.durationSec || 0) || 0;
  record[H.AUDIT_DEDUCT_SEC] = Number(rec.auditDeductSec || 0) || 0;
  record[H.NOTE] = String(rec.note || '').slice(0, 500);
  record[H.META_JSON] = String(rec.metaJson || '').slice(0, 8000);

  const sh = CoordinatorLog_sheet_();

  // Ensure headers exist if sheet is empty
  try {
    if (sh.getLastRow() === 0) {
      const headers = Object.keys(record);
      sh.getRange(1, 1, 1, headers.length).setValues([headers]);
    }
  } catch (_) {}

  const lastCol = sh.getLastColumn();
  const headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(x => String(x || '').trim());
  const row = headers.map(h => (record[h] !== undefined ? record[h] : ''));
  sh.appendRow(row);
}
