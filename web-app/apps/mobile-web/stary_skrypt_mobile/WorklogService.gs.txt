/**
 * Worklog service — API wrapper for CLEAN-only month list & summary.
 * Uses script timezone for month boundaries ("czas aplikacji").
 */
function Worklog_getMonthDaysForWorker_(worker, year, month1to12) {
  const login = String(worker && worker.login || '').trim();
  if (!login) return { days: [] };
  const y = Number(year);
  const m = Number(month1to12);
  const days = Worklog_listMonthDaysClosed_(login, y, m);
  return { year: y, month: m, days: days };
}

function Worklog_getSummaryTwoMonths_(worker, year, month1to12) {
  const login = String(worker && worker.login || '').trim();
  if (!login) return { current: { totalSec: 0 }, prev: { totalSec: 0 } };

  const y = Number(year);
  const m = Number(month1to12);
  let py = y, pm = m - 1;
  if (pm <= 0) { pm = 12; py = y - 1; }

  const curSec = Worklog_sumMonthClosed_(login, y, m);
  const prevSec = Worklog_sumMonthClosed_(login, py, pm);

  return {
    current: { year: y, month: m, totalSec: curSec },
    prev: { year: py, month: pm, totalSec: prevSec }
  };
}
