/**
 * Router — single entry for API calls.
 */
function Router_handleApiRequest_(e) {
  try {
    // Load optional overrides from WEB_Config (if present)
    try { if (typeof Config_init_ === 'function') Config_init_(); } catch (_) {}
    const payload = parseJsonBody_(e);
    const path = String(payload.path || '').trim();
    if (!path) return jsonError_('BAD_REQUEST', 'Missing "path" in request body.');

    // Public routes
    if (path === 'auth/login') {
      const data = Auth_login_(payload);
      return jsonOk_(data);
    }

    // Authenticated routes
    const token = String(payload.token || '').trim();
    const authDeviceId = String(payload.deviceId || '').trim();
    const worker = Auth_requireWorker_(token, authDeviceId); // throws if invalid

    switch (path) {
      case 'auth/me': {
        // Enrich session with worker type (role) from Workers sheet if missing.
        if (!worker.type) {
          try { worker.type = Workers_getTypeByLogin_(worker.login); } catch (_) {}
        }
        return jsonOk_({ worker });
      }

      // Coordinator (audit logging)
      case 'coordinator/audit/log': {
        // RBAC: only allowed coordinator roles can write coordinator logs
        const role = normalizeRoleServer_(worker.type || '');
        const allowed = ['koordynator', 'audytor', 'kierownik', 'admin'];
        if (!allowed.includes(role)) {
          return jsonError_('FORBIDDEN', 'Brak dostępu do modułu.');
        }

        const r = payload.record || {};
        // Safety: limit meta size (avoid huge payloads)
        let meta = '';
        try {
          meta = r.metaJson ? String(r.metaJson) : '';
          if (meta.length > 8000) meta = meta.slice(0, 8000);
        } catch (_) {}

        CoordinatorLog_append_({
          createdAt: r.createdAt || nowIso_(),
          date: r.date || '',
          workerId: worker.workerId || r.workerId || '',
          role: worker.type || r.role || '',
          functionName: r.functionName || '',
          module: r.module || 'AUDIT',
          sessionId: r.sessionId || '',
          startAt: r.startAt || '',
          stopAt: r.stopAt || '',
          durationSec: r.durationSec || 0,
          auditDeductSec: r.auditDeductSec || 0,
          note: r.note || '',
          metaJson: meta
        });

        return jsonOk_({ ok: true });
      }
      // Workday
      case 'workday/utilityId': {
        const startRoomId = (typeof Zones_getStartRoomId_ === 'function') ? Zones_getStartRoomId_() : Zones_getUtilityRoomId_();
        const stopRules = (typeof Zones_getStopRules_ === 'function') ? Zones_getStopRules_() : [];
        // Backward compatibility: utilityRoomId previously meant START/STOP; now it is START only.
        return jsonOk_({ utilityRoomId: startRoomId, startRoomId: startRoomId, stopRules: stopRules });
      }
      case 'workday/active': {
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const chk = Workday_getActiveForTodayForClient_(worker.login, clientNow || nowIso_(), tzOffsetMinutes);
        return jsonOk_({ workday: (chk && chk.todayWorkday) ? chk.todayWorkday : null, staleOpen: !!(chk && chk.staleOpen), staleStartAt: (chk && chk.staleStartAt) ? chk.staleStartAt : '' });
      }
      case 'audit/workdayOpen': {
        // Check Backup_Log for open WORKDAY entry (START exists and not closed)
        const open = BackupLog_findOpenWorkdayCycleForWorker_(worker.login);
        return jsonOk_({ open: !!(open && open.cycleId), workday: open || null });
      }
case 'workday/todaySummary': {
  const data = Workday_getTodaySummary_(worker);
  return jsonOk_(data);
}
case 'workday/summaryMonth': {
  const year = Number(payload.year || 0);
  const month = Number(payload.month || 0);
  const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
  const data = WorkdaySummary_getMonth_(worker, year, month, tzOffsetMinutes);
  return jsonOk_(data);
}

case 'workday/summaryMonthDays': {
  const year = Number(payload.year || 0);
  const month = Number(payload.month || 0);
  const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
  const data = WorkdaySummary_getMonthDays_(worker, year, month, tzOffsetMinutes);
  return jsonOk_(data);
}


case 'workday/monthSummary': {
  const data = Workday_getMonthSummary_(worker, { monthsBack: 3 });
  return jsonOk_(data);
}

      case 'workday/eventsMonth': {
  const year = Number(payload.year || 0);
  const month = Number(payload.month || 0);
  const clientNow = String(payload.clientNow || '');
  const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
  const data = WorkdayEvents_getMonthDaysForWorker_(worker, year, month, clientNow || nowIso_(), tzOffsetMinutes);
  return jsonOk_(data);
}

case 'workday/start': {
        const roomId = String(payload.roomId || '').trim(); // must be utility room
        const deviceId = String(payload.deviceId || '').trim();
        const startEventId = String(payload.startEventId || '').trim();
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const result = Workday_start_(worker, { roomId, deviceId, startEventId, clientNow, tzOffsetMinutes });
        return jsonOk_(result);
      }
      case 'workday/beginEnding': {
        const roomId = String(payload.roomId || '').trim(); // must be utility room
        const deviceId = String(payload.deviceId || '').trim();
        const endEventId = String(payload.endEventId || '').trim();
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        // Guard: STOP endpoints must apply only when there is a START today (device time)
        const __clientNow = String(payload.clientNow || '');
        const __tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const chk = Workday_getActiveForTodayForClient_(worker.login, __clientNow || nowIso_(), __tzOffsetMinutes);
        if (!chk || !chk.todayWorkday) return jsonError_('NO_START_TODAY', 'Brak START w dniu dzisiejszym. Zeskanuj QR START, aby rozpocząć dzień pracy.');

const result = Workday_beginEnding_(worker, { roomId, deviceId, endEventId, clientNow, tzOffsetMinutes });
        return jsonOk_(result);
      }
      case 'workday/cancelEnding': {
        const deviceId = String(payload.deviceId || '').trim();
        const result = Workday_cancelEnding_(worker, { deviceId });
        return jsonOk_(result);
      }

      case 'workday/closeNow': {
        const deviceId = String(payload.deviceId || '').trim();
        const endEventId = String(payload.endEventId || '').trim();
        const clientAt = String(payload.clientAt || '').trim();
        const result = Workday_closeNow_(worker, { deviceId, endEventId, clientAt });
        return jsonOk_(result);
      }

      // Cleaning cycles
      case 'zones/resolve': {
        const roomId = String(payload.roomId || '').trim();
        const zone = QrResolver_resolveRoom_(roomId);
        return jsonOk_({ zone });
      }
      case 'cycles/get': {
        const cycleId = String(payload.cycleId || '').trim();
        const cycle = Cycles_getById_(cycleId);
        if (!cycle) return jsonError_('NOT_FOUND', 'Cycle not found.');
        if (cycle.workerLogin !== worker.login) return jsonError_('FORBIDDEN', 'Cycle belongs to another worker.');
        return jsonOk_({ cycle });
      }
      case 'cycles/active': {
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const cycle = (clientNow)
          ? Cycles_findActiveForWorkerForClient_(worker.login, clientNow, tzOffsetMinutes)
          : Cycles_findActiveForWorker_(worker.login);
        return jsonOk_({ cycle: cycle || null });
      }
      case 'cycles/start': {
        const roomId = String(payload.roomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const startEventId = String(payload.startEventId || '').trim();
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const result = Cycles_start_(worker, { roomId, deviceId, startEventId, clientNow, tzOffsetMinutes });
        return jsonOk_(result);
      }
      case 'cycles/stop': {
        const cycleId = String(payload.cycleId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const endEventId = String(payload.endEventId || '').trim();
        const reason = String(payload.reason || '').trim() || 'QR_SAME';
        const result = Cycles_stop_(worker, { cycleId, deviceId, reason, comment: String(payload.comment || '').trim().slice(0, 300), endEventId });
        return jsonOk_(result);
      }

      // --- Atomic endpoints (Stop+Start / Utility ENDING) ---
      case 'cycles/stopAndStart': {
        const cycleId = String(payload.cycleId || '').trim();
        const nextRoomId = String(payload.nextRoomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const opEventId = String(payload.opEventId || '').trim();
        const comment = String(payload.comment || '').trim().slice(0, 300);
        const reason = String(payload.reason || '').trim().slice(0, 40);
        const result = Cycles_stopAndStart_(worker, { cycleId, nextRoomId, deviceId, comment, opEventId, reason });
        return jsonOk_(result);
      }
      case 'workday/stopCycleAndBeginEnding': {
        const cycleId = String(payload.cycleId || '').trim();
        const roomId = String(payload.roomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const opEventId = String(payload.opEventId || '').trim();
        const comment = String(payload.comment || '').trim().slice(0, 300);
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const endEventId = String(payload.endEventId || '').trim();
        const workdayEndEventId = String(payload.workdayEndEventId || '').trim();
        // Guard: STOP endpoints must apply only when there is a START today (device time)
        const __clientNow = String(payload.clientNow || '');
        const __tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const chk = Workday_getActiveForTodayForClient_(worker.login, __clientNow || nowIso_(), __tzOffsetMinutes);
        if (!chk || !chk.todayWorkday) return jsonError_('NO_START_TODAY', 'Brak START w dniu dzisiejszym. Zeskanuj QR START, aby rozpocząć dzień pracy.');

const result = Workday_stopCycleAndBeginEnding_(worker, { cycleId, roomId, deviceId, reason: 'STOP_END_DAY', comment, endEventId, workdayEndEventId, clientNow, tzOffsetMinutes });
        return jsonOk_(result);
      }
      case 'workday/cancelEndingAndStartCycle': {
        const roomId = String(payload.roomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const opEventId = String(payload.opEventId || '').trim();
        const result = Workday_cancelEndingAndStartCycle_(worker, { roomId, deviceId, opEventId });
        return jsonOk_(result);
      }

      
      case 'workday/pauseStart': {
        const deviceId = String(payload.deviceId || '').trim();
        const pauseEventId = String(payload.pauseEventId || '').trim();
        const result = Workday_pauseStart_(worker, { deviceId, pauseEventId });
        return jsonOk_(result);
      }
      case 'workday/pauseStopAndStartCycle': {
        const roomId = String(payload.roomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const startEventId = String(payload.startEventId || '').trim();
        const result = Workday_pauseStopAndStartCycle_(worker, { roomId, deviceId, startEventId });
        return jsonOk_(result);
      }
      case 'workday/pauseStopAndBeginEnding': {
        const stopRoomId = String(payload.stopRoomId || payload.roomId || '').trim();
        const deviceId = String(payload.deviceId || '').trim();
        const endEventId = String(payload.endEventId || '').trim();
        const clientNow = String(payload.clientNow || '');
        const tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        // Guard: STOP endpoints must apply only when there is a START today (device time)
        const __clientNow = String(payload.clientNow || '');
        const __tzOffsetMinutes = Number(payload.tzOffsetMinutes || 0);
        const chk = Workday_getActiveForTodayForClient_(worker.login, __clientNow || nowIso_(), __tzOffsetMinutes);
        if (!chk || !chk.todayWorkday) return jsonError_('NO_START_TODAY', 'Brak START w dniu dzisiejszym. Zeskanuj QR START, aby rozpocząć dzień pracy.');

const result = Workday_pauseStopAndBeginEnding_(worker, { stopRoomId, deviceId, endEventId, clientNow, tzOffsetMinutes });
        return jsonOk_(result);
      }

// Worklog (CLEAN) — lists & summaries from sheetId=629711994
case 'worklog/monthDays': {
  const year = Number(payload.year || 0);
  const month = Number(payload.month || 0);
  const data = Worklog_getMonthDaysForWorker_(worker, year, month);
  return jsonOk_(data);
}
case 'worklog/summaryTwoMonths': {
  const year = Number(payload.year || 0);
  const month = Number(payload.month || 0);
  const data = Worklog_getSummaryTwoMonths_(worker, year, month);
  return jsonOk_(data);
}

// Grafik
case 'schedule/myWeek': {
  const data = Schedule_getMyWeekForWorker_(worker);
  return jsonOk_(data);
}

// === QR: Edycja QR strefy ===
case 'qr/clients': {
  const clients = QrEdit_getClientNames_();
  return jsonOk_({ clients });
}

case 'qr/get': {
  // RBAC: only allowed coordinator roles can read QR mapping
  const role = normalizeRoleServer_(worker.type || '');
  const allowed = ['koordynator', 'audytor', 'kierownik', 'admin'];
  if (!allowed.includes(role)) {
    return jsonError_('FORBIDDEN', 'Brak dostępu do modułu.');
  }

  const qrId = String(payload.qrId || '').trim();
  if (!qrId) return jsonError_('BAD_REQUEST', 'Missing qrId.');

  const record = QrEdit_getQrRecord_(qrId);
  const location = QrEdit_getZoneLocationByRoomId_(qrId);
  // record may be null (not found) — front-end decides what to do
  if (record) record.location = location;
  return jsonOk_({ record: record || null, location });
}
case 'qr/upsert': {
  // RBAC: only allowed coordinator roles can edit QR mapping
  const role = normalizeRoleServer_(worker.type || '');
  const allowed = ['koordynator', 'audytor', 'kierownik', 'admin'];
  if (!allowed.includes(role)) {
    return jsonError_('FORBIDDEN', 'Brak dostępu do modułu.');
  }


  const rec = payload.record || {};
  const qrId = String(rec.qrId || '').trim();
  if (!qrId) return jsonError_('BAD_REQUEST', 'Missing qrId.');

  // Business rule: QR must already exist in QR base (added by external app)
  const existing = QrEdit_getQrRecord_(qrId);
  if (!existing) return jsonError_('NOT_FOUND', 'Brak kodu QR w bazie.');


  const clientName = String(rec.clientName || '').trim();
  const placeName = String(rec.placeName || '').trim();
  const functionName = String(rec.functionName || '').trim();
  const location = String(rec.location || '').trim();
  const clientAt = String(rec.clientAt || '').trim(); // device time (AI-friendly string)

  const result = QrEdit_upsertQrRecord_({
    qrId,
    clientName,
    placeName,
    functionName,
    editedBy: String(worker.login || '').trim(),
    clientAt
  });

  // Save zone location (kolumna G w arkuszu Strefy)
  QrEdit_setZoneLocationByRoomId_(qrId, location);

  return jsonOk_(result);
}

      // Checklist (DAILY + EXTRA)
      case 'checklist/forQr': {
        const qrId = String(payload.qrId || payload.roomId || '').trim();
        const data = ChecklistService_getForQr_(worker, { qrId });
        return jsonOk_(data);
      }

      case 'checklist/submit': {
        const data = ChecklistService_submit_(worker, payload);
        return jsonOk_(data);
      }





default:
        return jsonError_('NOT_FOUND', `Unknown path: ${path}`);
    }
  } catch (err) {
    const code = (err && err.code) ? String(err.code) : 'SERVER_ERROR';
    const message = (err && err.message) ? String(err.message) : String(err);
    const details = (err && err.stack) ? { stack: String(err.stack) } : undefined;
    return jsonError_(code, message, details);
  }
}



/** Normalize role on server (RBAC) */
function normalizeRoleServer_(v) {
  return String(v || '').trim().toLowerCase();
}

// Alias (backward compatibility)
function Router_route_(e) { return Router_handleApiRequest_(e); }
