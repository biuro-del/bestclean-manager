<script>
window.BC = window.BC || {};
BC.ApiClient = (function() {
  function normalizeError(err) {
    if (!err) return new Error('Błąd API');
    if (err instanceof Error) return err;
    const e = new Error(err.message || String(err));
    e.details = err;
    return e;
  }
  function isAuthErrorCode(code) {
    const c = String(code || '').trim().toUpperCase();
    return ['SESSION_EXPIRED','SESSION_REVOKED','AUTH_MISSING_TOKEN','UNAUTHORIZED','FORBIDDEN'].includes(c);
  }

  function handleAuthFailure_(code) {
    try {
      if (window.BC && BC.Store && typeof BC.Store.logout === 'function') {
        BC.Store.logout();
      }
    } catch (_) {}
    // Hard redirect to login state (no token) — avoids partial UI states.
    try { location.reload(); } catch (_) {}
  }



  function callViaGoogleScriptRun(payload) {
    return new Promise((resolve, reject) => {
      try {
        google.script.run
          .withSuccessHandler((res) => {
            try {
              if (!res || !res.ok) {
                const msg = (res && res.error && res.error.message) ? res.error.message : 'Błąd API';
                const code = (res && res.error && res.error.code) ? res.error.code : 'API_ERROR';
                const err = new Error(msg);
                err.code = code;
                err.details = res && res.error && res.error.details;
                if (isAuthErrorCode(code) && payload && payload.path && String(payload.path) !== 'auth/login') {
                  handleAuthFailure_(code);
                  return;
                }
                reject(err);
                return;
              }
              resolve(res.data);
            } catch (e) { reject(e); }
          })
          .withFailureHandler((err) => reject(normalizeError(err)))
          .Api_post(payload);
      } catch (err) { reject(err); }
    });
  }

  async function callViaFetch(payload) {
    const res = await fetch(location.href, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });

    const text = await res.text();

    try {
      const json = JSON.parse(text);
      if (!json.ok) {
        const msg = (json.error && json.error.message) ? json.error.message : 'Błąd API';
        const code = (json.error && json.error.code) ? json.error.code : 'API_ERROR';
        const err = new Error(msg);
        err.code = code;
        err.details = json.error && json.error.details;
        if (isAuthErrorCode(code) && payload && payload.path && String(payload.path) !== 'auth/login') {
          handleAuthFailure_(code);
          return;
        }
        throw err;
      }
      return json.data;
    } catch (_) {
      const snippet = text.slice(0, 180).replace(/\s+/g, ' ').trim();
      const err = new Error('Serwer nie zwrócił JSON. Fragment: ' + snippet);
      err.code = 'NON_JSON_RESPONSE';
      throw err;
    }
  }

  async function post(path, data) {
    const payload = Object.assign({}, data || {}, { path });
    if (window.google && google.script && google.script.run && typeof google.script.run.Api_post === 'function') {
      return callViaGoogleScriptRun(payload);
    }
    return callViaFetch(payload);
  }

  return { post };
})();
</script>
