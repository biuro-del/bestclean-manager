    try {
      const d = new Date(Number(year), Number(month1to12) - 1, 1);

      const w = d.toLocaleDateString('pl-PL', { weekday: 'long' });
      return String(w || '').toUpperCase();
    } catch (_) { return ''; }
  }

      const d = new Date(String(iso));
      return d.toLocaleTimeString('pl-PL', { hour: '2-digit', minute: '2-digit' });
    } catch (_) { return ''; }

  function startWorkHudTimer_(timerId) {
    try { if (window.__workHudTick) clearInterval(window.__workHudTick); } catch (_) {}
    window.__workHudTick = null;

    const elT = document.getElementById(String(timerId || 'workHudTimer'));
    if (!elT) return;

    const wd = (window.BC && BC.Store && typeof BC.Store.getActiveWorkday === 'function') ? (BC.Store.getActiveWorkday() || {}) : {};
    const st = String(wd.status || '').toUpperCase();
    const startIso = String(wd.startAt || '').trim();
    const hasOpen = !!(wd.workdayId && (st === 'RUNNING' || st === 'ENDING' || isPaused_(wd)));

    function fmtHMS(sec){
      const s = Math.max(0, Math.floor(sec || 0));
      const hh = String(Math.floor(s / 3600)).padStart(2,'0');
      const mm = String(Math.floor((s % 3600) / 60)).padStart(2,'0');
      const ss = String(s % 60).padStart(2,'0');
      return hh + ':' + mm + ':' + ss;
    }
    function tick(){
      try{
        if (!hasOpen || !startIso) { elT.textContent = '--:--:--'; return; }
        const startMs = new Date(startIso).getTime();
        if (!isFinite(startMs)) { elT.textContent = '--:--:--'; return; }
        const sec = Math.max(0, Math.floor((Date.now() - startMs) / 1000));
        elT.textContent = fmtHMS(sec);
      }catch(_){}
    }

    tick();
    if (hasOpen && startIso) window.__workHudTick = setInterval(tick, 1000);
  }

  function escapeHtml(s) {

    return String(s || '')
      .replaceAll('&', '&amp;')
      .replaceAll('<', '&lt;')
      .replaceAll('>', '&gt;')
      .replaceAll('"', '&quot;')
      .replaceAll("'", '&#039;');
  }

  function fmtHMS(sec){
    const s = Math.max(0, Math.floor(sec || 0));
    const hh = String(Math.floor(s / 3600)).padStart(2,'0');
    const mm = String(Math.floor((s % 3600) / 60)).padStart(2,'0');
    const ss = String(s % 60).padStart(2,'0');
    return hh + ':' + mm + ':' + ss;
  }
  function tickTimer_(){
    try{
      const elT = document.getElementById('homeTimer');
      if (!elT) return;
      if (!hasOpen || !startIso) { elT.textContent = '--:--:--'; return; }
      const startMs = new Date(startIso).getTime();
      if (!isFinite(startMs)) { elT.textContent = '--:--:--'; return; }
      const sec = Math.max(0, Math.floor((Date.now() - startMs) / 1000));
      elT.textContent = fmtHMS(sec);
    }catch(_){}
  }

// In-app dialogs (avoid browser alert/confirm that show URL)
const dialog = document.getElementById('dialogModal');
const dialogTitle = document.getElementById('dialogTitle');
const dialogMsg = document.getElementById('dialogMessage');
const btnDlgOk = document.getElementById('btnDialogOk');
const btnDlgCancel = document.getElementById('btnDialogCancel');

let dialogResolver = null;

function closeDialog_(result) {
  if (dialog) dialog.classList.add('hidden');
  const r = dialogResolver;
  dialogResolver = null;
  if (typeof r === 'function') r(result);
}

function openDialog_({ title, message, okText, cancelText, showCancel }) {
  if (!dialog) return Promise.resolve(!!showCancel ? false : undefined);

  dialogTitle.textContent = title || 'Informacja';
  dialogMsg.textContent = String(message || '');
  btnDlgOk.textContent = okText || 'OK';
  btnDlgCancel.textContent = cancelText || 'Anuluj';

  if (showCancel) btnDlgCancel.classList.remove('hidden');
  else btnDlgCancel.classList.add('hidden');

  dialog.classList.remove('hidden');

  return new Promise(resolve => {
    dialogResolver = resolve;

    btnDlgOk.onclick = () => closeDialog_(true);
    btnDlgCancel.onclick = () => closeDialog_(false);
    // No header close button (only OK / Cancel)

    // clicking outside content closes (cancel)
    dialog.onclick = (e) => { if (e.target === dialog) closeDialog_(false); };
  });
}



function dialogHtml(title, html, okText) {
  // Renders provided HTML inside the dialog message container.
  if (!dialog) return Promise.resolve();
  dialogTitle.textContent = title || 'Informacja';
  dialogMsg.innerHTML = html || '';
  btnDlgOk.textContent = okText || 'OK';
  btnDlgCancel.classList.add('hidden');
  dialog.classList.remove('hidden');

  return new Promise((resolve) => {
    dialogResolver = resolve;
    btnDlgOk.onclick = () => closeDialog_(true);
    btnDlgCancel.onclick = () => closeDialog_(false);
    dialog.onclick = (e) => { if (e.target === dialog) closeDialog_(false); };
  });
}

function alertBox(message, opts) {
  const o = opts || {};
  return openDialog_({
    title: o.title || 'Informacja',
    message,
    okText: o.okText || 'OK',
    cancelText: '',
    showCancel: false
  }).then(() => undefined);
}

function confirmBox(message, opts) {
  const o = opts || {};
  return openDialog_({
    title: o.title || 'Potwierdź',
    message,
    okText: o.okText || 'OK',
    cancelText: o.cancelText || 'Anuluj',
    showCancel: true
  }).then(v => !!v);
}

  commentInput.placeholder = o.placeholder || 'Np. brak papieru, zalana podłoga...';

  const maxLen = Number(o.maxLen || 200);

  // Update UI max length dynamically (needed e.g. maxLen=300 for zlecenie indywidualne)
  try { commentInput.maxLength = maxLen; } catch (_) {}
  try {
    const maxEls = commentModal.querySelectorAll('.commentMax');
    maxEls.forEach(el => { el.textContent = String(maxLen); });
  } catch (_) {}

  function updateCounter_() {
    const v = String(commentInput.value || '');
    if (v.length > maxLen) commentInput.value = v.slice(0, maxLen);
    if (commentCounter) commentCounter.textContent = String(commentInput.value.length);
  }

  updateCounter_();
  commentInput.oninput = updateCounter_;

  commentModal.classList.remove('hidden');
  setTimeout(() => { try { commentInput.focus(); } catch (_) {} }, 0);

  return new Promise(resolve => {
    commentResolver = resolve;

    btnCommentSkip.onclick = () => closeCommentPrompt_('');
    btnCommentSave.onclick = () => closeCommentPrompt_(String(commentInput.value || '').slice(0, maxLen));
    commentModal.onclick = (e) => { if (e.target === commentModal) closeCommentPrompt_(''); };
  });
}


function setLoading(isOn, text) {
  const ov = document.getElementById('loadingOverlay');
  const tx = document.getElementById('loadingText');
  if (!ov) return;
  if (tx && text) tx.textContent = String(text);
  if (isOn) ov.classList.remove('hidden');
  else ov.classList.add('hidden');
}

  // =========================
  // Checklist (DAILY + EXTRA) — state, UI, validation
  // =========================
  const __CHECKLIST = { byCycle: {} };
  const __CL_PREFIX = 'BC_CHECKLIST_';

  function clKey_(cycleId) {
    return __CL_PREFIX + String(cycleId || '').trim();
  }

  function checklistEnsureCtx_(cycleId, qrId) {
    const cid = String(cycleId || '').trim();
    if (!cid) return null;
    if (!__CHECKLIST.byCycle[cid]) {
      __CHECKLIST.byCycle[cid] = {
        cycleId: cid,
        qrId: String(qrId || '').trim(),
        loading: true,
        dailyTasks: [],
        extraTasks: [],
        dailyState: {},
        extraState: {}
      };
      // Restore state from localStorage (best-effort)
      try {
        const raw = localStorage.getItem(clKey_(cid));
        if (raw) {
          const s = JSON.parse(raw);
          if (s && s.dailyState) __CHECKLIST.byCycle[cid].dailyState = s.dailyState;
          if (s && s.extraState) __CHECKLIST.byCycle[cid].extraState = s.extraState;
        }
      } catch (_) {}
    } else {
      try { if (qrId) __CHECKLIST.byCycle[cid].qrId = String(qrId || '').trim(); } catch (_) {}
    }
    return __CHECKLIST.byCycle[cid];
  }

  function checklistSave_(ctx) {
    if (!ctx || !ctx.cycleId) return;
    try {
      localStorage.setItem(clKey_(ctx.cycleId), JSON.stringify({
        dailyState: ctx.dailyState || {},
        extraState: ctx.extraState || {}
      }));
    } catch (_) {}
  }

  function checklistClear_(cycleId) {
    const cid = String(cycleId || '').trim();
    if (!cid) return;
    try { delete __CHECKLIST.byCycle[cid]; } catch (_) {}
    try { localStorage.removeItem(clKey_(cid)); } catch (_) {}
  }

  function checklistSetLoading_(cycleId, qrId) {
    const ctx = checklistEnsureCtx_(cycleId, qrId);
    if (!ctx) return;
    ctx.loading = true;
    try {
      const wrap = document.getElementById('checklistWrap');
      if (wrap) wrap.innerHTML = '<div class="muted">Wczytuję checklistę…</div>';
    } catch (_) {}
  }

  function checklistSetData_(params) {
    const p = params || {};
    const ctx = checklistEnsureCtx_(p.cycleId, p.qrId);
    if (!ctx) return;
    ctx.loading = false;
    ctx.dailyTasks = Array.isArray(p.dailyTasks) ? p.dailyTasks : [];
    ctx.extraTasks = Array.isArray(p.extraTasks) ? p.extraTasks : [];
    checklistRender_(ctx);
  }

  function checklistAskReason_() {
    return commentPrompt({
      title: 'Podaj powód (wymagane)',
      maxLen: 200,
      placeholder: 'Wpisz powód…'
    }).then((v) => String(v || '').trim());
  }

  function checklistSetStatus_(ctx, kind, id, status, reason) {
    const k = String(kind || '').trim();
    const key = String(id || '').trim();
    if (!ctx || !k || !key) return;

    const target = (k === 'extra') ? ctx.extraState : ctx.dailyState;
    if (!target[key]) target[key] = { status: '', reason: '' };

    target[key].status = String(status || '').trim();
    target[key].reason = (target[key].status === 'NOT_DONE') ? String(reason || '').trim() : '';

    checklistSave_(ctx);
    checklistRender_(ctx);
  }

  function checklistMissingRequired_(ctx) {
    if (!ctx) return [];
    if (ctx.loading) return ['LOADING'];

    const miss = [];
    const isReq = (t) => {
      const v = (t && (t.required !== undefined ? t.required : (t.Required || ''))) ;
      const s = String(v || '').toUpperCase();
      return (s === 'TAK' || v === true || v === 1);
    };

    (ctx.dailyTasks || []).forEach((t) => {
      if (!isReq(t)) return;
      const id = String(t.id || t.taskId || '').trim();
      const st = (ctx.dailyState && ctx.dailyState[id] && ctx.dailyState[id].status) ? String(ctx.dailyState[id].status) : '';
      if (!st) miss.push('daily:' + id);
    });

    (ctx.extraTasks || []).forEach((t) => {
      if (!isReq(t)) return;
      const id = String(t.id || t.extraId || '').trim();
      const st = (ctx.extraState && ctx.extraState[id] && ctx.extraState[id].status) ? String(ctx.extraState[id].status) : '';
      if (!st) miss.push('extra:' + id);
    });

    return miss;
  }

  function checklistUpdateStatusLine_(ctx) {
    try {
      const el = document.getElementById('clStatus');
      if (!el) return;

      const miss = checklistMissingRequired_(ctx);
      if (!miss.length) {
        el.textContent = 'Wszystkie wymagane pozycje uzupełnione.';
        return;
      }
      el.textContent = 'Brakuje: ' + String(miss.length) + ' wymag.';
    } catch (_) {}
  }

  function checklistRender_(ctx) {
    const wrap = document.getElementById('checklistWrap');
    if (!wrap) return;

    const daily = Array.isArray(ctx.dailyTasks) ? ctx.dailyTasks : [];
    const extra = Array.isArray(ctx.extraTasks) ? ctx.extraTasks : [];

    const sectionHtml = (title, items, kind) => {
      const list = (items || []).map((t) => {
        const id = String(t.id || t.taskId || t.extraId || '').trim();
        const name = String(t.name || t.taskName || '').trim();
        const required = String(t.required || t.Required || '').toUpperCase() === 'TAK' || t.required === true || t.required === 1;
        const stMap = (kind === 'daily') ? (ctx.dailyState || {}) : (ctx.extraState || {});
        const st = (stMap[id] && stMap[id].status) ? String(stMap[id].status) : '';
        const reason = (stMap[id] && stMap[id].reason) ? String(stMap[id].reason) : '';

        const doneActive = (st === 'DONE') ? ' is-active' : '';
        const notActive = (st === 'NOT_DONE') ? ' is-active' : '';
        const reqBadge = required ? '<span class="cl-req">WYM.</span>' : '';

        const reasonHtml = (st === 'NOT_DONE' && reason)
          ? '<div class="cl-reason muted">Powód: ' + escapeHtml(reason) + '</div>'
          : '';

        return `
          <div class="cl-item" data-kind="${escapeHtml(kind)}" data-id="${escapeHtml(id)}">
            <div class="cl-left">
              <div class="cl-name">${escapeHtml(name || '')}</div>
              ${reqBadge}
            </div>
            <div class="cl-actions">
              <button type="button" class="cl-btn cl-btn--done${doneActive}" data-act="done" aria-label="Wykonano">✓</button>
              <button type="button" class="cl-btn cl-btn--not${notActive}" data-act="not" aria-label="Nie wykonano">✕</button>
            </div>
            ${reasonHtml}
          </div>
        `;
      }).join('');

      if (!list) {
        return `<div class="cl-section"><div class="cl-sec-title">${escapeHtml(title)}</div><div class="muted">Brak pozycji.</div></div>`;
      }

      return `
        <div class="cl-section">
          <div class="cl-sec-title">${escapeHtml(title)}</div>
          ${list}
        </div>
      `;
    };

    const header = `
      <div class="cl-top">
        <div class="scan-title" style="margin:0">Checklista</div>
        <div id="clStatus" class="muted"></div>
      </div>
    `;

    wrap.innerHTML = header
      + sectionHtml('Codzienne', daily, 'daily')
      + sectionHtml('Uzupełniające', extra, 'extra');

    // Bind buttons
    const btns = wrap.querySelectorAll('.cl-btn');
    btns.forEach((b) => {
      b.onclick = async (e) => {
        try {
          const btn = e && e.currentTarget ? e.currentTarget : b;
          const item = btn.closest('.cl-item');
          if (!item) return;
          const kind = String(item.getAttribute('data-kind') || '').trim();
          const id = String(item.getAttribute('data-id') || '').trim();
          const act = String(btn.getAttribute('data-act') || '').trim();
          if (!kind || !id) return;

          if (act === 'done') {
            checklistSetStatus_(ctx, kind, id, 'DONE', '');
            return;
          }
          if (act === 'not') {
            const reason = await checklistAskReason_();
            const r = String(reason || '').trim();
            if (!r) return;
            checklistSetStatus_(ctx, kind, id, 'NOT_DONE', r);
            return;
          }
        } catch (_) {}
      };
    });

    checklistUpdateStatusLine_(ctx);
  }

  function checklistValidate_(cycleId) {
    const cid = String(cycleId || '').trim();
    const ctx = __CHECKLIST.byCycle[cid];
    if (!ctx) return { ok: true };
    if (ctx.loading) return { ok: false, message: 'Czekaj… wczytuję checklistę.' };

    const miss = checklistMissingRequired_(ctx);
    if (miss.length) {
      return { ok: false, message: 'Uzupełnij checklistę (wymagane pozycje).' };
    }
    return { ok: true };
  }

  function checklistGetSubmission_(cycleId) {
    const cid = String(cycleId || '').trim();
    const ctx = __CHECKLIST.byCycle[cid];
    if (!ctx) return null;
    if (ctx.loading) return null;

    const daily = (ctx.dailyTasks || []).map((t) => {
      const id = String(t.id || t.taskId || '').trim();
      const name = String(t.name || t.taskName || '').trim();
      const st = (ctx.dailyState && ctx.dailyState[id]) ? ctx.dailyState[id] : null;
      return st && st.status ? { taskId: id, taskName: name, status: st.status, reason: st.reason || '' } : null;
    }).filter(Boolean);

    const extra = (ctx.extraTasks || []).map((t) => {
      const id = String(t.id || t.extraId || '').trim();
      const name = String(t.name || t.taskName || '').trim();
      const st = (ctx.extraState && ctx.extraState[id]) ? ctx.extraState[id] : null;
      return st && st.status ? { extraId: id, taskName: name, status: st.status, reason: st.reason || '' } : null;
    }).filter(Boolean);

    const hasAnyTasks = (ctx.dailyTasks && ctx.dailyTasks.length) || (ctx.extraTasks && ctx.extraTasks.length);
    if (!hasAnyTasks) return { daily: [], extra: [] };

    return { daily, extra, qrId: ctx.qrId || '' };
  }

  return {
    setTopBackHandler,
    hideAllPublic,
    renderCoordinatorHome,
    renderCoordinatorAuditStart,
    renderCoordinatorAudit,
    renderCoordinatorEditQr,

    renderLogin,
    renderMenu,
    renderWorklogList,
    renderWorkSummary,
    renderWorkSummaryDetails,
    renderStartDay,
    renderScanRoom,
    renderPause,
    renderCleaning,
    renderEndDay,
    setElapsedText,
    setPauseElapsedText,
    setLeftText,
    setCleaningLocation,
    setLoading,
    alert: alertBox,
    dialogHtml,
    confirm: confirmBox,
    commentPrompt,

    // Checklist API
    checklistSetData: checklistSetData_,
    checklistValidate: checklistValidate_,
    checklistGetSubmission: checklistGetSubmission_,
    checklistClear: checklistClear_
  };
})();
